\begin{abstract}
For even $N$ tending to infinity, we deduce cancellation of $\mu(N-p)$
over primes and equal densities of its two nonzero signs from a
zero-free input and specified arguments of L\"u, Wang and Wu.
We extend the determinant reduction to both positive-argument families
$\mu(\varepsilon p+h)$, uniformly for $\varepsilon=\pm1$ and
$0<|h|\le x(\log x)^C$, with error
$O_C(\pi(x)/\sqrt{\log\log x})$ under the inputs
$\mathrm H1$ and $\mathrm H2'$ stated below.
We give the corresponding progression result for
$q\le(\log x)^{C_q}$ and a relative rough-value result for
$z\le\exp((\log x)^a)$, every fixed $0<a<1/2$, with even $h$.
All have Liouville counterparts. The small-shift and reflected
small-radical cases retain their weaker analytic inputs.
We also give radical criteria using a uniform real-zero gap or
classical exceptional-conductor avoidance.
\end{abstract}

\section{Introduction}\label{sec:intro}
All logarithms are natural. The M\"obius function is denoted by $\mu$,
and $\lambda(n)=(-1)^{\Omega(n)}$ is the Liouville function on positive
integers. Here $\Omega$ counts prime factors with multiplicity.
Write $\rad(n)=\prod_{p\mid n}p$ and let $\pi(x)$ count the primes at
most $x$. Put
\begin{equation}\label{eq:squarefree-product}
 A_{\GN}=\prod_{p\nmid\GN}\left(1-\frac1{p(p-1)}\right),\qquad
 A_h=\prod_{p\nmid h}\left(1-\frac1{p(p-1)}\right).
\end{equation}
These products are at least Artin's constant $0.3739\ldots$.

We specify the inputs to the statements. The assertion $\mathrm H2'$ is
that the arguments of LWW Sections~4.2--4.7, including the proofs of
Propositions~4.3--4.5 and Proposition~4.2, and Appendix~A,
(A.2)--(A.4), are correct as written
\LWW{pp.~17--39, 48--50}. The assertion $\mathrm H1$ is
\begin{equation}\label{eq:H1}
 L(s,\chi)\ne0\quad(\Re s>7/8)
 \quad\hbox{for every Dirichlet character $\chi$},
\end{equation}
with the principal pole at $s=1$ allowed. This is the Dirichlet
conclusion of \cite[Section~1, Theorem~1.1, p.~4]{OpenAIQ78}.
For $L=\log\GN$ and fixed $A_0$, write
$\mathrm H1^\star(C_{\mathrm{zf},1},C_{\mathrm{zf},2};A_1)$ at $\GN$
for the following assertion: for every $k\le L^{A_0}$ and every primitive
character $\chi^*$ of conductor $q^*\mid k\GN$ with $q^*>L^{A_1}$,
\begin{equation}\label{eq:Hstar}
 L(\sigma+it,\chi^*)\ne0\quad\left(
 \sigma\ge1-C_{\mathrm{zf},1}\frac{\log L}{L},\quad
 |t|\le L^{C_{\mathrm{zf},2}}\right).
\end{equation}
For every fixed choice of these constants, \eqref{eq:H1} implies
\eqref{eq:Hstar} at all sufficiently large $\GN$, since
$1-C_{\mathrm{zf},1}\log L/L>7/8$.

\begin{theoremG}[Goldbach shift]\label{thm:G}
Assume $\mathrm H2'$ and $\mathrm H1^\star$ at every sufficiently large
even $\GN$, with the fixed constants chosen in
Theorem~\ref{thm:energy} and Section~\ref{sec:assembly}.
Then, as $\GN\to\infty$ through even integers,
\begin{equation}\label{eq:G}
 \sum_{p<\GN}\mu(\GN-p)=o(\pi(\GN)),\qquad
 \#\{p<\GN:\mu(\GN-p)=\kappa\}
 =\left(\frac{A_{\GN}}2+o(1)\right)\pi(\GN)
 \quad(\kappa=\pm1).
\end{equation}
In particular these conclusions are deduced from $\mathrm H2'$ and
\eqref{eq:H1}. The errors are uniform in even $\GN$ and ineffective.
\end{theoremG}

\begin{theoremS}[Small radical]\label{thm:S}
Assume $\mathrm H2'$. For each fixed $C>0$, the conclusions
\eqref{eq:G} hold as $\GN\to\infty$ through even integers satisfying
$\rad(\GN)\le(\log\GN)^C$, using Iwaniec's region
\eqref{eq:Iwaniec} and the Burgess bound \eqref{eq:Burgess}.
The errors are uniform in this class for fixed $C$ and ineffective.
\end{theoremS}

\begin{theoremH}[Small shifts]\label{thm:H}
Assume the arguments of LWW as written: $\mathrm H2'$ and, in addition,
Sections~2.3, 3, 5 and 6 \LWW{pp.~6--14, 39--47}.
For each fixed $C>0$, uniformly in integers $0<|h|\le(\log x)^C$,
\begin{equation}\label{eq:H}
 \sum_{\substack{p\le x\\p+h\ge1}}\mu(p+h)=o(\pi(x)),\qquad
 \#\{p\le x:p+h\ge1,\ \mu(p+h)=\kappa\}
 =\left(\frac{A_h}2+o(1)\right)\pi(x)
 \quad(\kappa=\pm1).
\end{equation}
The errors are ineffective.
\end{theoremH}

\begin{theorem}[Liouville versions]\label{thm:lambda}
Under the respective hypotheses of Theorems~\ref{thm:G}, \ref{thm:S}
and~\ref{thm:H}, their conclusions hold with $\lambda$ in place of
$\mu$ and each sign density replaced by $1/2+o(1)$.
The domain $p+h\ge1$ is retained in the small-shift statement.
\end{theorem}

\paragraph{Affine notation and classical inputs.}
All statements below evaluate the arithmetic functions only on positive
integers. For $\varepsilon\in\{-1,1\}$ put
\begin{equation}\label{eq:eligible}
 \mathcal P_{\varepsilon,h}(x)=\{p\le x:\varepsilon p+h\ge1\},\qquad
 \Pi_{\varepsilon,h}(x)=|\mathcal P_{\varepsilon,h}(x)|,\qquad
 E(x)=(\log\log x)^{-1/2}.
\end{equation}
Here and below $p,r$ in products over primes are positive primes,
$P^-(1)=\infty$, and the theorem range parameters $C,C_q,a$ are fixed
before $x\to\infty$. Band locations vary only within the uniform families
of Section~\ref{subsec:band-uniform}.
In addition to the hypotheses specified in each theorem, we use the
prime number theorem, Siegel--Walfisz, Bombieri--Vinogradov for primes
and for M\"obius on reduced classes with endpoint maxima,
Montgomery--Vaughan mean values, fixed exponent-pair bounds for
logarithmic phases, and the dyadic M\"obius estimate of
\cite[Lemma~3.9(ii)]{MSTT}. Their uses, interval extensions and
Liouville consequences are given in the proofs.

\begin{theoremA}[Uniform affine shifts]\label{thm:affine}
Assume $\mathrm H1$ and $\mathrm H2'$. For each fixed $C\ge0$,
uniformly for $\varepsilon=\pm1$ and integers
$0<|h|\le x(\log x)^C$, and for $f\in\{\mu,\lambda\}$,
\begin{equation}\label{eq:affine-main}
 \sum_{p\in\mathcal P_{\varepsilon,h}(x)}f(\varepsilon p+h)
       =O_C(\pi(x)E(x)).
\end{equation}
For either $\kappa=\pm1$ the M\"obius sign main term is
$A_h\Pi_{\varepsilon,h}(x)/2$, and the Liouville sign main term is
$\Pi_{\varepsilon,h}(x)/2$, with the same error as in
\eqref{eq:affine-main}.
\end{theoremA}

For the progression statement let $\Pi_{q,a_0;\varepsilon,h}(x)$
count the primes in \eqref{eq:eligible} with $p\equiv a_0\pmod q$.
Define
\begin{equation}\label{eq:progression-factor}
\begin{split}
 \mathfrak A(q,a_0;\varepsilon,h)
 ={}&\prod_{r\nmid qh}\left(1-\frac1{r(r-1)}\right)
      \prod_{\substack{r\Vert q\\r\mid\varepsilon a_0+h}}
                    \left(1-\frac1r\right)\\
 &{}\cdot\prod_{\substack{r^2\mid q\\r^2\mid\varepsilon a_0+h}}0.
\end{split}
\end{equation}
The last product sets the value to zero when a square is forced;
an empty product is one.
\begin{theoremB}[Progressions]\label{thm:progression}
Under the hypotheses of Theorem~\ref{thm:affine}, for each fixed
$C_q\ge0$, uniformly for $q\le(\log x)^{C_q}$ and $(a_0,q)=1$,
\begin{equation}\label{eq:progression-main}
 \sum_{\substack{p\in\mathcal P_{\varepsilon,h}(x)\\p\equiv a_0\ (q)}}
 f(\varepsilon p+h)
 =O_{C,C_q}\left(\frac{\pi(x)E(x)}{\varphi(q)}\right).
\end{equation}
The sign main terms are
$\mathfrak A(q,a_0;\varepsilon,h)\Pi_{q,a_0;\varepsilon,h}(x)/2$
for M\"obius and $\Pi_{q,a_0;\varepsilon,h}(x)/2$ for Liouville,
with the error in \eqref{eq:progression-main}.
\end{theoremB}

\begin{theoremC}[Explicit rate]\label{thm:rate}
In Theorems~\ref{thm:affine}, \ref{thm:progression} and
\ref{thm:rough} the displayed function
$E(x)=(\log\log x)^{-1/2}$ is valid with fixed $K$, fixed
positive gaps between bands in each weight, and fixed analytic
precision. The same rate holds for the weaker-input A/D branches
specified below. In particular it gives a rate for G and S on their
respective classes and for the shift range of H.
The constants and thresholds may be ineffective.
\end{theoremC}

For nonzero even $h$ and $z\ge2$ define
\begin{equation}\label{eq:rough-factors}
 \begin{split}
 V_h(z)&=\prod_{\substack{r\le z\\r\nmid h}}
                       \left(1-\frac1{r-1}\right),\\
 U_h(z)&=V_h(z)\prod_{\substack{r>z\\r\nmid h}}
                       \left(1-\frac1{r(r-1)}\right).
 \end{split}
\end{equation}
\begin{theoremD}[Rough affine values]\label{thm:rough}
Assume $\mathrm H1$ and $\mathrm H2'$. For fixed $C\ge0$ and
$0<a<1/2$, uniformly for nonzero even $h$ in the A range,
$\varepsilon=\pm1$ and $2\le z\le\exp((\log x)^a)$,
\begin{equation}\label{eq:rough-main}
 \sum_{p\in\mathcal P_{\varepsilon,h}(x)}
 f(\varepsilon p+h)\ind_{P^-(\varepsilon p+h)>z}
 =O_{a,C}(V_h(z)\pi(x)E(x)),\qquad f=\mu,\lambda.
\end{equation}
For each $\kappa=\pm1$, the corresponding sign main terms are
$U_h(z)\Pi_{\varepsilon,h}(x)/2$ and
$V_h(z)\Pi_{\varepsilon,h}(x)/2$, respectively, with the error in
\eqref{eq:rough-main}.
\end{theoremD}

The A and D errors are on the displayed ambient scales. They are
not relative errors in an arbitrarily short eligible interval.
For example $\varepsilon=1$, $h=-x$ and integral $x$ give an empty
set in \eqref{eq:eligible}. At fixed $z$ the tail product in $U_h(z)$
is retained. Uniformly in $h$,
$0\le1-U_h(z)/V_h(z)\ll1/z$.

\paragraph{Weaker-input branches.}
For each fixed $C_0>0$, A and D on $0<|h|\le(\log x)^{C_0}$
require $\mathrm H2'$ and the listed classical inputs alone.
For $\varepsilon=-1$, $x=h=\GN$ even, A and D instead require
$\mathrm H2'$ and the sufficiently strong fixed instance of
$\mathrm H1^\star$ selected by the reflected energy proof, together
with Burgess. On $\rad(\GN)\le(\log\GN)^{C_0}$, Iwaniec and
Burgess supply this branch without $\mathrm H1$.
The D branch always retains even $h$ and fixed $a<1/2$.
These assertions concern A and D; no progression refinement of
these weaker-input branches is asserted.

\begin{theoremF}[Radical criteria]\label{thm:F}
Fix $A_0,A_1,C_1,C_2>0$, interpreting $C_1,C_2$ as the constants
in \eqref{eq:Hstar}, and write $L=\log\GN$.
\begin{enumerate}
\item Suppose $\mathrm{LS}(c)$: for some fixed $c>0,q_0$, every
primitive real character of conductor $q\ge q_0$ and every real
zero $\beta\in(0,1)$ satisfy $1-\beta\ge c/\log q$.
For each prescribed nonnegative function $\eta(L)\to0$, Iwaniec's
region and $\mathrm{LS}(c)$ imply the selected $\mathrm H1^\star$
uniformly on
\begin{equation}\label{eq:F-envelope}
 \log\rad(\GN)\le\eta(L)L/\log L.
\end{equation}
The threshold may depend on $\eta$. In particular one may take
$\rad(\GN)\le\exp(L/(\log L)^2)$.
\item Let $c_{\rm LP}>0$ be a Landau--Page constant, choose
\begin{equation}\label{eq:F-kappa}
 0<\kappa\le\frac{\min(c_{\rm LP},1/(4\cdot10^4))}{32C_1},
 \quad \ell=\log X,\quad
 R_X=\exp(\kappa\ell/\log\ell),\quad Q_X=8(2\ell)^{A_0}R_X.
\end{equation}
Let $q_X$ be the conductor of the possible Landau--Page character
at $Q_X$. The selected $\mathrm H1^\star$ holds for all sufficiently
large even $X\le\GN\le2X$ with $\rad(\GN)\le R_X$ if the
exception is absent, if $q_X\le\ell^{A_1}$, or if
\begin{equation}\label{eq:F-avoid}
 q_X/\gcd(q_X,\GN)>(\log\GN)^{A_0}.
\end{equation}
This analytic criterion uses Iwaniec and Landau--Page.
\end{enumerate}
After the precision required by the reflected energy proof is fixed,
these criteria give reflected A and D under $\mathrm H2'$,
Burgess and the listed classical inputs, with $\mathrm{LS}(c)$
needed only in the first criterion. Choose $\kappa$ after that
precision is fixed. The analytic criteria themselves do not use
$\mathrm H2'$.
\end{theoremF}

Under $\mathrm H1$ and $\mathrm H2'$, A contains G and the shift
range of H; C quantifies both. The separate G, S, H and Liouville
statements above keep their original hypotheses. A correlation at
two distinct nonzero shifted arguments remains an open problem
for the present method.

\paragraph{Related averaged and fixed-shift statements.}
Lichtman \cite[Theorem~1.1]{Lichtman} states cancellation averaged over
positive shifts $h\le H<X$ when $\log H/\log\log X\to\infty$.
Lichtman and Ter\"av\"ainen \cite[Corollary~1.4]{LichtmanTeravainen}
state an almost-all-shifts result for
$(\log X)^{1+\epsilon}\le H\le X$.
Carella \cite[Theorem~1.1]{Carella} states, for a small fixed nonzero
shift $a$, a bound $O_{c,a}(x(\log x)^{-c})$ for every fixed $c>1$;
its Theorem~1.2 gives a two-shift statement under its
Hypothesis~3.1.
Tao and Ter\"av\"ainen \cite[Theorem~1.6]{TaoTeravainen}
state hybrid correlations at distinct fixed natural-number shifts, with
at most two von Mangoldt factors and a fixed number of Liouville factors,
on scales governed
by the conductor and quality of a hypothetical Siegel zero.
The real-zero bounds stated by Banks \cite{Banks} and Kerr
\cite{Kerr} motivate the separately named assumption
$\mathrm{LS}(c)$; here that assumption is used in exactly its
displayed form.
For integer summands, Mangerel \cite{Mangerel} states
$|\sum_{1\le n<N}\lambda(n)\lambda(N-n)|<N-1$ for all
sufficiently large $N$. The rough-summand statement in
\cite{GuoRGP} gives prescribed Liouville signs for two rough
integer summands. The prime-factor law in \cite{OpenAIPD}
states convergence of normalized logarithms of the prime factors
of $p-1$ to the Poisson--Dirichlet law with parameter one.
For ordinary short intervals, Matom\"aki, Pandey, Pliego,
Ter\"av\"ainen and Wang \cite{MPPTW} state M\"obius cancellation
at exponent $19/35+\epsilon$ and Gowers uniformity of the von
Mangoldt function minus its Cram\'er model at exponent
$3/5+\epsilon$.

\paragraph{Background.}\label{par:background}
In their preprint submitted on 14 October 2026, L\"u, Wang and Wu
\cite{LWW} state in Theorem~1 that
$\sum_{p\le x}\mu(p+2)=o(\pi(x))$, and in Theorem~2 give the sign
densities $A_2/2$ among all primes. Their abstract describes the
conjecture as ``dating back at least to Hildebrand's work in 1989'';
Section~1 attributes its appearance in print to Hildebrand's 1989
paper \LWW{p.~3}. Their Section~4.4 describes the construction of its
auxiliary operators as similar to Sections~4.1--4.3 of OpenAI's
\emph{The Poisson--Dirichlet law for prime predecessors}, dated
24 September 2026 \cite{OpenAIPD}; their Appendix~A says that its
root-counting proof follows the argument of that paper's Lemma~3.3,
with an explicit error for arbitrary squarefree $S$
\LWW{pp.~26, 47}.
The author's related papers, \emph{Shifted primes with a prescribed
parity of the number of prime factors} \cite{GuoSPP} and
\emph{Sums of two rough numbers with prescribed Liouville signs}
\cite{GuoRGP}, are dated 9 October 2026 in their repository entries.

\section{Setting and determinant reduction}\label{sec:setting}
The reflected subsections retain the version~1 notation: $\GN$
is the global bound and $x$ is a prime-dyad scale. Subsections
headed ``Changes for general $h$'' use global bound $x$ and
dyad scale $y$. In each case $L$ is the logarithm of the global
bound, and the dyadic logarithm differs by $O_B(\log L)$.
The normalized bases and later estimates are shared through the
explicit specialization $\Delta=H=\GN$.

\begin{definition}[Parameters and coefficients]\label{def:parameters}
For the Goldbach shift, $\GN$ is even and $L=\log\GN$.
Fix $B>2$, let $p\in(x,2x]$, and put
\begin{equation}\label{eq:ranges}
 \begin{gathered}
 \frac{\GN}{2L^B}\le x\le\frac{\GN}2,\quad
 Z=\exp(L^{3/4}),\quad U=V=x^{1/5},\\
 X=MN\asymp x,\quad M,N\ge x^{1/5}/2,\quad MN\le2x\le\GN.
 \end{gathered}
\end{equation}
The letter $N$ always denotes the length of the Vaughan variable $n$.
Write $R_Z(n)$ for the indicator that every prime divisor of $n$
exceeds $Z$. Fix $0<\rho<1$ and
\begin{equation}\label{eq:bands}
 \frac1{10}<a_1<\cdots<a_K<\theta<\frac12,\quad
 \mathcal P_i=\{p:\exp(L^{a_i})<p\le\exp(2L^{a_i}),\ p\nmid\GN\},
 \quad V_i=\sum_{p\in\mathcal P_i}\frac1p.
\end{equation}
Put $\Aset=\{p_1\cdots p_K:p_i\in\mathcal P_i\}$,
$\omega_i(n)=\#\{p\in\mathcal P_i:p\mid n\}$ and
$\omega(n)=\sum_i\omega_i(n)$. The weight of LWW (2.13), p.~7, is
\begin{equation}\label{eq:weight}
 W(n)=\prod_{i=1}^K\frac{\rho^{\omega_i(n)-1}\omega_i(n)}{V_i}.
\end{equation}
With $j_*,r_*$ fixed in Section~\ref{subsec:retune}, let $J=\lfloor L^{j_*}\rfloor$, $R=\lceil L^{r_*}/2\rceil$,
and let $d_0,D$ have the meanings of LWW Section~4.2, pp.~17--21.
The real smooth dyadic cutoff $\eta$ is supported on $[1,4]$
(LWW Section~4.1, p.~15), so its product scale $Y$ has
$a,b\in[Y,4Y]\subset[Y,16Y]$. Define
\begin{equation}\label{eq:rectangle}
 \Omega=[d_0YM,16d_0YM]\times[N,2N].
\end{equation}
For $|v|\le L^{2B}$, the coefficients are
\begin{equation}\label{eq:coefficients}
 \begin{split}
 \alpha_m&=\mu(m)R_Z(m)\ind_{(m,\GN)=1}m^{iv},\\
 \beta_n&=c_V(n)R_Z(n)\ind_{(n,\GN)=1}n^{iv},\qquad
 c_V(n)=\sum_{\substack{\ell\mid n\\\ell>V}}\Lambda(\ell).
 \end{split}
\end{equation}
Thus $|\alpha_m|\le1$, $|\beta_n|\le L$, and the untwisted
coefficient $c_V(n)R_Z(n)\ind_{(n,\GN)=1}$ is nonnegative.
For a character $\xi$ of modulus dividing $k\GN$, put
\begin{equation}\label{eq:polynomials}
 \begin{split}
 B_\xi(t)&=\frac1{Y\prod_i V_i}
    \sum_{b\in\Aset}\eta(b/Y)e(-\lambda_0b/Y)\xi(b)b^{it},\\
 M_\xi(t)&=\frac1M\sum_{M<m\le2M}\alpha_m\xi(m)m^{it},\qquad
 N_\xi(t)=\frac1N\sum_{N<n\le2N}\beta_n\xi(n)n^{it}.
 \end{split}
\end{equation}
Here $e(t)=\exp(2\pi it)$, $k\le L^{A_0}$,
$|\lambda_0|\le2L^{A_0}$, $T_0=\lceil L^{B_1}\rceil$, and
$|B_\xi(t)|\le C_B=16\|\eta\|_\infty/\prod_i V_i$.
We write the arc displacement as $\lambda_0$ to distinguish it from
the Liouville function.
The major-arc target, at a fixed requested precision $A$, is
\begin{equation}\tag{$\alpha^*$}\label{eq:alpha-star}
 \frac1{\ph(\GN)}\sum_{\chi\bmod\GN}
 \int_{|t|\le T_0}|B_{\chi\psi_2}(t)M_{\chi\psi}(t)N_{\chi\psi}(t)|^2\,dt
 \ll_A\frac{L^{-A}}{\GN}.
\end{equation}
\stepcounter{equation}
The fixed twists $\psi,\psi_2$ are those constructed in
Section~\ref{subsec:arc-reduction}. The trivial bound is
$L^{O(1)}/\GN$; $1/\GN$ is the normalization.
\end{definition}

\begin{remark}[Ranges]\label{rem:ranges}
Primes $p\le\GN/L^B$ contribute $O(\GN/L^B)$ to the bounded target
sum, and $O(\GN L^{1-B})$ with $\log p$. The rest splits into
$O(\log L)$ dyadic intervals in \eqref{eq:ranges}.
Before separating the product cutoff,
$\GN-mn\in[\GN-2x,\GN-x)\cap(0,\GN)$.
For fixed $a\asymp Y$ the positive cofactor belongs to
\begin{equation}\label{eq:fixed-cofactor}
 \left[\frac{\GN-2x}{a},\frac{\GN-x}{a}\right],
\end{equation}
an interval of length $x/a\asymp X/Y$. As $a$ varies, its centre
varies, and the common positive support has $O(\GN/Y)$ integers.
Terms with $0<\GN-mn<Y$ vanish, since $W$ requires a factor from
every band. No dyadic condition on $\GN-p$ is needed.
\end{remark}

\begin{proposition}[Sifted Vaughan reduction]\label{prop:vaughan}
Let $y\in[x/(\log x)^B,x]$, $U=V=y^{1/5}$, and
$Z=\exp((\log x)^\gamma)$ for fixed $\gamma<1$.
For any bounded function $F$ on the positive integers set
\begin{align}
 T_1(F)&=\sum_{d\le U}\mu(d)R_Z(d)
       \sum_{y<dk\le2y}R_Z(k)(\log k)F(dk),\label{eq:T1}\\
 T_2(F)&=\sum_{\substack{d\le U\\\ell\le V}}
       \mu(d)\Lambda(\ell)R_Z(d\ell)
       \sum_{y<d\ell k\le2y}R_Z(k)F(d\ell k),\label{eq:T2}\\
 T_3(F)&=\sum_{\substack{m>U,n>V\\y<mn\le2y}}
       \mu(m)R_Z(m)c_V(n)R_Z(n)F(mn),\quad
       c_V(n)=\sum_{\substack{\ell\mid n\\\ell>V}}\Lambda(\ell).
       \label{eq:T3}
\end{align}
Then $0\le c_V(n)\le\log n$ and
\begin{equation}\label{eq:sifted-vaughan}
 \sum_{y<p\le2y}(\log p)F(p)
 =T_1(F)-T_2(F)+T_3(F)
  +O\bigl(\|F\|_\infty[Z+\sqrt y(\log(2y))^2]\bigr).
\end{equation}
The same holds for a partial dyad by including its endpoint in $F$.
\end{proposition}
\begin{proof}
For Dirichlet convolution $*$, $\log=1*\Lambda$ and
$\mu*1=\delta_1$. Splitting at $U,V$ gives the exact identity
\begin{equation}\label{eq:vaughan-convolution}
 \Lambda=\Lambda_{\le V}+\mu_{\le U}*\log
    -\mu_{\le U}*\Lambda_{\le V}*1
    +\mu_{>U}*\Lambda_{>V}*1.
\end{equation}
The three convolution terms combine to
$\mu*1*\Lambda_{>V}=\Lambda_{>V}$.
On the dyad the first term vanishes. Multiplication by $R_Z(n)F(n)$
and complete multiplicativity of $R_Z$ yield $T_1-T_2+T_3$.
Deleting the sifting indicator from the von Mangoldt sum costs
$O(\|F\|_\infty Z)$: in $(y,2y]$ a fixed prime has at most one
power, and $\sum_{p\le Z}\log p\ll Z$. Deleting all higher
prime powers, including bases above $Z$, costs at most
$O(\|F\|_\infty\sqrt y(\log(2y))^2)$.

In $T_2$ the outer product $d\ell\le y^{2/5}$ has coefficient
bounded by $\log(d\ell)$. Both variables in $T_3$ exceed
$y^{1/5}$. Dyadic localization and smoothing therefore produce
\begin{equation}\label{eq:Trect}
 T=\sum_{m,n}\alpha_m\beta_n F(mn),\qquad
 M,N\ge y^{1/5}/2,\quad MN\asymp y,
 \quad |\alpha_m|\le1,\quad|\beta_n|\ll\log x.
\end{equation}
For a partial dyad $(y,t]$, write its bounded factor as
$F_0(n)\ind_{n\le t}$ and expose the full product cutoff
$\ind_{y<mn\le t}$ before the rectangular reduction; for a full
dyad take $t=2y$. Smooth this product cutoff as a function of $mn/y$
at relative width $\delta=(\log x)^{-B_s}$. The positivity endpoint
of the shifted target remains in $F_0$. On each rectangle choose
the shorter variable as the counting variable. An endpoint strip contains
$O(\sum_{m\asymp M}(\delta y/m+1))
 =O(\delta y+\sqrt y)$ pairs, with fixed logarithmic coefficient
losses. Mellin inversion of the smooth cutoff has a fixed power
of $\delta^{-1}$ as its $L^1$ norm; truncate at
$|v|\le\delta^{-2}$, raising the fixed integration-by-parts order
for any desired logarithmic precision. The coefficients acquire
only $m^{iv},n^{iv}$. The shifted target, its positivity endpoint
and its roughness multiplier are kept intact and are never
differentiated. The extra factor $\log k$ in $T_1$ is treated by
real-weight partial summation after its endpoint-uniform estimate.
\end{proof}

\begin{lemma}[Puncturing and unit support]\label{lem:puncturing}
The punctured bands satisfy $V_i=\log2+o(1)$, uniformly in $\GN$.
The restrictions $(m,\GN)=(n,\GN)=1$ in \eqref{eq:coefficients}
cost $O_A(XL^{-A})$ for every fixed $A$.
The remaining variables are units modulo $k\GN$ whenever $k\le L^{A_0}$.
\end{lemma}
\begin{proof}
The removed reciprocal mass is at most
\begin{equation}\label{eq:puncture-mass}
 \sum_{\substack{p\mid\GN\\p>\exp(L^{0.1})}}\frac1p
 \le\omega_{\rm all}(\GN)e^{-L^{0.1}}\le Le^{-L^{0.1}},
\end{equation}
where $\omega_{\rm all}$ counts all distinct prime divisors.
Thus LWW (2.14)--(2.15), p.~7, apply with the same limits.
A common prime of a sifted variable and $\GN$ exceeds $Z$.
The number of discarded pairs is bounded, up to an absolute constant, by
\begin{equation}\label{eq:unit-discard}
 \sum_{\substack{p\mid\GN\\p>Z}}\frac Xp
 \le\frac{X\omega_{\rm all}(\GN)}Z\le XLe^{-L^{3/4}}.
\end{equation}
The bound $|\alpha_m\beta_n|\ll L$ absorbs this into the claimed
error. The restrictions preserve the coefficient and sifting
hypotheses of Proposition~4.2. Every prime factor of $k$ is below
$Z$, so the sifted variables are also units modulo $k$.
Every list prime and every $a,b\in\Aset$ is a unit modulo $\GN$ by
puncturing, and modulo $k$ because the bands lie above $L^{A_0}$.
\end{proof}

\begin{proposition}[The cofactor square]\label{prop:cofactor-square}
Specialize the bounded function in Proposition~\ref{prop:vaughan} to
\[
 F(n)=\begin{cases}
 \mu(\GN-n)W(\GN-n),&1\le n<\GN,\\
 0,&\text{otherwise}.
 \end{cases}
\]
After a smooth dyadic partition $\eta(a/Y)$, the resulting
\eqref{eq:Trect} has the form
\begin{equation}\label{eq:expansion-mu}
 T=\frac{(-1)^K}{\prod_i V_i}
 \sum_{\ell_0\ge1}\mu(\ell_0)\rho^{\omega(\ell_0)}
 \sum_{\substack{p_i\in\mathcal P_i,m,n\\mn+a\ell_0=\GN}}
 \alpha_m\beta_n+O(XLe^{-L^{0.1}}).
\end{equation}
Its dyadic pieces satisfy
\begin{equation}\label{eq:cauchy-all}
 |T_Y|^2\ll\frac{\GN}{Y}Q_Y^+\le\frac{\GN}{Y}\QQ,
 \qquad
 \QQ=\sum_{\substack{\ell_0\in\Z\\\ell_0\ne0}}
 \left|\frac1{\prod_i V_i}
 \sum_{\substack{a\in\Aset,m,n\\\GN-mn=a\ell_0}}
 \eta(a/Y)\alpha_m\beta_n\right|^2\ge0.
\end{equation}
Here $Q_Y^+$ is the same square restricted to $\ell_0\ge1$.
All cofactors in $\QQ$ satisfy $|\ell_0|\ll\GN/Y$.
\end{proposition}
\begin{proof}
If $\GN-mn=a\ell_0$ and $(a,\ell_0)=1$, then
$\mu(a\ell_0)=(-1)^K\mu(\ell_0)$ and $\omega(a\ell_0)=K+\omega(\ell_0)$.
Also $(a,m)=1$: a common list prime would divide $\GN$.
When $(a,\ell_0)>1$, some list prime has square dividing $\GN-mn$.
For fixed $m$ this is one residue class of $n$ modulo that square,
and the count in LWW (4.3), p.~15, gives $O(XLe^{-L^{0.1}})$.
The coprimality restriction can be removed in the expansion at the
same cost; its new terms have that square-divisibility property.
Cauchy--Schwarz uses $|\mu(\ell_0)\rho^{\omega(\ell_0)}|\le1$ and the
common range $O(\GN/Y)$ from Remark~\ref{rem:ranges}.
Each square is nonnegative, proving $Q_Y^+\le\QQ$.
On a supported rectangle $mn\ll MN\le\GN$, whence
$|\GN-mn|\ll\GN$. The cofactor $0$ is impossible because
$mn=\GN$ contradicts $(mn,\GN)=1$ for $\GN>1$.
Only this upper-bound quadratic form introduces negative cofactors;
$F$ and the expansion of its target always use positive arguments.
\end{proof}

\begin{lemma}[Disjoint lists and determinant equation]\label{lem:determinant}
Terms with $(a,b)>1$ may be removed from $\QQ$ at a cost
$O_A(\GN YL^{-A})$. For $(a,b)=1$ the exact determinant equation is
\begin{equation}\label{eq:determinant}
 b(\GN-mn)=a(\GN-rs)
 \quad\Longleftrightarrow\quad
 t:=bmn-ars=\GN(b-a).
\end{equation}
Equality on the left is equality of the two integer cofactors,
with no positivity condition. In particular $\GN\mid t$ and
$|t|\ll\GN Y$, while $t'=t/\GN=b-a$ has size $O(Y)$.
\end{lemma}
\begin{proof}
The elementary divisor-square estimate proved in
\eqref{eq:affine-divisor-square}, specialized to $\Delta=\GN$, has the form
\begin{equation}\label{eq:divisor-square}
 \sum_{\ell_0\in I_\ell}\tau(\GN-\ell \ell_0)^2\ll(X/Y)L^3,\qquad |I_\ell|\ll X/Y
\end{equation}
where $I_\ell$ is the translated cofactor interval for fixed
$\ell\asymp Y$, with $\GN-\ell \ell_0$ in the positive dyadic
product range. The argument depends on
$d\mid\GN-\ell \ell_0\Rightarrow(d,\ell)\mid\GN$; since
$(\ell,\GN)=1$, it gives $(d,\ell)=1$, exactly the needed hypothesis.
For the all-integer square use the full interval $|\ell_0|\ll\GN/Y$
and $|\GN-\ell \ell_0|$ in its absolute counts. Its length costs at most
$\GN/X\ll L^B$ relative to the displayed local estimate.
The absolute estimates in Section~\ref{subsec:affine-transport}
and the common-list saving then give the
stated $\GN YL^{-A}$ after increasing the fixed precision.
Reducing \eqref{eq:determinant} modulo $a$ and $b$, and using
$(a,b)=1$, proves divisibility by $a,b$ of the corresponding
numerators and equality of their integer quotients.
\end{proof}

\subsection{Changes for general \texorpdfstring{$h$}{h}: the common square}\label{subsec:affine-setting}
For the general statements $x$ denotes the global prime bound,
$L=\log x$, and $y\ge x/L^B$ is a prime-dyad scale.
In \eqref{eq:Trect} put $X=MN\asymp y$ and
\begin{equation}\label{v2:2-1}
 \Delta=-\varepsilon h,\qquad H=|h|,\qquad W_0=X+H,
 \qquad H/X\ll L^{B+C},\qquad H/(d_0YX)=o(1).
\end{equation}
Use the bands and weight \eqref{eq:weight}, punctured at $p\mid h$.
The common retuned parameters are specified in
Section~\ref{subsec:retune}; their band ceiling is $\theta<1/2$.
The puncturing and unit-support proof above applies with
$\omega_{\rm all}(H)=O_C(L)$ and $Z=\exp(L^{3/4})$.
In particular the coefficient sequences remain sifted and bounded.
\begin{proposition}[Affine cofactor square]\label{prop:affine-square}
In \eqref{eq:Trect} take $F(n)=f(\varepsilon n+h)W(\varepsilon n+h)$ when $\varepsilon n+h\ge1$ and $F(n)=0$ otherwise, for $f=\mu$ or $f=\lambda$. On unit Vaughan support set $w=\varepsilon mn+h=\varepsilon(mn-\Delta)$ and require $w\ge1$. The selected-prime expansion gives a sum over $w=a\ell$, $\ell\ge1$, after the square-overlap error $O_{K,\rho}(XL\exp(-cL^{0.1}))$. After a smooth dyadic partition in $a$,


\begin{equation}
|T_Y|^2\ll (W_0/Y)\,Q_Y,
\end{equation}


where the nonnegative enlarged square is


\begin{equation}\label{v2:3-4}
Q_Y=\sum_{j\in\mathbb Z\setminus\{0\}}
 \left|\frac1{\prod_iV_i}\sum_{mn-\Delta=aj}
 \eta(a/Y)\alpha_m\beta_n\right|^2.
\end{equation}


For coprime $a,b$, equality of the integer cofactors is equivalent to


\begin{equation}\label{v2:3-5}
bmn-ars=\Delta(b-a).
\end{equation}


For D, define $P_h(z)=\prod_{r\le z,\ r\nmid h}r$ and $\mathcal R_{z,h}(n)=1_{(n,P_h(z))=1}$. If every selected list prime exceeds $z$, then


\begin{equation}\label{v2:3-6}
\mathcal R_{z,h}(a\ell)=\mathcal R_{z,h}(\ell).
\end{equation}


Multiplying the target by $\mathcal R_{z,h}$ leaves exactly the same upper-bound square \eqref{v2:3-4}.

\end{proposition}\begin{proof} Puncturing at $r\mid h$ changes a band's reciprocal mass by at most $O(L\exp(-L^{0.1}))$. Sifted variables sharing a prime with $h$ share one exceeding $Z$; counting their multiples costs $O(XL^2/Z)$, at every fixed logarithmic precision. Each list prime is coprime to $m$. Hence its square dividing $w$ selects one residue class of $n$, and $N\gg r^2$ allows the bound $O(N/r^2)$. The bounded-weight square-overlap calculation in the reflected proof applies, including its separate Liouville expansion; it uses no sign of the affine coefficient. Once overlaps are removed, the common factor is $f(\ell)\rho^{\omega(\ell)}$ of modulus at most one. The positive cofactors lie in $[1,O(W_0/Y)]$. Cauchy gives the square over cofactors of the permitted sign, and adding the other sign adds nonnegative terms, producing \eqref{v2:3-4}. A zero cofactor would mean $mn=\Delta$; for $H>1$ this contradicts unit support, and for $H=1$ the dyadic product range excludes it. The equation \eqref{v2:3-5}, reduced modulo $a,b$, forces both divisibilities because $(a,b)=1$; it then gives equality of integer quotients. Finally $(a,P_h(z))=1$ proves \eqref{v2:3-6}, so the extra roughness indicator stays in the common bounded cofactor factor. \end{proof}

For actual primes $p\nmid h$, the shifted value is coprime to $h$. Thus $\mathcal R_{z,h}(\varepsilon p+h)$ equals the full roughness indicator. The omitted primes dividing $h$ contribute $O(L)$ without logarithmic weight and $O(L^2)$ with it.

\section{Blocks and relative coordinates}\label{sec:blocks}
Endpoint positions, with the list data of LWW attached, are
\begin{equation}\label{eq:positions}
 P=(Dbm,s)^T,\qquad Q=(Dar,n)^T,\qquad
 \det(P,Q)=D(bmn-ars)=Dt.
\end{equation}
All endpoint coordinates, and $D,a,b$, are units modulo $\GN$.

\subsection{The major-arc congruence}\label{subsec:kernel}
Let $\Mone$ be LWW's union of arcs with denominators
$k\le L^{A_0}$ and width $2L^{A_0}/Y$
\LWW{Section~4.1, p.~16}. Define
\begin{equation}\label{eq:arc-kernel}
 \begin{split}
 \MN&=\{\theta\in\T:\GN\theta\in\Mone\},\\
 H_{\GN}(t;a,b)&=\psi_1(t/(\GN Y))
 \int_{\MN}e(\theta(t-\GN(b-a)))\,d\theta,\\
 \Qmaj&=\prod_iV_i^{-2}
 \sum_{a,b,m,n,r,s}\eta(a/Y)\eta(b/Y)
 \alpha_m\overline{\alpha_r}\beta_n\overline{\beta_s}
 H_{\GN}(bmn-ars;a,b).
 \end{split}
\end{equation}
The cutoff $\psi_1$ is the fixed smooth cutoff of LWW, equal to $1$
on $[-4,4]$.
\begin{lemma}[Covering map]\label{lem:cover}
For every integer $t$,
\begin{equation}\label{eq:cover}
 \int_{\MN}e(\theta(t-\GN(b-a)))\,d\theta
 =\ind_{\GN\mid t}\int_{\Mone}e(\phi(t/\GN-b+a))\,d\phi.
\end{equation}
In particular $|\MN|=|\Mone|\ll L^{3A_0}/Y$.
\end{lemma}
\begin{proof}
For a fundamental domain of $\Mone$, the change of variables gives
\[
 \int_{\MN}F(\theta)\,d\theta
 =\frac1{\GN}\sum_{j\bmod\GN}\int_{\Mone}F((\phi+j)/\GN)\,d\phi.
\]
For the integrand in \eqref{eq:cover},
\[
 F((\phi+j)/\GN)
 =e(\phi t/\GN)e(jt/\GN)e(-\phi(b-a))e(-j(b-a)).
\]
The last factor is $1$ and the $j$-sum is
$\GN\ind_{\GN\mid t}$. This also makes the identity independent of
the chosen representatives.
\end{proof}
The raw-minus-major kernel is
\begin{equation}\label{eq:K}
 \begin{split}
 K_{\GN}(P,Q)={}&\frac{d_0}{D}\eta(a/Y)\eta(b/Y)\psi_1(t/(\GN Y))\\
 &\cdot\left\{\ind_{t=\GN(b-a)}
 -\int_{\MN}e(\theta(t-\GN(b-a)))\,d\theta\right\}
 \rho^{(\omega(P_1)+\omega(Q_1)-2K(J+1))/2}.
 \end{split}
\end{equation}
Both terms vanish unless $\GN\mid t$. On that congruence the brace is
\begin{equation}\label{eq:reduced-brace}
 \ind_{t'=b-a}-\int_{\Mone}e(\phi(t'-b+a))\,d\phi,
 \qquad t'=t/\GN,
\end{equation}
with cutoff $\psi_1(t'/Y)$, the transformed kernel of LWW
Section~4.2, p.~18.

\subsection{Lines and their sizes}\label{subsec:lines}
\begin{definition}[Unit endpoints and labelled blocks]\label{def:blocks}
Let $\CC_{\GN}$ consist of $P\in\Omega$ primitive in $\Z^2$
with $(P_1P_2,\GN)=1$. For a unit $\varrho\bmod\GN$, put
\begin{equation}\label{eq:lattice}
 \Lambda_\varrho=\{(u,v)\in\Z^2:v\equiv\varrho u\pmod{\GN}\}.
\end{equation}
This has basis $(1,\varrho),(0,\GN)$ and index $\GN$.
The slope class of a unit-coordinate endpoint is
$\varrho(P)=P_2\overline{P_1}\bmod\GN$.
\end{definition}
\begin{lemma}[Block diagonality]\label{lem:blocks}
For unit-coordinate endpoints,
\begin{equation}\label{eq:disjoint-union}
 \GN\mid\det(P,Q)\ \Longleftrightarrow\ \varrho(P)=\varrho(Q),
 \qquad
 \CC_{\GN}=\coprod_{\varrho\in(\Z/\GN\Z)^\times}
      (\CC_{\GN}\cap\Lambda_\varrho).
\end{equation}
Thus the endpoint kernel is block diagonal. Each full lattice is
regarded as a labelled copy when the state space is enlarged.
\end{lemma}
\begin{proof}
Invert $P_1Q_1$ in the determinant congruence. Since $(D,\GN)=1$,
$\GN\mid t$ is equivalent to $\GN\mid\det(P,Q)$.
The unit restriction makes the slope class unique.
\end{proof}
\begin{lemma}[Block size]\label{lem:block-size}
The number of vectors in $\Omega\cap\Lambda_\varrho$ primitive in
$\Lambda_\varrho$ is
\begin{equation}\label{eq:block-size}
 O\left(1+\frac{d_0YMN}{\GN}\right).
\end{equation}
In the application $d_0YMN/\GN\to\infty$ and $MN\le\GN$, so this is
$O(d_0YMN/\GN)\le O(d_0Y)$, uniformly in $\varrho$.
\end{lemma}
\begin{proof}
If the block is occupied, choose a lattice-primitive $P_0$ in it.
For any $Q$ in the block,
\[
 \det(P_0,Q)\in\GN\Z,\qquad
 |\det(P_0,Q)|\le2\cdot16d_0YM\cdot2N
 =64d_0YMN\le64d_0Y\GN.
\]
There are $O(1+d_0YMN/\GN)$ possible determinants. For each, two
solutions differ by an element of
$\Q P_0\cap\Lambda_\varrho=\Z P_0$; their first coordinates permit
at most $16$ solutions. Here $\Q$ denotes the rational numbers.
In the application $\GN/(d_0YMN)\ll L^B/(d_0Y)\to0$.
\end{proof}
This partition supplies pieces of the size of LWW's slope intervals
\LWW{Section~4.6, p.~36}. Physical slope locality is replaced by
relative-coordinate bounds.

\subsection{A normalized basis}\label{subsec:basis}
Let $P=(u,v)^T\in\CC_{\GN}$, and choose $ud-vc=1$, $0\le c<u$.
Define
\begin{equation}\label{eq:basis}
 \begin{split}
 c'&\equiv\GN c\pmod u,\quad0\le c'<u,
 \qquad c'v\equiv-\GN\pmod u,\quad \beta_P=c'/u,\\
 d''&=(\GN+c'v)/u,\qquad P^*=(c',d'')^T,\qquad
 \gN(P)=\begin{pmatrix}u&c'\\v&d''\end{pmatrix},\quad\det\gN(P)=\GN.
 \end{split}
\end{equation}
Equivalently
\begin{equation}\label{eq:shear}
 P^*=\GN(c,d)^T-\lfloor\GN c/u\rfloor P,\qquad
 \gN(P)=gE_{\GN}U_0,\quad
 g\in\operatorname{SL}_2(\Z),\quad E_{\GN}=\operatorname{diag}(1,\GN),
\end{equation}
where $U_0$ is the corresponding unimodular shear.
\begin{lemma}[Relative coordinates]\label{lem:coordinates}
Writing $\varrho=\varrho(P)$ and $Q=\gN(P)z$, one has:
\begin{enumerate}
\item $\Lambda_\varrho=\Z P\oplus\Z P^*$.
\item With $\tauN(z)=z_1+\beta_Pz_2$,
\begin{equation}\label{eq:coordinates}
 \det(P,Q)=\GN z_2,\quad Q_1=u\tauN(z),\quad
 Q_2=v\tauN(z)+\frac{\GN z_2}{u}.
\end{equation}
\item $Q$ is primitive in $\Lambda_\varrho$ precisely when
$(z_1,z_2)=1$. It is primitive in $\Z^2$ precisely when also
$(Q_1,\GN)=1$.
\item For primitive $z$ and every prime $p\nmid\GN$,
\begin{equation}\label{eq:kernel-line}
 p\mid Q_1\quad\Longleftrightarrow\quad
 [z]_p=\ker(u,c')\in\PP^1(\mathbb F_p).
\end{equation}
\item For $Q\in\Omega$,
\begin{equation}\label{eq:z-bound}
 |z_2|\le\frac{64d_0YMN}{\GN}\le64d_0Y,\qquad
 |z_1|\le16+|z_2|\le2^7d_0Y.
\end{equation}
Conversely every integer $z$ gives $Q\in\Lambda_\varrho$.
\end{enumerate}
\end{lemma}
\begin{proof}
The two columns belong to $\Lambda_\varrho$ and have its determinant.
Formula~\eqref{eq:coordinates} follows from
$d''=\GN/u+\beta_Pv$.
The lattice bijection proves lattice-primitivity.
At $p\mid\GN$ the matrix has rank one, with unit-coordinate column
$P$, and $Q\equiv0\pmod p$ precisely when $p\mid Q_1$.
At $p\nmid\GN$ the matrix is invertible, so $Q\equiv0$ precisely
when $z\equiv0$. Also $(u,c')=1$, since
$c'v\equiv-\GN\pmod u$ and $(u,\GN)=1$.
This proves the kernel assertion for primitive $z$.
Finally use the determinant bound in Lemma~\ref{lem:block-size},
$Q_1/u\le16$ and $0\le\beta_P<1$.
\end{proof}
The continuous extension and its comparison with LWW are
\begin{equation}\label{eq:continuous}
 \gN(u,v,\beta)=\begin{pmatrix}u&\beta u\\v&\GN/u+\beta v\end{pmatrix},
 \quad\tauN(z)=z_1+\beta z_2,
 \quad
 g'(u,v,r)=\begin{pmatrix}u&-2ur\\(u-v)/2&r(v-u)+1/u\end{pmatrix}.
\end{equation}
Thus $-2r$ is replaced by $\beta$ and $1/u$ by $\GN/u$; both
entries are retained exactly. The ratio in LWW (4.26), p.~21,
becomes $\beta_P=\{\GN r_P\}$. Multiplication by $\GN$ preserves
Lebesgue measure on $\T$.

\subsection{Enlarged states and intrinsic ratios}\label{subsec:enlargement}
\begin{definition}[States and good-state ratios]\label{def:good-ratio}
The ambient position space is the disjoint union of labelled sets
\begin{equation}\label{eq:states}
 \mathcal S'_\varrho=
 \{(\varrho,Q):Q\in\Omega\cap\Lambda_\varrho,
                  \ Q\text{ primitive in }\Lambda_\varrho\},
\end{equation}
with LWW's lists attached. For such a state choose $W\in\Lambda_\varrho$
with $\det(Q,W)=\GN$ and put
\begin{equation}\label{eq:intrinsic-ratio}
 \beta_Q=\frac{W_1}{Q_1}\pmod1.
\end{equation}
The good-state tests of LWW (4.14)--(4.16), pp.~18--19, use this
ratio at every state.
\end{definition}
\begin{lemma}[Intrinsic ratio and endpoint compression]\label{lem:compression}
Definition~\ref{def:good-ratio} is independent of the complement and
of the chosen block basis. In relative coordinates,
\begin{equation}\label{eq:ratio-relative}
 \beta_Q\equiv\frac{\tauN(w)}{\tauN(z)}\pmod1
 \quad\bigl(Q=\gN(P)z, W=\gN(P)w, \det(z,w)=1\bigr).
\end{equation}
For a unit endpoint it equals $\{\GN r_Q\}$.
LWW's test that $z_1$ is odd is replaced by endpoint support
$(Q_1Q_2,\GN)=1$, which is not a parity statement about the basis used here.
The restricted-root trace in LWW Section~4.6 is valid in this space.
\end{lemma}
\begin{proof}
A lattice-primitive $Q$ has a determinant-$\GN$ complement; any two
such complements differ by an integer multiple of $Q$.
Two oriented bases differ by $\operatorname{SL}_2(\Z)$, proving
\eqref{eq:ratio-relative} and basis independence.
At a unit endpoint the normalized complement in \eqref{eq:basis}
gives $\{\GN r_Q\}$. A nonunit vector can belong to several full
lattices, giving a different labelled state in each. Such a vector
is not primitive in $\Z^2$, by Lemma~\ref{lem:coordinates}; the
endpoint vector vanishes there because its coefficients and lists
are supported on units. For example, $\GN=2$, $P=(3,1)$,
$P^*=(1,1)$ and $z=(0,1)$ give a unit endpoint with $z_1$ even.
Thus no assertion $z_1$ odd is made after this shear.

What the moment argument uses is a completion at a root, the
primitive-coordinate equivalence
\[
 D\mid\det(z,z')\ \Longleftrightarrow\ [z]_p=[z']_p
 \quad(p\mid D),
\]
for squarefree $D$, and the ratio at the current state.
All hold here. The diagonal operators $S,G$ preserve the endpoint
pairing under enlargement. For positive $B_0$, the matrix
\begin{equation}\label{eq:compressed-gram}
 \bigl(\langle u_P,B_0^R u_Q\rangle\bigr)
       _{P,Q\in\CC_{\GN}\cap\Lambda_\varrho}
 \succeq0,\qquad
 \lambda_{\max}\le d_\varrho:
 =\sum_{P\in\CC_{\GN}\cap\Lambda_\varrho}
        \langle u_P,B_0^R u_P\rangle
\end{equation}
is a principal submatrix of a Gram matrix.
Spectral H\"older in LWW (4.65), p.~36, is applied in the ambient
space before this restriction. The sum $\sum_\varrho d_\varrho$
is exactly the root average over $\Omega\cap\CC_{\GN}$ counted below.
No estimate for the full trace on added nonunit states is required.
The second good-state test in LWW (4.47)--(4.48), p.~30, is applied
at the source position $Q$, using $\beta_Q$.
For the exceptional-measure bound, multiplication by $\GN$ preserves
Haar measure; for the subsequent cube count we count $\beta$ directly.
\end{proof}

\subsection{Changes for general \texorpdfstring{$h$}{h}: signed bases and slope cells}
\label{subsec:affine-geometry}
The unit blocks now have modulus $H=|h|$, including the single
block $\Z^2$ when $H=1$. The determinant orientation is
$\Delta=-\varepsilon h$.
\begin{lemma}[Signed normalized basis]\label{lem:affine-basis}
Let $P=(u,v)^T$ have $u,v>1$, $(u,v)=(uv,H)=1$. Define
\begin{equation}\label{eq:affine-basis}
 0\le c<u,\quad cv\equiv-\Delta\pmod u,\quad
 d=(\Delta+cv)/u,\quad\beta=c/u,\quad
 g_\Delta(P)=\begin{pmatrix}u&c\\v&d\end{pmatrix}.
\end{equation}
It is an oriented basis of
$\Lambda_\varrho=\{(s,t):t\equiv\varrho s\pmod H\}$,
$\varrho=v\bar u\pmod H$, with determinant $\Delta$.
For $Q=g_\Delta(P)z$ and $\tau_\beta(z)=z_1+\beta z_2$,
\begin{equation}\label{v2:3-1}
 Q_1=u\tau_\beta(z),\qquad
 Q_2=v\tau_\beta(z)+\Delta z_2/u,\qquad
 \det(P,Q)=\Delta z_2.
\end{equation}
Lattice-primitivity is $(z_1,z_2)=1$; plane primitivity also
requires $(Q_1,H)=1$. For primitive $z$ and $r\nmid H$,
\begin{equation}\label{v2:3-2}
 r\mid Q_1\quad\Longleftrightarrow\quad
 [z]_r=\ker(u,c)\pmod r.
\end{equation}
\end{lemma}
\begin{proof}
The congruence defining $c$ gives $ud-vc=\Delta$ and
$(u,c)=1$. Both columns lie in the index-$H$ lattice, and their
index is $H$. This proves the basis assertion for either sign.
Matrix multiplication gives \eqref{v2:3-1}. At a prime outside
$H$ the matrix is invertible; at a prime dividing $H$ it has rank
one with a first column whose coordinates are units. The
primitivity and kernel proofs in Lemma~\ref{lem:coordinates}
therefore apply with these signed entries.
\end{proof}

For odd $u,v$, reduction modulo two gives
\begin{equation}\label{eq:affine-parity}
 Q_1,Q_2\text{ odd}\quad\Longleftrightarrow\quad
 \begin{cases}z_1\text{ odd},\ z_2\text{ even},&H\text{ odd},\\
 z_1+cz_2\text{ odd},&H\text{ even}.
 \end{cases}
\end{equation}
The even-$H$ case uses the labelled enlargement of
Section~\ref{subsec:enlargement} with unit endpoint support;
it does not impose a parity test on $z_1$.
For odd $H$ retain the odd-odd physical projection, equivalently
the first line of \eqref{eq:affine-parity}. Nonunit intermediate
positions remain labelled lattice-primitive states.
At every source state use a determinant-$\Delta$ lattice
complement to define its intrinsic ratio. For
$\det(z,w)=1$ this ratio and its derivative are
\begin{equation}\label{v2:4-4}
 \beta_Q=\frac{\tau_\beta(w)}{\tau_\beta(z)}\pmod1,
 \qquad\partial_\beta\frac{\tau_\beta(w)}{\tau_\beta(z)}
       =\frac1{\tau_\beta(z)^2}.
\end{equation}
Changing orientation changes the ratio's sign and preserves both
good-state tests' distances to integers. In particular both tests
are imposed at the current source, including added nonunit states.

\begin{lemma}[Uniform local cells]\label{lem:slope-cells}
Set $D_0=d_0Y$ and retain
$\Omega=[D_0M,16D_0M]\times[N,2N]$. In each labelled block
partition physical slopes into half-open intervals of width
$H/(D_0M^2)$. Each cell contains $O(D_0)$ lattice-primitive
positions, and an edge with determinant $O(HD_0)$ interacts with
only $O(1)$ neighboring cells. A path of at most $2R$ such edges
from a root has relative coordinates
\begin{equation}\label{v2:3-3}
 |(z_j)_2|\ll RD_0,\qquad |(z_j)_1|\ll1+RD_0.
\end{equation}
All constants are independent of $H$.
\end{lemma}
\begin{proof}
Fix a lattice-primitive $P$ in one cell. For another $Q$ in it,
$|\det(P,Q)|\le256HD_0$ and the determinant is in $H\Z$.
For each of its $O(D_0)$ possible values, differences of solutions
belong to $\Q P\cap\Lambda_\varrho=\Z P$; the fixed ratio of
first coordinates leaves $O(1)$ choices. This proves occupancy.
An edge changes slope by $O(H/(D_0M^2))$. Sum the slope changes
along the path and multiply by the two first coordinates to get
$|\det(P,P_j)|\ll RHD_0$. Divide by $\Delta$ in
\eqref{v2:3-1}, and use $\tau_\beta(z_j)\in[1/16,16]$ and
$0\le\beta<1$, to prove \eqref{v2:3-3}.
\end{proof}
This cell partition extends the version~1 whole-block partition
to every $H$. Its occupancy improves the earlier unnormalized
small-shift bound $O(|h|d_0Y)$ to $O(d_0Y)$.
The finite checks in Appendix~\ref{app:checks} include these identities
and the signed swapped complement below.

\section{Root counting}\label{sec:roots}
\begin{remark}[Changes in version 2: signed root count]
The root count below replaces the version~1 positive-determinant
Lemma R. It counts the normalized signed basis directly, includes
all three swapped-complement branches, and permits every fixed
subpower modulus exponent below one.
\end{remark}
\begin{lemma}[Lemma R, signed form]\label{lem:R}
Put $U_0=d_0YM$, $V_0=N$ and assume $U_0,V_0\ge x^\delta$, $0<H<U_0V_0$, $H\le xL^C$. Let $S$ be odd, squarefree, coprime to $H$, with $S\le\exp(C_0L^\sigma)$ for any fixed $\sigma<1$. For a matrix $g_0=\left(\begin{smallmatrix}u_0&c_0\\v_0&d'_0\end{smallmatrix}\right)$ modulo $S$ of determinant $\Delta$, and intervals $I_1\subset[1,16]$, $I_2\subset[1,2]$, $I_3\subset[0,1]$, the formula is


\begin{equation}\label{v2:4-1}
\begin{aligned}&\#\left\{(u,v):\begin{array}{l}
(u,v)=1,\ (uv,2H)=1,\ u/U_0\in I_1,\ v/V_0\in I_2,\\
\beta=c/u\in I_3,\ g_\Delta(u,v)\equiv g_0\pmod S
\end{array}\right\}\\
&\quad=\frac{c_{2H}U_0V_0|I_1||I_2||I_3|}{|\mathrm{SL}_2(\mathbb Z/S\mathbb Z)|}
+O(U_0V_0x^{-c\delta}),
\end{aligned}
\end{equation}


where


\begin{equation}
c_{2H}=\zeta(2)^{-1}\prod_{r\mid2H}\frac{r-1}{r+1}.
\end{equation}


The error is uniform in the signed $\Delta$, intervals, and residue class, with a threshold allowed to depend on the fixed $\sigma,C_0$. Prescribed first-row kernel lines at $r\mid S$ instead have main mass $c_{2H}\prod_{r\mid S}(r+1)^{-1}$ and a power-saving error with a possibly smaller fixed $c>0$.

\end{lemma}\begin{proof}\emph{The shorter-coordinate case.} Fix $u\equiv u_0\pmod S$, $(u,2H)=1$, and replace the Appendix A inverse graph by


\begin{equation}
\mathcal R_u=\{(c,v)\pmod{uS}:c\equiv c_0\ (S),\ v\equiv v_0\ (S),\ cv\equiv ud_0'-\Delta\ (uS)\}.
\end{equation}


The last right-hand side is a unit at every prime dividing $u$, including the primes shared by $u,S$, because $u_0d'_0-c_0v_0\equiv\Delta$ and $(S,H)=1$. Exactly as in the CRT calculation of LWW (A.3), with $S_1=(u,S)$, $S_2=S/S_1$, and $\gamma S_2\equiv1\pmod{uS_1}$, its exponential sum is bounded by


\begin{equation}
\frac1{S_1}\sum_{j\bmod S_1}
 |K_{uS_1}(\gamma r+ju,\gamma s(ud'_0-\Delta))|.
\end{equation}


Here
$K_m(r,s)=\sum_{y\bmod m,(y,m)=1}e((ry+s\bar y)/m)$
is the Kloosterman sum. The common-divisor factor is at most $S_1(r,s,u)$, independently of the size or sign of $\Delta$. The stated Appendix A estimates (A.2)–(A.4) therefore give the same discrepancy $O(u^{0.99}S^2)$ for every rectangle. Its total mass is


\begin{equation}
|\mathcal R_u|=u\prod_{p\mid u,\ p\nmid S}(1-1/p).
\end{equation}


Sieve $(v,2H)=1$ by $e\mid\operatorname{rad}(2H)$. Each such $e$ is coprime to $uS$; replacing $v$ by $ev'$ replaces the inverse-graph constant by a unit multiple and leaves the rectangle error unchanged. Period splitting gives a main term


\begin{equation}
\frac{V_0|I_2||I_3|}{eS^2}\prod_{p\mid u,p\nmid S}(1-1/p)
\end{equation}


and error $O((V_0/(uSe)+1)u^{0.99}S^2)$. Summing the signed divisor sieve multiplies the main term by $\prod_{p\mid2H}(1-1/p)$ and costs at most $2^{\omega_{\rm all}(2H)}=x^{o(1)}$ in the error.

For the $u$ average expand its Euler weight as $\sum_{\ell\mid u,(\ell,S)=1}\mu(\ell)/\ell$ and its unit condition by $d\mid\operatorname{rad}(2H)$. Counting the fixed residue modulo $S$ gives the factor $1/\operatorname{lcm}(\ell,d)$. Sum first over $d$:


\begin{equation}\label{v2:4-2}
\sum_{d\mid\operatorname{rad}(2H)}\frac{\mu(d)}{\operatorname{lcm}(\ell,d)}
=\begin{cases}0,&(\ell,2H)>1,\\
\ell^{-1}\prod_{p\mid2H}(1-1/p),&(\ell,2H)=1.
\end{cases}
\end{equation}


This proves the convergent tail bound $O(1/U_0)$ after the additional $\mu(\ell)/\ell$; it avoids an unnecessary divisor-count factor in the main term. The local Euler factors and $|\mathrm{SL}_2(\mathbb Z/S\mathbb Z)|=S^3\prod_{p\mid S}(1-p^{-2})$ give precisely $c_{2H}$ above. The summed rectangle error is


\begin{equation}
O\big(x^{o(1)}S^2(V_0U_0^{0.99}+U_0^{1.99})\big)
=O(U_0V_0x^{-c_0\delta}),
\end{equation}


for a fixed $c_0>0$, since $U_0\le V_0$ and $S=x^{o(1)}$. No comparison $H\asymp x$ was used; the polynomial upper bound for $H$ suffices for the divisor cost.

\emph{Swapping the coordinates.} If $V_0<U_0$, write $\epsilon_0=\Delta/(uv)$; it lies strictly between $-1$ and 1. Since $d/v=\beta+\epsilon_0$, let $k=\lfloor\beta+\epsilon_0\rfloor\in\{-1,0,1\}$. The normalized complement of $(v,u)^T$ with determinant $\Delta$ is


\begin{equation}\label{v2:4-3}
\widetilde c=(1+k)v-d,\qquad \widetilde d=(1+k)u-c,\qquad
\widetilde\beta=1+k-\beta-\epsilon_0.
\end{equation}


Here $0<\widetilde c<v$: an endpoint value would imply $v\mid\Delta$, excluded by $v>1$ and $(v,H)=1$. Direct multiplication gives $v\widetilde d-u\widetilde c=\Delta$. For each of the three $k$ branches use the corresponding fixed matrix residue and the interval


\begin{equation}
((1+k)-I_3-\Delta/(uv))\cap[0,1).
\end{equation}


Their total length is $|I_3|$, since they describe one reflected interval modulo 1. Partition $I_1\times I_2$ into relative cells of side $x^{-c_0\delta/4}$. On a cell $\Delta/(uv)$ varies by $O(x^{-c_0\delta/4})$, because its absolute value is less than one. Enlarge or contract the \emph{unclipped} interval by that amount before intersecting with $[0,1)$; this includes branches that are empty at a corner but become nonempty inside the cell. The shorter-coordinate discrepancy is uniform in interval length. Summing it over $O(x^{c_0\delta/2})$ cells leaves a power saving; the boundary main mass is $O(U_0V_0x^{-c_0\delta/4})$. This proves \eqref{v2:4-1}, with smaller $c$. The negative branch $k=-1$ is essential and is not present in the positive-Delta proof.

At a prime $p\mid S$, the determinant-Delta matrix coset has $p(p^2-1)$ elements. A prescribed first-row kernel admits $p-1$ nonzero first rows and $p$ second rows each, giving probability $1/(p+1)$. CRT multiplies the probabilities. Summing per-matrix errors costs at most $S^3=x^{o(1)}$, absorbed by decreasing $c$. This proves the line version. The factor at 2 is already in $c_{2H}$, so is not inserted again.

\end{proof}
For the reflected specialization $\Delta=H=\GN$ even, write
\begin{equation}\label{eq:cN}
 c_{\GN}=c_{2H}=\frac1{\zeta(2)}
           \prod_{r\mid\GN}\frac{r-1}{r+1}\le1.
\end{equation}
Thus \eqref{v2:4-1} also supplies the matrix and line counts used
in the reflected transport below. Only the upper bound on this
density is used.

\begin{lemma}[Reduced determinant and boundary geometry]\label{lem:transport-geometry}
For $Q=\gN(P)z$, $Q'=\gN(P)z'$ in one labelled block,
\begin{equation}\label{eq:reduced-det}
 \det(\gN(P)z,\gN(P)z')=\GN\det(z,z'),\qquad
 t'=\frac{\det(Q,Q')}{\GN D}=\frac{\det(z,z')}{D}.
\end{equation}
The kernel in relative coordinates is \eqref{eq:reduced-brace} with

\end{equation}

\begin{equation}\label{eq:boundaries}

\end{equation}
The second term is bounded by $64$ in absolute value on allowed

$\partial_{t_0}E_2=\tauN(z)\ge1/16$.
\end{lemma}
\begin{proof}
The determinant identities follow from $\det\gN=\GN$ and

Lemma~\ref{lem:coordinates} and the cutoff in \eqref{eq:K}.
In the last boundary formula $s\in[1,16]$, so the extra term and its
$s$ derivative are $O(1)$. The full boundary has
$\partial_\beta E_2=t_0z_2$, with
$|\partial_\beta E_2|\ll\exp(O_K(L^{j_*+\theta}))$. Transversality, rather than smallness of
this term, is what the cube count uses.

Here $M_R^{(\GN)}$ is LWW's $M_R$ with $g'$ replaced by $\gN$

\end{proposition}

A closed path from a root $P\in\CC_{\GN}$ has
$P_j=\gN(P)z_j$, with $z_j$ primitive,

$c_{\GN}\,d(u/(d_0YM))\,d(v/N)\,d\beta$, and each prescribed line

 \partial_\beta\frac{\tauN(w)}{\tauN(z)}

the boundary surfaces have a transverse partial derivative of modulus

\end{equation}
At $P=(Dbm,s)$, $Q=(Dar,n)$ this gives exactly

$|\Delta z_2/(D_0MN)|\ll RH/X\ll L^{B+C+r_*}$.
The good-state ratio has the transverse derivative
\eqref{v2:4-4}. The odd-$H$ parity projection depends only on
$z$. Partition normalized root space into cubes of side
$x^{-c\delta/8}$. Boundary volume is
$x^{-c\delta/8+o(1)}$, and the summed interval-uniform root errors
still save a fixed power after all retained residue and path
choices. The independent-line moment has density $c_{2H}\le1$.
The same reasoning with one edge also applies when goodness fails
at either endpoint. Section~\ref{subsec:positive-tail} proves the
new model-tail bound; Section~\ref{subsec:retune} checks its separate
arithmetic discrepancy budget at the retuned parameters.

\paragraph{Signed operators.}
The relative primitive box now has radius $O(RD_0)$, and its
physical cutoff is $\ind_{g_\Delta z\in\Omega}$. The shared-prime
lattice, intrinsic good-state tests, weighted adjoints, signed

\paragraph{Endpoint pairing.}
The coefficient $f(Db m,s)=\overline{\alpha_m}\beta_s$ gives
\eqref{eq:pairing-product}. Its norm and supremum satisfy
\[
 \|f\|\ll(D_0X)^{1/2}L^{C_6},\qquad \sup|f|\le L^{C_6},
\]
where the exponent is fixed independently of $K$.
The complete band-prime part of $P_1=Dbm$ is exactly $Db$,
because $m$ is sifted. Thus this coefficient depends on the

endpoint primitivity follows from $H\mid t$, $0<|t/\Delta|\ll Y$
and $Y=o(Z)$ by the argument above.

For a slope cell $j$, let $f_j$ be its endpoint vector and $g_j$
the vector on its boundedly many interacting cells. For positive
$B_0=\mathcal A\mathcal A^*$, the principal coefficient-space
Gram matrix on that cell has largest eigenvalue at most its trace
$d_j$. Cell occupancy gives
$\langle f_j,B_0^Rf_j\rangle\le CD_0\sup|f|^2d_j$.
Apply spectral H\"older in the ambient space first, and then
Cauchy and H\"older over the cells, to obtain
\begin{equation}\label{eq:cell-pairing}
 |\langle f,\mathcal A g\rangle|
 \ll\|g\|\|f\|^{1-1/R}
       \left(CD_0\sup|f|^2\sum_jd_j\right)^{1/(2R)}.
\end{equation}
Only actual unit endpoint roots enter $\sum_jd_j$, so no factor
for the number of labelled blocks occurs. Root averaging and
$\mathrm H2'$ give
$\sum_jd_j\ll D_0XL^{-2RE_0}$. The trace loss is
$D_0^{1/(2R)}=O(1)$ by Section~\ref{subsec:retune}.
This gives the pairing scale $D_0X$ with arbitrary fixed saving.

Here are the absolute estimates required for those operations.
For fixed $\ell\asymp Y$ coprime to $H$, put

Deleting a largest factor in the four-factor decomposition that
majorizes $\tau(k)^2$, the remaining product is at most

\begin{equation}\label{eq:affine-divisor-square}
 \sum_{j:\Delta+\ell j\asymp X}\tau(\Delta+\ell j)^2
 \ll (X/Y)L^3+X^{3/4}L^2\ll(X/Y)L^3.
\end{equation}
Cauchy for each product pair therefore bounds the absolute raw

 \ell_0\equiv ba^{-1}k_0\pmod H,\qquad
 |bk_0-a\ell_0|\ll HY.

For either fixed variable the other has $O(1)$ choices, in a

Use $2\tau(k_0)\tau(\ell_0)\le\tau(k_0)^2+\tau(\ell_0)^2$,
sum over the $O(Y^2)$ product pairs, and multiply by arc mass

\subsection{Changes in version 2: the positive active-prime remainder}\label{subsec:positive-tail}
The signed main moment still uses the memory and ghost operators
in $\mathrm H2'$. The following auxiliary positive estimate is
used for the model remainder in root averaging.
\begin{lemma}[Positive active-prime bound]\label{lem:positive}
Consider the relative-coordinate path expansion of a $2R$-edge root moment, with the same active lists, normalized list permutations, position restrictions and shared-prime restrictions as above. Keep the definition $V_i=\sum_{p\in\mathcal P_i}1/p$ and assume these masses are bounded above and below by fixed positive constants, $2KR\le L^2$, and the absolute geometric kernels admit $O(1)$ raw endpoints and $O(Y)$ comparison endpoints, the latter carrying $O(L^{3A_0}/Y)$. Let $\mathcal A_{\rm act}$ be the sum of absolute geometric path coefficients, including the list normalization, times the independent-line factors for primes that occur in active lists, while all factors for primes that never occur in a list are replaced by one. Then, uniformly in the real root variables and the permitted projections,


\begin{equation}\label{v2:9-1}
\mathcal A_{\rm act}\le L^{C_*R}
\end{equation}


for a fixed $C_*=C_*(K,A_0,\rho)$ and sufficiently large $L$.

\end{lemma}\begin{proof} For a prime $p$ in group $i$, let $A_p\subset\{0,\ldots,2R\}$ be its active time set, and let $\kappa_p$ count its consecutive active runs. Put $q_0=q_{2R}=\sqrt\rho$, $q_j=\rho$ for $0<j<2R$. If $A_p$ is nonempty, its line average vanishes unless all active positions have one common projective line $\ell_p$. In the consistent case its factor is exactly


\begin{equation}\label{v2:9-2}
\frac{V_i^{-\kappa_p}}{p+1}
\prod_{j\notin A_p}q_j^{1_{[z_j]_p=\ell_p}}.
\end{equation}


Indeed the first occurrence charges one uniform-line probability, and the physical list normalization charges one $1/V_i$ at the beginning of each active run. Shared entries continue a run without a new charge. This is also obtained directly from the local factor (4.37) of LWW before introducing $b'_p$.

Write $a_p=\min A_p$, $c_p=\max A_p$. Dropping the factors at $j<a_p$ and $j>c_p$ increases \eqref{v2:9-2}, because each belongs to $[0,1]$. All remaining inactive positions are internal, so their $q_j$ equals $\rho$. It remains to bound the model with prime factors


\begin{equation}\label{v2:9-3}
\frac{V_i^{-\kappa_p}}{p+1}
\prod_{\substack{a_p<j<c_p\\j\notin A_p}}
\rho^{1_{[z_j]_p=\ell_p}}.
\end{equation}


Define $\nu_i^+(p)=1/(V_i(p+1))$ and $\lambda_i^+(p,\ell)=\nu_i^+(p)$. Active lists have product measure $\nu_i^+$; symmetric memory tuples have measure $(\lambda_i^+)^{\otimes k}/k!$. Since $V_i=\sum_{p\in\mathcal P_i}1/p$, one has $\sum_p\nu_i^+(p)\le1$ exactly. Use the active-list leave/discard and recall/fresh operations with \emph{no ghost operators}. A fresh target entry has measure $\nu_i^+$; recalling a memory entry contributes $1/V_i$. At an edge multiply source memory hits and target memory hits by $\vartheta_+^{h_s+h_t}$, with $\vartheta_+=\sqrt\rho$. Use a nonnegative geometric upper kernel consisting of the absolute raw term plus the absolute comparison upper bound. Retain list permutations and all required shared-prime and complete-list restrictions.

First require all initial/fresh prime births to be distinct and the initial and final memory to be empty. A prime appearing in two separated active runs must remain in memory throughout the gap: discarding it would require a second fresh birth, and retaining it after its last run would leave final memory nonempty. Thus its history is forced. Each gap position receives two factors $\sqrt\rho$, one from each incident edge. The first active run costs $\nu_i^+(p)$, and each later run costs $1/V_i$, reproducing \eqref{v2:9-3}. Consistency of active lines is enforced by shared-prime conditions and by the line recorded in a recalled memory entry. There is no contribution from a never-active prime. Multiplicity factorials cancel the orderings of memory coordinates, so this is an identity of positive path sums.

Remove distinctness of births. All weights are nonnegative, so the sum increases. Starting empty, at most $K$ memory coordinates are appended per edge; therefore every canonical path has memory size at most $2KR\le L^2$. Restrict the enlarged space to this cap, which still contains every canonical path.

We now prove a two-sided absolute bound for each positive edge operator. With Schur weight $v_0^{k_1+\cdots+k_K}$, fixed $v_0>1$, an edge changes the exponent by at most $K$. There are at most $L^{2K}$ choices of recalled memory coordinates, and sums of fresh prime measures are at most one per group. The normalized list permutations have row and column sum one. For a fixed determinant, B\'ezout and the bounded $\tau$ ratio give $O(1)$ positions. There are $O(Y)$ comparison determinants, canceling the $1/Y$ comparison factor. Thus the weighted row sum is at most


\begin{equation}\label{v2:9-4}
C_{K,\rho}L^{2K}(1+L^{3A_0}).
\end{equation}


The adjoint coefficients follow from the complete factorial state measure. In one group let $S,I\in\{0,1\}$ indicate appending the departing prime and recalling the arriving prime. In the unordered memory description, each token type has measure $\nu(p)^n/n!$. If the source active prime is $p$ and the target is $q$, cross-edge exclusion gives $p\ne q$. Multiplication by source measure divided by target measure gives:

\begin{center}\small
\begin{longtable}{@{}>{\raggedright\arraybackslash}p{0.13\textwidth}>{\raggedright\arraybackslash}p{0.23\textwidth}>{\raggedright\arraybackslash}p{0.23\textwidth}>{\raggedright\arraybackslash}p{0.31\textwidth}@{}}
\toprule
$(S,I)$ & Forward coefficient & Adjoint coefficient & Reversed rule\\
\midrule
$(0,0)$ & $\nu(q)$ & $\nu(p)$ & $(0,0)$\\
$(1,0)$ & $\nu(q)$ & $n'_p$ & $(0,1)$\\
$(0,1)$ & $n_q/V_i$ & $\nu(p)/V_i$ & $(1,0)$\\
$(1,1)$ & $n_q/V_i$ & $n'_p/V_i$ & $(1,1)$\\
\bottomrule
\end{longtable}
\end{center}

Here $n_q$ and $n'_p$ are the relevant source/target anchored memory multiplicities. Equivalently the adjoint is the reversed rule times $V_i^{S-I}$. Shared active-list measures cancel. This calculation includes repeated token types and active/memory overlaps in the enlarged space. In every reverse creation the memory line is anchored at its position; there is no free factor $p+1$. Taking the product of these identities over groups, and using the reversed $O(1)/O(Y)$ geometric count, proves the same polynomial column bound as \eqref{v2:9-4}, up to a fixed constant. The cap and endpoint projections only decrease absolute sums. Weighted Schur gives the polynomial edge norm.

The vector $b$ at relative position $e_1$, with empty memory and arbitrary admissible active lists, has squared norm at most $\prod_i(\sum_p\nu_i^+(p))^{J+1}\le1$. Its positive matrix element across $2R$ such operators is bounded by the product of their norms, giving \eqref{v2:9-1}. The main signed-memory construction retains its original parameter $\vartheta^2>\sqrt\rho$; that condition concerns ghost operators, which are absent from this auxiliary positive bound. The geometric input was the elementary endpoint count; the arithmetic root average enters only in the later transfer. \end{proof}

\begin{lemma}[Bonferroni model remainder]\label{lem:bonferroni}
For damping Bonferroni depth $B_d$, the total independent-line main mass of the absolute remainder is at most


\begin{equation}\label{v2:9-5}
L^{C_*R}\frac{(C_KR)^{B_d}}{B_d!}.
\end{equation}


\end{lemma}\begin{proof} Fix a path and its active prime values first. For a never-active prime define


\begin{equation}
d_p(\ell)=1-\prod_{j=0}^{2R}q_j^{1_{[z_j]_p=\ell}}\in[0,1].
\end{equation}


Its mean is at most $(2R+1)/(p+1)$ by $1-\prod(1-u_j)\le\sum u_j$. Distinct inactive primes have independent lines, independently also of active-prime consistency. The mean of the elementary-symmetric Bonferroni error of degree $B_d$ is therefore at most $(C_KR)^{B_d}/B_d!$, uniformly in the path and its active values. Multiply by the exact active factors and absolute geometric coefficients, sum, and apply Lemma~\ref{lem:positive}. This uses independence only after the path and active values have been fixed. It does not assert a relative error for arithmetic roots; their discrepancy is treated separately. \end{proof}

\subsection{Retuned numerical parameters}\label{subsec:retune}
Fix $0<a<1/2$ and choose


\begin{equation}
\theta=\frac{\max(a,1/10)+1/2}{2},\qquad
j_*=\frac{1-2\theta}{10},\qquad
r_*=\theta+2j_*,\qquad b_*=\theta+3j_*.
\end{equation}


Replace the version~1 numerical parameters by


\begin{equation}\label{v2:9-6}
Z=\exp(L^{3/4}),\quad J=\lfloor L^{j_*}\rfloor,\quad
R=\lceil L^{r_*}/2\rceil,\quad
B_d=\lceil L^{b_*}\rceil,\qquad
\max(a,1/10)<a_1<\cdots<a_K<\theta.
\end{equation}


The positive margins are


\begin{equation}\label{v2:9-7}
r_*-j_*-\theta=j_*,\qquad
b_*-r_*=j_*,\qquad
1-(r_*+j_*+\theta)=1-(b_*+\theta)=7j_*>0.
\end{equation}


The changes to every affected use of the old numerical exponents are as follows.

\paragraph{1.} \emph{Vaughan and Type I:} $Z=x^{o(1)}$, $Z^{O(\log L)}=x^{o(1)}$, and every marking prime and selected product $a\asymp Y$ is below $Z$ eventually. The Type I small-divisor BV margin remains $19/40-2/5>0$. Finite target-presieve moduli are still $\exp(O(L^a\log L))=x^{o(1)}$. All factorial and Rankin arguments in Section~\ref{sec:typeI} retain only fixed logarithmic losses.
\paragraph{2.} \emph{Sifted M\"obius:} the smooth-divisor reciprocal tail in Lemma~\ref{input:sifted} becomes $O(L^{3/4}\exp(-c_\delta L^{1/4}))$, still smaller than any fixed logarithmic power. No other part of the small-conductor estimate changes.
\paragraph{3.} \emph{Paths and root moduli:} $\log(d_0Y)=O_K(L^{j_*+\theta})$; the crude absolute path/list/permutation coefficient count has logarithm $O_K(L^{r_*+j_*+\theta})$. Retained prime moduli have logarithm $O_K(L^{r_*+j_*+\theta}+L^{b_*+\theta})=O_K(L^{1-7j_*})$. Both are subpower. The Appendix A discrepancy and the signed root lemma are valid for any fixed modulus exponent below one, since every extra fixed power of $S$ is absorbed by the coordinate power saving. Boundary derivatives and counts remain $x^{o(1)}$.
\paragraph{4.} \emph{Root-model tail:} use \eqref{v2:9-5}, not the crude path count, for the model Bonferroni error. Stirling bounds its logarithm by $O(R\log L)-(b_*-r_*+o(1))L^{b_*}\log L$, which is smaller than $-2AR\log L$ for every fixed $A$ eventually. The arithmetic discrepancy uses the crude subpower coefficient count in item 3.
\paragraph{5.} \emph{Prime measures and signed memory:} $(J+1)R/p_{\min}\to0$, so $b'_p>0$, $s_i^{J+1}=1+o(1)$ and the original finite primewise identities are unchanged. The main signed model's memory cap remains $L^2$. Its truncation error has factor $2^{-L^2+O(R)}$ times $\exp(O_K(L^{1-7j_*}))$, hence remains negligible. The address count is polynomial in $L$, and the collision-rank threshold $\lceil C_0 2R\log L/L^{0.1}\rceil$ is $o(R)$. The high-/low-rank restoration argument therefore keeps the same saving.
\paragraph{6.} \emph{Signed edge norms:} the clean bound has $L^{\theta-A_0}$ in place of $L^{0.2-A_0}$, so $A_0>G+3$ still suffices. The dirty raw bound becomes $L^{1-j_* K/2}+L^{-A}$; choose fixed $K>2(G+3)/j_*$. Dirty comparison bounds retain exponential minimum-prime savings. The good-state powers of $Y$, the balanced-box powers, and the Vaughan exponent $1/5$ are not changed.
\paragraph{7.} \emph{Moment to pairing:} the trace factor has logarithm $O_K(L^{j_*+\theta-r_*})=O_K(L^{-j_*})$. Thus it is bounded. There are $O_K(L)$ auxiliary-product dyads as before. Goodness and padding collisions still have exponential savings, because $J$ is only a fixed power of $L$.
\paragraph{8.} \emph{Major arcs:} the short-prime contour proof works uniformly up to $\exp(O(L^\theta))$. The sparse-frequency count has logarithm $O(L^{0.9}\log L+L^\theta)$ and still satisfies the Gram estimates. In the reflected $\mathrm H1^\star$ interface, $a_K<\theta<1/2$ allows the $\log Y$ term to be absorbed into $L^{1-a_K}\log L$ in Lemma LS. Its Burgess/P\'olya–Vinogradov energy bounds do not depend on $J$ or the marking ceiling. Deleting prime powers with base at most $Z$ costs $O(ZL/V)$ after the long-prime cofactor sum, negligible since $V$ is a fixed positive power of the large variable.

The determinant estimates at \eqref{v2:9-6} hold under $\mathrm H2'$. The model-tail bound in item~4 follows from Lemmas~\ref{lem:positive}--\ref{lem:bonferroni}. The resulting roughness range is subpower.

The strict endpoint $a=1/2$ is not obtained. In this fixed-parameter scheme the trace condition requires $r_*>j_*+\theta$, while the active-prime modulus budget requires $r_*+j_*+\theta<1$. These already imply $2\theta+2j_*<1$. This is an obstruction to the displayed parameter scheme, not a claim about all methods.

\section{Major arcs}\label{sec:major}
\subsection{Reduction to the averaged energy}\label{subsec:arc-reduction}
\begin{proposition}[Major-arc reduction]\label{prop:arc-reduction}
For every fixed $D_*>0$, \eqref{eq:alpha-star} at a sufficiently
large fixed precision $A=D_*+O(A_0+B_1)$ implies
\begin{equation}\label{eq:major-bound}
 |\Qmaj|\ll_{D_*}\GN YL^{-D_*}.
\end{equation}
\end{proposition}
\begin{proof}
By Lemma~\ref{lem:cover}, on the six-variable unit support,
\begin{equation}\label{eq:character-congruence}
 \begin{split}
 H_{\GN}(bmn-ars;a,b)
 &=\psi_1(t'/Y)\ind_{bmn\equiv ars\ (\GN)}
      \int_{\Mone}e(\phi(t'-b+a))\,d\phi,\\
 t'&=(bmn-ars)/\GN,\qquad
 \ind_{bmn\equiv ars\ (\GN)}
 =\frac1{\ph(\GN)}\sum_{\chi\bmod\GN}\chi(bmn)\overline{\chi(ars)}.
 \end{split}
\end{equation}
Write $\phi=h_0/k+\lambda_0/Y$,
$(h_0,k)=1$, $k\le L^{A_0}$, $|\lambda_0|\le2L^{A_0}$,
and $d\phi=d\lambda_0/Y$. Its factor is
\begin{equation}\label{eq:arc-phases}
 e(h_0t'/k)e(-h_0(b-a)/k)
 e(\lambda_0t'/Y)e(-\lambda_0(b-a)/Y).
\end{equation}
The factors in $a,b$ modulo $k$ expand in characters, with mass
at most $k$ each, since $a,b$ are units modulo $k$.

For $e(h_0t'/k)$, put $u_1=bmn$, $u'_1=ars$.
Both are units modulo $k\GN$, and
\[
 \ind_{\GN\mid u_1-u'_1}e(h_0t'/k)
 =\ind_{\GN\mid u_1-u'_1}
    e(h_0u_1/(k\GN))e(-h_0u'_1/(k\GN)).
\]
Let
\begin{equation}\label{eq:fibre-group}
 H=\ker((\Z/k\GN\Z)^\times\longrightarrow(\Z/\GN\Z)^\times),
 \qquad |H|=k\prod_{\substack{p\mid k\\p\nmid\GN}}(1-1/p)\le k.
\end{equation}
The reduction is surjective. For each unit class $a'\bmod\GN$,
choose a unit representative $\widetilde a'\bmod k\GN$; its fibre
is $\widetilde a'H$. Expand
$w\mapsto e(h_0\widetilde a'w/(k\GN))$ in characters
$\eta\in\widehat H$, with coefficients $|c_{a',\eta}|\le1$.
Every such character extends to a character $\psi_\eta$ of the
finite abelian group $(\Z/k\GN\Z)^\times$, independently of $a'$.
Hence on that fibre
\begin{equation}\label{eq:fibre-expansion}
 e(h_0u_1/(k\GN))
 =\sum_{\eta\in\widehat H}
 c_{a',\eta}\psi_\eta(\widetilde a')^{-1}\psi_\eta(u_1).
\end{equation}
The same representative works on the $u'_1$ side. Cauchy gives
\begin{equation}\label{eq:fibre-cauchy}
 \sum_{a'}\left|\sum_\eta
 c_{a',\eta}\psi_\eta(\widetilde a')^{-1}X_{a',\eta}\right|^2
 \le k\sum_\eta\sum_{a'}|X_{a',\eta}|^2.
\end{equation}
Thus class-dependent coefficients disappear at fixed powers of
$k$, including when $(k,\GN)>1$. Together with the $a,b$ factors,
the character mass is at most $k^4$ (LWW use $k^6$).

For the smooth factor put
$w=u_1/(\GN Y)$, $w'=u'_1/(\GN Y)$ and
$\kappa_X=MN/\GN$. The dyadic variables satisfy
\begin{equation}\label{eq:natural-scale}
 w,w'\in[\kappa_X,16\kappa_X],\qquad
 \kappa_X\gg L^{-B},\quad\kappa_X\le1,
 \qquad
 \psi_1(t'/Y)e(\lambda_0t'/Y)
       =\psi_1(w-w')e(\lambda_0(w-w')).
\end{equation}
If $\kappa_X\le1/4$, the difference has absolute value at most
$4$ and $\psi_1=1$. Otherwise it varies on a bounded interval.
Set $z=\log(w'/\kappa_X)$ and $\zeta=\log(w/w')$.
With a fixed smooth cutoff $\varpi$ equal to $1$ on $[1,16]$,
the kernel is
\begin{equation}\label{eq:mellin-kernel}
 \varpi(e^z)\varpi(e^{z+\zeta})
 \psi_1(\kappa_Xe^z(e^\zeta-1))
 e(\lambda_0\kappa_Xe^z(e^\zeta-1)).
\end{equation}
Its support lies in a fixed box of side $O(1)$.
In the variables $w/\kappa_X,w'/\kappa_X$, derivatives are bounded
by fixed powers of $L^{A_0}$ because $\kappa_X\le1$.
Consequently for a fixed sufficiently large $B_0$ the transform obeys
\begin{equation}\label{eq:mellin-decay}
 F_{\lambda_0}(v,s)\ll_j
 (1+|v|/L^{B_0})^{-j}(1+|s|/L^{B_0})^{-j}.
\end{equation}
Taking $B_1$ sufficiently large leaves frequencies
$|t|,|s|\le L^{B_1}$. This is the Mellin separation of LWW
Section~5, pp.~42--43, at scale $1$ rather than $X$.
The cutoff here does not force a narrow product difference;
the arithmetic is in the character congruence.

With the normalized polynomials \eqref{eq:polynomials}, each product
sum has size $YMN\cdot BMN$ with its displayed normalization.
The $1/Y$ arc measure therefore gives the scale
\begin{equation}\label{eq:major-normalization}
 \frac{(YMN)^2}{Y}\,\frac1{\ph(\GN)}
       \sum_\chi\int|BMN|^2
 =Y(MN)^2\,\frac1{\ph(\GN)}\sum_\chi\int|BMN|^2.
\end{equation}
To obtain \eqref{eq:major-bound} it suffices that the energy be
$\ll L^{-D_*}\GN/(MN)^2$, which follows from
$L^{-A}/\GN$ because $MN\le\GN$.
Cauchy in $t,\chi$, and the dyadic tail estimates of LWW
Section~5, p.~43, leave \eqref{eq:alpha-star} with
$\xi=\chi\psi$ on $m,n$ and $\xi_b=\chi\psi_2$ on $b$,
where $\psi=\psi_\eta$, $\psi_2=\psi\psi'_b$ and
$\psi'_b$ is modulo $k$. The primed product uses
$-\lambda_0,\overline\alpha,\overline\beta$; the estimates below
also hold after these conjugations and the corresponding sign
change of the frequency twist.
The arc count $L^{2A_0}$, character mass $k^4$, $\lambda_0$ integral
$L^{A_0}$ and $s$ integral $L^{B_1}$ are fixed logarithmic costs,
absorbed by $A=D_*+O(A_0+B_1)$.

For completeness, Parseval for a product polynomial gives
\begin{equation}\label{eq:trivial-energy}
 \frac1{\ph(\GN)}\sum_\chi
 \left|\sum_u c_u\chi(u)\psi(u)u^{it}\right|^2
 =\sum_{a'\in(\Z/\GN\Z)^\times}
      \left|\sum_{u\equiv a'\ (\GN)}c_u\psi(u)u^{it}\right|^2
 \ll(YX/\GN+1)\sum_u|c_u|^2
 \ll\frac{L^{O(1)}}{\GN}.
\end{equation}
Here the product length is $\asymp XY$ and
$|c_u|\ll L^{C}\tau_3(u)/(XY)$.
For $MN$ alone the squared coefficient norm is
$L^{O(1)}/(MN)\le L^{O(1)+B}/\GN$; this extra power is absorbed.
\end{proof}

\subsection{Prime sums from a fixed zero-free region}\label{subsec:P}
\begin{lemma}[Lemma P]\label{lem:P}
Let $2\le q\le\GN^3$, and let $\chi\bmod q$ be induced by a
primitive $\chi^*$ of conductor $q^*>1$.
Fix positive $C_{\mathrm{zf},1},C_{\mathrm{zf},2}$ and
$C_3\ge1$ with $C_{\mathrm{zf},2}\ge C_3+1$.
Suppose $L(s,\chi^*)\ne0$ in
\eqref{eq:Hstar}'s rectangle for these constants.
For $|u|\le L^{C_3}$ and $\GN^{1/6}\le y\le\GN$,
\begin{equation}\label{eq:P}
 S(y):=\sum_{\ell\le y}\Lambda(\ell)\chi(\ell)\ell^{iu}
 \ll yL^3(L^{-C_{\mathrm{zf},1}/12}+L^{-C_{\mathrm{zf},2}}).
\end{equation}
The implied constant can be absolute once
$L\ge L_0(C_{\mathrm{zf},1},C_{\mathrm{zf},2},C_3)$.
\end{lemma}
\begin{proof}
The imprimitive difference is supported on prime powers at primes
dividing $q$, and contributes at most
$\omega_{\rm all}(q)\log y\le5L^2$, absorbed since
$y\ge\GN^{1/6}$.
For $\chi^*$ put
\begin{equation}\label{eq:Perron-data}
 F(s)=-\frac{L'}{L}(s-iu,\chi^*),\quad
 c=1+1/L,\quad
 T=L^{C_{\mathrm{zf},2}}-L^{C_3}-1\ge\tfrac12L^{C_{\mathrm{zf},2}}.
\end{equation}
Truncated Perron \cite[Chapter~17]{Davenport} gives
\begin{equation}\label{eq:Perron}
 \sum_{\ell\le y}\Lambda(\ell)\chi^*(\ell)\ell^{iu}
 =\frac1{2\pi i}\int_{c-iT}^{c+iT}F(s)\frac{y^s}{s}\,ds
       +O(y(\log y)^2/T+\log y).
\end{equation}
The usual error is the sum of
$\Lambda(\ell)(y/\ell)^c
\min(1,(T|\log(y/\ell)|)^{-1})$.
The choice $y\le\GN$ ensures $y^c\le ey$ and
$\sum\Lambda(\ell)\ell^{-c}\ll L$.

For $1/2\le\sigma\le2$, the partial-fraction formula and local
zero count \cite[Chapter~16]{Davenport} read
\begin{equation}\label{eq:partial-fraction}
 \frac{L'}L(s,\chi^*)
 =\sum_{|\gamma-t|\le1}\frac1{s-\rho_0}
       +O(\log(q^*(|t|+2))),\quad
 N(t+1,\chi^*)-N(t,\chi^*)\ll\log(q^*(|t|+2)).
\end{equation}
Put $\sigma_0=1-C_{\mathrm{zf},1}\log L/(2L)$.
For $\sigma\ge\sigma_0$ and $|t|+1\le L^{C_{\mathrm{zf},2}}$,
all nearby zeros have real part at most
$1-C_{\mathrm{zf},1}\log L/L$ and horizontal distance at least
$C_{\mathrm{zf},1}\log L/(2L)$.
Their number is $O(\log(q^*L^{C_{\mathrm{zf},2}}))=O(L)$ after a
threshold depending on the fixed constants. Hence
\begin{equation}\label{eq:log-derivative}
 |L'/L(s,\chi^*)|\ll
 L\frac{2L}{C_{\mathrm{zf},1}\log L}+L\ll L^2.
\end{equation}
The shifted $F(s)$ has this bound on $\sigma\ge\sigma_0$, $|t|\le T$.

Move the contour to $\sigma_0$.
There is no zero pole in the rectangle: a pole $s=\rho_0+iu$ would
have $\Re\rho_0\ge\sigma_0$ and
$|\Im\rho_0|\le T+|u|\le L^{C_{\mathrm{zf},2}}-1$, contradicting
the region. Also $\sigma_0>1/2$, so $s=0$ is outside, and
$\chi^*$ is nonprincipal. The vertical and horizontal integrals
are respectively
\[
 \ll y^{\sigma_0}L^2\int_{-T}^T\frac{dt}{|\sigma_0+it|}
 \ll y^{\sigma_0}L^3,
 \qquad
 \ll(c-\sigma_0)y^cL^2/T\ll yL^2/T.
\]
Since $L/6\le\log y\le L$,
$y^{\sigma_0}\le yL^{-C_{\mathrm{zf},1}/12}$.
Combining these estimates yields
\begin{equation}\label{eq:P-end}
 S(y)\ll yL^{3-C_{\mathrm{zf},1}/12}+yL^{2-C_{\mathrm{zf},2}}
       +y(\log y)^2/T+\log y+L^2
 \ll yL^3(L^{-C_{\mathrm{zf},1}/12}+L^{-C_{\mathrm{zf},2}}).
\end{equation}
\end{proof}
\begin{remark}\label{rem:width}
The width $C\log L/L$ is what gives the arbitrary fixed logarithmic
saving on $y\ge\GN^{1/6}$. A width $c/\log q^*$ with
$q^*\asymp\GN$ gives only a factor of the size $e^{-c/6}$.
Only the logarithmic derivative on the contour is used.
\end{remark}

\begin{corollary}[Pointwise prime-factor polynomial]\label{cor:P}
Suppose $\xi=\chi\psi\bmod k\GN$ is induced by $\chi^*$ of
conductor $L^{A_1}<q^*\le k\GN$ and Lemma~\ref{lem:P} applies.
If
\begin{equation}\label{eq:P-constants}
 C_{\mathrm{zf},1}\ge12(D+4),\quad
 C_{\mathrm{zf},2}\ge D+4,\quad
 C_{\mathrm{zf},2}\ge C_3+1,\quad C_3=\max(B_1,2B)+1,
\end{equation}
then, uniformly for $|t|\le T_0+1$ and sufficiently large $L$
depending on the fixed parameters,
\begin{equation}\label{eq:N-pointwise}
 |N_\xi(t)|\le8L^{1-D}+\frac{4Z+2L}{V}\le9L^{1-D}.
\end{equation}
\end{corollary}
\begin{proof}
Put $u=t+v$, so $|u|\le L^{C_3}$.
Both $R_Z$ and the unit indicator are completely multiplicative.
Thus
\begin{equation}\label{eq:N-convolution}
 \begin{split}
 c_V(n)R_Z(n)\ind_{(n,\GN)=1}
 &=\sum_{\substack{\ell k'=n\\\ell>V}}
 \Lambda(\ell)R_Z(\ell)\ind_{(\ell,\GN)=1}
 R_Z(k')\ind_{(k',\GN)=1},\\
 N_\xi(t)&=\frac1N\sum_{k'<2N/V}
 R_Z(k')\ind_{(k',\GN)=1}\xi(k')(k')^{iu}
 \sum_{\ell\in I_{k'}}\Lambda(\ell)R_Z(\ell)
                    \ind_{(\ell,\GN)=1}\xi(\ell)\ell^{iu},\\
 I_{k'}&=(\max(V,N/k'),2N/k'].
 \end{split}
\end{equation}
In a ratio-two interval at most one power of each prime is present.
Dropping the indicators in the inner sum therefore costs at most
$\sum_{p\le Z}\log p+\sum_{p\mid\GN}\log p\le2Z+L$,
and after the $k'$ sum at most $(4Z+2L)/V$.
For every nonempty $I_{k'}$, the two endpoints are between $V$
and $2N\le\GN$, and
\[
 \log V=(\log x)/5\ge(L-B\log L-\log2)/5\ge L/6
\]
for sufficiently large $L$. The modulus is at most $\GN^2$.
Lemma~\ref{lem:P} and \eqref{eq:P-constants} give
$|S(y)|\le2C_PyL^{-D-1}\le yL^{-D}$ for $L\ge2C_P$.
Consequently
\[
 |N_\xi(t)|\le4L^{-D}(\log(2N/V)+1)+(4Z+2L)/V
 \le8L^{1-D}+(4Z+2L)/V.
\]
Since $V\ge\GN^{1/6}$ and $Z=\exp(L^{3/4})$, the last ratio is
at most $L^{1-D}$ for large $L$. Only endpoint sums with $y\ge V$
are used, even when $I_{k'}$ is short.
\end{proof}

The mean value theorem used below has the form
\begin{equation}\label{eq:MV-general}
 \int_{-T}^T\left|\sum_u b_u u^{it}\right|^2dt
 =\sum_u|b_u|^2\bigl(2T+O(\delta_u^{-1})\bigr),
\end{equation}
where $\delta_u$ is the separation of $\log u$ from the other
logarithmic frequencies \cite{MV}.

\subsection{Exceptional values of the short polynomial}\label{subsec:LS}
For $0\le i<2T_0$ let $J_i=[-T_0+i,-T_0+i+1]$.
For each $\chi\bmod\GN$ choose a point $t_{\chi,i}$ maximizing
$|B_{\xi_b}M_\xi N_\xi|$ on $J_i$.
Let $\mathcal P_0$ use the even $i$ and $\mathcal P_1$ the odd $i$.
Within each family, points for a fixed character are $1$-spaced, and
\begin{equation}\label{eq:discretize}
 \sum_\chi\int_{|t|\le T_0}|BMN|^2\,dt
 \le\sum_{\varepsilon=0,1}
       \sum_{(\chi,t)\in\mathcal P_\varepsilon}
        |B_{\xi_b}(t)M_\xi(t)N_\xi(t)|^2.
\end{equation}
Below $\mathcal P$ is either family, and each argument applies to both.

\begin{lemma}[Sobolev estimate]\label{lem:Sobolev}
For $F\in C^1([-T_0-1,T_0+1])$ and $1$-spaced points
$t_1<\cdots<t_r$ in $[-T_0,T_0]$,
\begin{equation}\label{eq:Sobolev}
 \sum_j|F(t_j)|^2\le\int_{|t|\le T_0+1}|F|^2\,dt
 +2\left(\int_{|t|\le T_0+1}|F|^2\,dt\right)^{1/2}
   \left(\int_{|t|\le T_0+1}|F'|^2\,dt\right)^{1/2}.
\end{equation}
\end{lemma}
\begin{proof}
Average
$|F(t_j)|^2=|F(s)|^2+\int_s^{t_j}(|F|^2)'$
over $s\in[t_j-1/2,t_j+1/2]$.
These intervals have disjoint interiors; use
$|(|F|^2)'|\le2|F||F'|$ and Cauchy--Schwarz.
\end{proof}

\begin{lemma}[Mean value over residue classes]\label{lem:MV}
If $(c_u)$ is supported on positive $u\le4\GN$ and $|\psi_3|\le1$,
then for $T\ge1$,
\begin{equation}\label{eq:MV}
 \sum_{\chi\bmod\GN}\int_{-T}^T
 \left|\sum_uc_u\chi(u)\psi_3(u)u^{it}\right|^2dt
 \le\ph(\GN)(2T+C_0')\sum_u|c_u|^2
\end{equation}
with absolute $C_0'$.
\end{lemma}
\begin{proof}
Character orthogonality gives
\[
 \sum_\chi\left|\sum_uc_u\chi(u)d_u\right|^2
 =\ph(\GN)\sum_{a'\in(\Z/\GN\Z)^\times}
           \left|\sum_{u\equiv a'\ (\GN)}c_ud_u\right|^2.
\]
Each class has at most four positive integers at most $4\GN$.
If $u<u'$ are in it, $u'\ge u+\GN\ge(4/3)u$ since
$u\le3\GN$. The Montgomery--Vaughan mean value theorem
\cite{MV} therefore gives $C_0'=O(1/\log(4/3))$.
For length at most $\GN$, a unit class has at most one term and the
integral in that class is exactly $2T$ times its squared coefficient.
\end{proof}

\begin{lemma}[Lemma LS]\label{lem:LS}
Fix $A_s\ge1$ and put
$E=\{(\chi,t)\in\mathcal P:|B_{\xi_b}(t)|>L^{-A_s}\}$.
For $L\ge L_0(K,a_K,A_s,B_1,C_B)$,
\begin{equation}\label{eq:LS}
 |E|\le\exp\bigl(L^{1-a_K}(2A_s+K+10)\log L\bigr)
       \le\exp(L^{0.9}).
\end{equation}
\end{lemma}
\begin{proof}
Let $j=\lfloor L/\log(16Y)\rfloor\ge1$; then
$(16Y)^j\le\GN<(16Y)^{j+1}$.
Write $B_{\xi_b}(t)^j=\sum_uc_u^{(j)}\xi_b(u)u^{it}$, where
\begin{equation}\label{eq:B-power}
 c_u^{(j)}=\sum_{\substack{b_1\cdots b_j=u\\b_i\in\Aset}}
      \prod_i c_{b_i},\qquad
 c_b=\frac{\eta(b/Y)e(-\lambda_0b/Y)}{Y\prod_iV_i},\qquad
 |c_b|\le C_B/(16Y).
\end{equation}
This polynomial is supported on $[Y^j,(16Y)^j]\subset[1,\GN]$.
For each disjoint band the $j$ prime factors of $u$, with
multiplicity, can be assigned to the $j$ slots in at most $j!$ ways.
Hence there are at most $(j!)^K$ representations and
\begin{equation}\label{eq:B-power-norm}
 |c_u^{(j)}|^2\le(j!)^K\sum_{\rm reps}\prod_i|c_{b_i}|^2,
 \qquad
 \sum_u|c_u^{(j)}|^2
 \le(j!)^K\left(\sum_b|c_b|^2\right)^j
 \le(j!)^K\left(\frac{C_B^2}{16Y}\right)^j.
\end{equation}
Use Lemmas~\ref{lem:Sobolev} and~\ref{lem:MV} on $B^j$;
its derivative inserts a coefficient $i\log u$, with $|\log u|\le L$.
Since $\xi_b=\chi\psi_2$ and $|\psi_2|\le1$,
\begin{equation}\label{eq:B-moment}
 \sum_{(\chi,t)\in\mathcal P}|B_{\xi_b}(t)|^{2j}
 \le3L\ph(\GN)(2T_0+2+C_0')(j!)^K(C_B^2/(16Y))^j.
\end{equation}
The left side is at least $|E|L^{-2jA_s}$.
Since $Y^{-j}<16^{j+1}Y/\GN$ and $\ph(\GN)\le\GN$,
\begin{equation}\label{eq:E-count}
 |E|\le L^{2jA_s}3L(2T_0+2+C_0')(j!)^KC_B^{2j}16Y.
\end{equation}
Every $b\in\Aset$ exceeds $\exp(L^{a_K})$, so
$j\le L^{1-a_K}$, $\log j\le\log L$, and
$\log Y\le2\sum_iL^{a_i}+\log16\le3KL^{a_K}$.
It follows that
\[
 \log|E|\le2A_sj\log L+Kj\log j+2j\log C_B+\log Y+O(\log L)
 \le L^{1-a_K}(2A_s+K+10)\log L.
\]
For fixed $a_K>1/10$, this is eventually at most $L^{0.9}/2$.
\end{proof}

\begin{corollary}[Bulk estimate]\label{cor:bulk}
For each family $\mathcal P$,
\begin{equation}\label{eq:bulk}
 \begin{split}
 \sum_{\mathcal P\setminus E}|BMN|^2
 &\le L^{-2A_s}\sum_\mathcal P|M_\xi N_\xi|^2\\
 &\le L^{-2A_s}3L\ph(\GN)(2T_0+O(1))
           \frac{L^2\sum_{u\le4MN}\tau(u)^2}{(MN)^2}\\
 &\ll L^{-2A_s+B_1+B+8}\frac{\ph(\GN)}{\GN}.
 \end{split}
\end{equation}
\end{corollary}
\begin{proof}
The product $MN$ polynomial is supported in $(MN,4MN]$, with
$4MN\le8x\le4\GN$ and coefficients at most $L\tau(u)/(MN)$.
Apply Lemmas~\ref{lem:Sobolev} and~\ref{lem:MV}, and
$\sum_{u\le y}\tau(u)^2\ll y(\log y)^3$.
\end{proof}

\subsection{Energy on the exceptional set}\label{subsec:HM}
\begin{lemma}[Lemma HM]\label{lem:HM}
Let $(\xi_r,t_r)_{r\le R'}$ be character-frequency pairs modulo $q$,
$1$-spaced in $t$ whenever their characters agree.
Let $a_m$ be supported on $(X',X'']$, $X'<X''\le8X'$, and let
$w\ge0$ be at least $1$ on that interval. Then
\begin{equation}\label{eq:HM}
 \sum_r\left|\sum_ma_m\xi_r(m)m^{it_r}\right|^2
 \le\|a\|_2^2\max_r\sum_s|G_{rs}|,\qquad
 G_{rs}=\sum_mw(m)\xi_r(m)\overline{\xi_s(m)}m^{i(t_r-t_s)}.
\end{equation}
\end{lemma}
\begin{proof}
Put $S_r=\sum_ma_m\xi_r(m)m^{it_r}$ and
$b_r=\overline{S_r}/(\sum_s|S_s|^2)^{1/2}$; the zero case is
immediate. Cauchy gives
\[
 (\sum_r|S_r|^2)^{1/2}=\sum_rb_rS_r
 \le\|a\|_2\left(\sum_mw(m)
           \left|\sum_rb_r\xi_r(m)m^{it_r}\right|^2\right)^{1/2}.
\]
The squared last factor is $\sum_{r,s}b_r\bar b_sG_{rs}$.
The Hermitian Schur bound and $\sum|b_r|^2=1$ prove the claim.
\end{proof}

Choose $q=k\GN$ and a smooth $w_0$ with
$\ind_{(X',X'']}\le w_0\le\ind_{[X'/2,16X']}$ and
$|w_0^{(i)}|\le C_i(X')^{-i}$. There are three kinds of Gram entry.
\begin{enumerate}
\item On the diagonal, $G_{rr}\le16X'$.
\item If $\xi_r=\xi_s$, $r\ne s$, and $\tau=t_r-t_s$, then
$|\tau|\ge1$ and
\begin{equation}\label{eq:Gram-same}
 G_{rs}=\sum_{d\mid q}\mu(d)d^{i\tau}
              \sum_{m'}w_0(dm')(m')^{i\tau}.
\end{equation}
For $d\le16X'$, Euler summation replaces the inner sum by its
integral with error $O(1+|\tau|)$: for
$f(y)=w_0(dy)y^{i\tau}$,
\[
 \int|f'|\le\int|w_0'|
       +|\tau|\int_{X'/(2d)}^{16X'/d}dy/y\le C(1+|\tau|).
\]
Two integrations by parts give integral size
$O((X'/d)(1+|\tau|)^{-2})$.
For $d>16X'$ the sum is empty. Thus
\begin{equation}\label{eq:Gram-same-bound}
 |G_{rs}|\le C'X'(1+|\tau|)^{-2}\log L
                +\tau(k\GN)(1+|\tau|),
\end{equation}
using $\sum_{d\mid q}1/d\ll\log\log q$.
The $1$-spacing gives
\begin{equation}\label{eq:Gram-same-row}
 \sum_{\substack{s\ne r\\\xi_s=\xi_r}}|G_{rs}|
 \le C''X'\log L+(2T_0+1)^2\tau(k\GN).
\end{equation}
\item If $\xi_r\ne\xi_s$, their quotient is nonprincipal.
On units it is $\chi_r\overline{\chi_s}$, and these are distinct
characters modulo $\GN$.
Partial summation gives
\begin{equation}\label{eq:Gram-distinct}
 |G_{rs}|\le C(1+|\tau|)\Delta_\xi(16X'),\qquad
 \Delta_\xi(Y')=\max_{y\le Y'}\left|\sum_{m\le y}\xi(m)\right|,
 \quad |\tau|\le2T_0+1.
\end{equation}
If $\xi$ is induced by primitive $\xi^*\bmod q^*>1$, put
$q_1=\prod_{p\mid k\GN,p\nmid q^*}p$. Then
\begin{equation}\label{eq:imprimitive}
 \sum_{m\le y}\xi(m)
 =\sum_{d\mid q_1}\mu(d)\xi^*(d)\sum_{m'\le y/d}\xi^*(m'),
 \qquad \Delta_\xi(Y')\le2^{\omega_{\rm all}(k\GN)}\Delta_{\xi^*}(Y').
\end{equation}
The two inputs are P\'olya--Vinogradov and Burgess at $r=2$:
\begin{align}
 \Delta_{\xi^*}(Y')&\le C(q^*)^{1/2}\log q^*
       \le C(L^{A_0}\GN)^{1/2}\,2L,\label{eq:PV}\\
 \left|\sum_{H_0<n\le H_0+H}\xi^*(n)\right|
   &\ll_\varepsilon H^{1/2}(q^*)^{3/16+\varepsilon}
       \quad(H\ge1),\label{eq:Burgess}
 \end{align}
where \eqref{eq:Burgess} holds for every modulus, as in
\cite[introduction, (1)]{HeathBrown2012} (arbitrary moduli are
allowed for $r\le3$).
Also
\[
 2^{\omega_{\rm all}(k\GN)}
 \le2^{A_0\log L}2^{\omega_{\rm all}(\GN)}
 \le L^{A_0}\exp(O(L/\log L))=\GN^{o(1)},\qquad
 \tau(k\GN)=\GN^{o(1)}.
\]
\end{enumerate}

\begin{corollary}[Exceptional energy]\label{cor:exceptional-energy}
Let $E$ be as in Lemma~\ref{lem:LS} and fix $\eta_0=1/40$.
Then
\begin{align}
 M\ge\GN^{3/8+\eta_0}&\quad\Longrightarrow\quad
        \sum_E|M_\xi(t)|^2\ll\log L,\label{eq:EM}\\
 \sum_E|M_\xi(t)|^6&\ll L^9,\label{eq:EM3}\\
 N\ge\GN^{1/2+2\eta_0}&\quad\Longrightarrow\quad
        \sum_E|N_\xi(t)|^2\ll L^2\log L.\label{eq:EN}
\end{align}
\end{corollary}
\begin{proof}
For \eqref{eq:EM} use $a_m=\alpha_m/M$,
$\|a\|_2^2\le2/M$, $X'=M$ and Burgess with
$\varepsilon=\eta_0/8$.
The distinct-character row contribution is at most
\begin{equation}\label{eq:EM-margin}
 \begin{split}
 &|E|C(2T_0+2)2^{\omega_{\rm all}(k\GN)}
       C_\varepsilon(16M)^{1/2}(L^{A_0}\GN)^{3/16+\varepsilon}\\
 &\hspace{4mm}\le M\,\GN^{-\eta_0/2+\varepsilon+o(1)}
                      \exp(L^{0.9})
   =M\,\GN^{-3/320+o(1)}\exp(L^{0.9})=o(M).
 \end{split}
\end{equation}
Together with $16M+C''M\log L+(2T_0+1)^2\tau(k\GN)$ this proves
\eqref{eq:EM}.
For \eqref{eq:EM3},
$M^3\ge(x^{1/5}/2)^3\ge\GN^{3/5-o(1)}$.
The coefficients of $M_\xi^3$ are at most $M^{-3}\tau_3(u)$ on
$(M^3,8M^3]$, with squared norm $\ll L^8/M^3$.
P\'olya--Vinogradov makes the distinct-character row at most
\[
 |E|L^{O(1)}\GN^{1/2+o(1)}
 \le M^3\GN^{1/2-3/5+o(1)}\exp(L^{0.9})=o(M^3),
\]
yielding $L^9$.
For \eqref{eq:EN}, take $a_n=\beta_n/N$,
$\|a\|_2^2\le2L^2/N$; P\'olya--Vinogradov gives row
$\le N\GN^{-2\eta_0+o(1)}\exp(L^{0.9})=o(N)$.
In all cases the same-character row is $O(X'\log L)$, because
$X'\ge\GN^{1/5-o(1)}$ absorbs $(2T_0+1)^2\tau(k\GN)$.
\end{proof}

\begin{lemma}[Twisted sifted M\"obius sums]\label{input:sifted}
For a primitive character of conductor at most $L^D$, every fixed $C_t,A,\delta>0$ admits


\begin{equation}\label{v2:5-11}
\sup_{|t|\le x^{C_t}}\left|\sum_{m\in I}\mu(m)R_Z(m)\xi^*(m)m^{it}\right|
\ll M L^{-A},\qquad I\subset[M,2M],\quad M\ge x^\delta.
\end{equation}


\end{lemma}\begin{proof}First use \cite[Lemma~3.9(ii)]{MSTT} at any dyadic scale $D_0\ge x^{\delta/2}$ with auxiliary variable $X_*=D_0^{R_*}$, fixed $R_*>2(C_t+1)/\delta$, and exponent $1/R_*$. Its freely chosen logarithmic precision covers both $L^D$ and the desired saving, uniformly over these scales. To obtain arbitrary subintervals, decompose a prefix ending at any $u\in[M,2M]$ into dyads down to scale $M L^{-E}$, for a large fixed $E$. The sum of their square-root lengths is $O(\sqrt M)$, while the remaining initial segment contributes at most $O(\sqrt M L^{-E/2})$ to the weighted sum with coefficient $m^{-1/2}$. All retained scales exceed $x^{\delta/2}$ for sufficiently large $x$. Thus weighted prefix sums, and hence their differences on every subinterval of $[M,2M]$, are $O(\sqrt M L^{-A})$ after selecting the precision and $E$. Partial summation against the \emph{real} factor $m^{1/2}$ now gives $O(ML^{-A})$ for the untapered interval sum; the phase is not differentiated. This derives the interval estimate from the full-dyad consequence of the cited lemma.

Choose the fixed rescaling exponent once large enough for the
same argument with $\delta/2$ as well as $\delta$; for example
$R_*>8(C_t+1)/\delta$ suffices for the frequency requirement.
After the convolution below, all retained prefix dyads then
have length at least $x^{\delta/4}$ for sufficiently large $x$.
The identity


\begin{equation}
\mu(m)R_Z(m)=\sum_{\substack{d\mid m\\P^+(d)\le Z}}\mu(m/d)
\end{equation}


is checked prime by prime. For $d\le M^{1/2}$ apply the preceding untapered estimate at scale $M/d\ge x^{\delta/2}$; summing $1/d$ over $Z$-smooth $d$ costs $O(\log Z)$. For $d>M^{1/2}$, Rankin with exponent $1/\log Z$ bounds its reciprocal mass by $O(\log Z\exp(-c_\delta L^{1/4}))$. This proves \eqref{v2:5-11}. If the coefficient also has the unit restriction $(m,h)=1$, remove it at cost $O(M\omega_{\rm all}(h)/Z)$, uniformly in the phase, since any excluded sifted factor exceeds $Z$. Consequently \eqref{v2:5-11} still holds. Fixed Mellin phases and smooth weights of logarithmic variation are absorbed by increasing the fixed precision.

\end{proof}
\begin{remark}[The published interval estimate]
Lemma~3.9(ii) of \cite[p.~46]{MSTT} is stated with a supremum
over subintervals of its dyadic range. The proof above uses its
full-dyad consequence and recovers the interval estimate by prefix
decomposition and partial summation against a real weight.
For a bounded-variation weight the estimate is multiplied by its
supremum plus variation.
\end{remark}

\subsection{The energy theorem}\label{subsec:A}
\begin{theorem}[Reflected energy estimate]\label{thm:energy}
Fix $A$ and $A_1\ge1$, and put
\begin{equation}\label{eq:A-constants}
 \begin{aligned}
 D&=\left\lceil\frac32(A+B_1+B+20)\right\rceil,
 & C_3&=\max(B_1,2B)+1,\\
 C_{\mathrm{zf},2}&=\max(D+4,C_3+1),
 & C_{\mathrm{zf},1}&=12(D+4),\\
 A_s&=A+B_1+B+10.
 \end{aligned}
\end{equation}
In the setting of Section~\ref{subsec:arc-reduction},
$\mathrm H1^\star(C_{\mathrm{zf},1},C_{\mathrm{zf},2};A_1)$ at $\GN$
implies \eqref{eq:alpha-star}, for a threshold depending on all
fixed parameters and $A_1$.
\end{theorem}
\begin{proof}
Split the characters into
$\mathcal X_s=\{\chi:\cond(\chi\psi)\le L^{A_1}\}$ and its
complement $\mathcal X_b$.
For $\chi\in\mathcal X_s$, let $\xi^*$ induce $\xi=\chi\psi$.
On the coefficient support,
\begin{equation}\label{eq:small-conductor-M}
 M_\xi(t)=\frac1M\sum_m\mu(m)R_Z(m)\ind_{(m,\GN)=1}
                    \xi^*(m)m^{i(t+v)}.
\end{equation}
Removing the unit indicator changes this by at most
\[
 \sum_{\substack{p\mid\GN\\p>Z}}\frac{2M/p}{M}
 \le2\omega_{\rm all}(\GN)/Z\le2Le^{-L^{3/4}}.
\]
Apply Lemma~\ref{input:sifted} and its bounded-variation extension to the
sum without the indicator: the modulus is at most $L^{A_1}$,
$M\ge x^\delta$ and $|t+v|\le x$ for large $x$.
For every fixed $A'$ this gives
$\sup_{|t|\le T_0}|M_\xi(t)|\ll L^{-A'}$.
This is the use of the sifted M\"obius input, whose proof cites
\cite[Lemma~3.9(ii)]{MSTT}.
For fixed $\psi$, the primitive inducing character determines
$\chi$, so $|\mathcal X_s|\le L^{2A_1}$.
The polynomial $B_{\xi_b}N_\xi$ is supported on $u\le32YN$ with
coefficients at most $C_BL\tau(u)/(16YN)$ and squared coefficient
norm $\ll L^5/(YN)$. The ordinary mean value theorem gives
$\int_{|t|\le T_0}|BN|^2\ll(T_0+YN)L^5/(YN)\ll L^5$.
Thus with $A'=A+A_1+4$,
\begin{equation}\label{eq:small-conductor-energy}
 \frac1{\ph(\GN)}\sum_{\chi\in\mathcal X_s}\int|BMN|^2
 \ll\frac{L^{2A_1-2A'+5}}{\ph(\GN)}
 \ll L^{-A}/\GN,
\end{equation}
using $\GN/\ph(\GN)\ll\log L$.

For $\chi\in\mathcal X_b$,
$q^*\mid k\GN$ and $L^{A_1}<q^*\le k\GN$.
The assumed region and Corollary~\ref{cor:P} yield
$|N_\xi(t)|\le9L^{1-D}$ for $|t|\le T_0+1$.
Discretize on this subset as above; restricting characters only
reduces the bounds for $\mathcal P$ and $E$.
The bulk is at most
\[
 L^{-2A_s+B_1+B+8}\ph(\GN)/\GN
 \le L^{-A-1}\ph(\GN)/\GN.
\]
If $M\ge\GN^{3/8+\eta_0}$, \eqref{eq:EM} gives
\begin{equation}\label{eq:caseA}
 \sum_E|BMN|^2
 \le C_B^2\sup_E|N|^2\sum_E|M|^2
 \ll L^{2-2D}\log L.
\end{equation}
If $M<\GN^{3/8+\eta_0}$, then
\begin{equation}\label{eq:caseB-length}
 \begin{gathered}
 N=\frac{MN}{M}\gg\frac{x}{M}
       \gg\frac{\GN^{5/8-\eta_0}}{L^B},\\
 N\ge\GN^{1/2+2\eta_0}\quad\text{for sufficiently large }\GN.
 \end{gathered}
\end{equation}
H\"older with exponents $3,3/2$, followed by
\eqref{eq:EM3} and \eqref{eq:EN}, gives
\begin{equation}\label{eq:caseB}
 \begin{split}
 \sum_E|B|^2|M|^2|N|^2
 &\le C_B^2\sup_E|N|^{2/3}\sum_E|M|^2|N|^{4/3}\\
 &\le C_B^2(9L^{1-D})^{2/3}
       (\sum_E|M|^6)^{1/3}(\sum_E|N|^2)^{2/3}
 \ll L^{(2/3)(1-D)+3+2}.
 \end{split}
\end{equation}
In either case the exceptional contribution is
\begin{equation}\label{eq:exceptional-end}
 \ll L^{6-2D/3}\le L^{-A-B_1-B-14}\le L^{-A-1}.
\end{equation}
One may also use the looser exponent
$(2/3)(1-D)+6$ in the second case and
$3-2D$ with a factor $\log L$ in the initial case; both leave the
same required margin. Sum the two families and divide by $\ph(\GN)$:
\[
 \frac1{\ph(\GN)}\sum_{\chi\in\mathcal X_b}\int|BMN|^2
 \ll L^{-A-1}/\GN+L^{-A-1}/\ph(\GN)\ll L^{-A}/\GN.
\]
\end{proof}
The inputs to Theorem~\ref{thm:energy} are the ordinary mean value theorem,
P\'olya--Vinogradov, Burgess at $r=2$, partial fractions and zero
counting, Perron, and Lemma~\ref{input:sifted}. The half-plane
\eqref{eq:H1} enters only by supplying \eqref{eq:Hstar}.
For one fixed region the theorem supplies its selected fixed
precision $A$; it does not assert all precisions from one fixed width.

\subsection{Changes for general \texorpdfstring{$h$}{h}: the hybrid analytic interface}\label{subsec:hybrid}
In this subsection $L=\log x$, $X=MN\asymp y$, $y\ge x/L^B$,
$\Delta=-\varepsilon h$ and $H=|h|$. Normalize $B,M,N$ as in
\eqref{eq:polynomials}, replacing the unit modulus $\GN$ by $H$.
The same fibre expansion used in
Section~\ref{subsec:arc-reduction} applies to the units modulo
$kH$ over those modulo $H$: the kernel of reduction has size at
most $k$, for either sign of $\Delta$. On the arc
$\phi=c/k+\lambda_0/Y$, expand
$e(\operatorname{sgn}(\Delta)cu/(kH))$ on each fibre.
Its Fourier coefficients have modulus at most one, and each
character of the fibre group extends to the full unit group.
The same representative at both endpoints and Cauchy as in
\eqref{eq:fibre-cauchy} remove the fibre-dependent coefficients
before orthogonality modulo $H$. Together with the $a,b$
phases modulo $k$, a bound $k^8$ for all character losses
suffices and is independent of $H$.
The characters occur in the two locations
\begin{equation}\label{v2:5-3}
 \xi=\chi\psi\text{ on }m,n,\qquad
 \xi_b=\chi\psi\psi'_b\text{ on }b,\qquad
 \chi\pmod H,\quad\psi'_b\pmod k.
\end{equation}
Choose $A_1>A_0$. If $\cond(\xi)>L^{A_1}$, then
$\xi_b$ is nonprincipal; otherwise $\xi$ would be induced from
modulus $k$. For each fixed extension $\psi$, at most
$L^{2A_1+1}$ characters $\chi$ lie in the complementary range,
since the primitive inducing character determines $\chi$.

\subsubsection{Uniform Mellin scale}
Let $\mathcal T=\max(1,X/H)$, $w=u/(YX)$, $w'=u'/(YX)$, so $w,w'$ belong to a fixed compact subset of $(0,\infty)$. The smooth factor is


\begin{equation}
\psi_1((X/\Delta)(w-w'))e(\lambda_0(X/\Delta)(w-w')).
\end{equation}


With $z_0=\log w'$ and $\zeta=\mathcal T\log(w/w')$, insert fixed cutoffs in $w,w'$ equal to one on their ranges and call the resulting function $K_{\lambda_0}(z_0,\zeta)$. If $H\le X$, the compact support of $\psi_1$ makes $\zeta$ lie in a fixed interval. If $H>X$, $\mathcal T=1$ and the two fixed cutoffs give that conclusion. All mixed derivatives have bounds by fixed powers of $L^{A_0}$, uniformly in $H$: in the former case $(X/H)/\mathcal T=1$, while in the latter $X/H\le1$.

Its Fourier transform satisfies, for every fixed integer $j$,


\begin{equation}\label{v2:5-4}
|\widehat K_{\lambda_0}(v,s)|\ll_j
(1+|v|/L^{B_0})^{-j}(1+|s|/L^{B_0})^{-j},
\end{equation}


with fixed $B_0$. Fourier inversion with $t=\mathcal T v$ yields an \emph{exact factor $1/\mathcal T$} and factors $w^{it}(w')^{i(s-t)}$. Including normalized endpoint polynomials and arc measure $d\phi=d\lambda_0/Y$ gives


\begin{equation}\label{v2:5-5}
\frac{YX^2}{\mathcal T}\cdot
\frac1{\varphi(H)}\sum_\chi
\iint\widehat K_{\lambda_0}(t/\mathcal T,s)
Q_\chi(t)Q'_\chi(s-t)\,dt\,ds.
\end{equation}


Here $Q=BMN$, with $M^{-1},N^{-1},Y^{-1}$ normalizations as in the reflected proof; the $b$-phase is $e(-\lambda_0b/Y)$ and the primed $a$-phase is its inverse. The precise sign has no effect on the estimates below. Frequency cutoffs can be chosen as


\begin{equation}\label{v2:5-6}
|s|\le L^{B_1},\qquad |t|\le\mathcal T L^{B_1},\qquad B_1=B_0+2,
\end{equation}


independently of the requested saving. Higher integration-by-parts order, not a growing $B_1$, supplies arbitrary fixed precision in the tails.

\subsubsection{Hybrid mean value}
For arbitrary bounded coefficient polynomials of lengths $M,N$, their product has coefficients $c_u$ on $u\asymp X$ with


\begin{equation}
\sum_u|c_u|^2\ll L^{C_2}/X.
\end{equation}


For a fixed character extension $\psi$ and any real interval $I$ of length $T\ge1$,


\begin{equation}\label{v2:5-7}
\frac1{\varphi(H)}\sum_{\chi\bmod H}
\int_I\left|\sum_u c_u\psi(u)\chi(u)u^{it}\right|^2dt
\ll (T+X/H)\sum_u|c_u|^2.
\end{equation}


Orthogonality leaves a sum of time integrals over residue classes modulo $H$. Distinct integers in one class, inside a fixed multiple of $X$, have logarithms spaced at least $cH/X$. Montgomery–Vaughan's Hilbert-inequality form of the mean value bounds each class by $(T+O(X/H))$ times its coefficient square sum. If a class has only one term its integral is exactly $T$ times that square. Summing proves \eqref{v2:5-7}. Translation of $I$ multiplies coefficients by phases and changes none of the estimates.

At $T\asymp\mathcal T L^{B_1}$, \eqref{v2:2-1} and $\mathcal T=\max(1,X/H)$ imply the bound $L^{C_3}/H$. This is also a uniform bound for polynomially weighted tails after multiplication by \eqref{v2:5-4}. Dyadic decomposition of $t$ and then $s$, using \eqref{v2:5-7} and Cauchy on the primed and unprimed sides, shows that the part outside \eqref{v2:5-6} contributes at most $L^{-A}/H$ for any fixed $A$, by taking $j$ large. The possible factor $H/X\ll L^{B+C}$ is included in the fixed precision budget. The trivial bound $|B|\ll_{K,\rho}1$ is enough for this tail argument.

\subsubsection{Short-prime cancellation under \texorpdfstring{$\mathrm H1$}{H1}}\label{subsec:short-prime}
Suppose $\xi$ is nonprincipal modulo $q\le x^{C_q}$, and $|t|\le x^{C_t}$ for fixed $C_q,C_t$. Under $\mathrm H1$, uniformly for $P\ge\exp(cL^{a_1})$ and $P\le\exp(CL^\theta)$,


\begin{equation}\label{v2:5-8}
\sum_{P<p\le P'}\frac{\xi(p)p^{it}}p\ll P^{-1/16}L^{C_4},\qquad P\le P'\le2P.
\end{equation}


To prove this, pass first to the primitive inducing character. On $\sigma=15/16$ the partial-fraction formula for $L'/L$ and the zero count in a unit-height interval give $|L'/L(\sigma+iu)|\ll\log(q(|u|+3))$: every nontrivial zero is at horizontal distance at least $1/16$, and there is no pole for a nonprincipal character. The same bound holds on the contour between $15/16$ and $1+1/\log P$. Apply truncated Perron to $-L'/L(s-it,\xi)$ at height $x^{C_t+C_q+20}$ and move to $15/16$. Taking half-integer endpoints, then restoring an endpoint term if necessary, bounds the truncation error by $O(PL^{O(1)}x^{-C_t-C_q-20}+L^{O(1)})$. The vertical integral is $O(P^{15/16}L^{O(1)})$, and the horizontal integrals are negligible. This proves the twisted von Mangoldt endpoint bound. Prime powers contribute $O(P^{1/2}L^{O(1)})$. Partial summation with the real weight $1/(u\log u)$ gives \eqref{v2:5-8}; the phase $u^{it}$ is part of the sum and is never differentiated. Imprimitive Euler factors and puncturing at $p\mid h$ cost $O(L/P)$ on these bands, absorbed by the bound.

To estimate $B_{\xi_b}$, put $g(y)=y\eta(y)e(-\lambda_0y)$ and use its Mellin transform. Isolating one prime from the first band gives


\begin{equation}\label{v2:5-9}
B_{\xi_b}(t)=\frac1{V_1}\int_{\mathbb R}\widehat g(v)Y^{-iv}
 \left(\sum_{p\in\mathcal P_1}\frac{\xi_b(p)p^{i(t+v)}}p\right)
 \prod_{i=2}^K\left(\frac1{V_i}\sum_{p\in\mathcal P_i}\frac{\xi_b(p)p^{i(t+v)}}p\right)dv.
\end{equation}


The product is at most one in modulus, including the empty product when $K=1$. The transform has $L^1$ norm at most a fixed logarithmic power. Truncate $|v|\le x^2$ using three or more integrations by parts; its tail is $O(x^{-1})$ times a fixed logarithmic factor. On \eqref{v2:5-6}, $|t+v|\le x^3$ for large $x$. Decompose $\mathcal P_1$ into ordinary dyadic intervals and apply \eqref{v2:5-8}. Thus for each fixed $D$,


\begin{equation}\label{v2:5-10}
\sup_{|t|\le2\mathcal T L^{B_1}}|B_{\xi_b}(t)|\ll_D L^{-D}
\end{equation}


whenever $\xi_b$ is nonprincipal. Combining \eqref{v2:5-10} with \eqref{v2:5-7} proves the large-conductor contribution $\ll_A L^{-A}/H$.

\subsubsection{Small conductors and sparse-frequency energy}\label{subsec:sparse}
Let $\mathcal P\subset[P,2P)$ with $cL^{0.1}\le\log P\le CL^\theta$, and


\begin{equation}
P_s(t)=P^{-1}\sum_{p\in\mathcal P}v_p\xi_b(p)p^{i(t+v)},\quad |v_p|\le1,
\end{equation}


where $v$ is any real number. For fixed $A_s,B_1$, let $\mathcal I$ be the unit intervals with integer endpoints meeting $|t|\le2XL^{B_1}$ and $|P_s(t)|>L^{-A_s}$. Then


\begin{equation}\label{v2:5-12}
|\mathcal I|\le\exp(O(L^{0.9}\log L)),\qquad
\sum_{I\in\mathcal I}\sup_{t\in I}|N_\xi(t)|^2\ll L^{C_5}.
\end{equation}


For the first assertion take $T_0=4XL^{B_1}$ and $j=\lfloor\log T_0/\log(2P)\rfloor$. Unique factorization gives coefficient-square mass of $P_s^j$ at most $j!(2/P)^j$, and its length is at most $T_0$. Choose a large-value point in each interval and divide the intervals into three classes modulo 3, making the points separated. On disjoint unit neighborhoods the inequality


\begin{equation}
|F(t_r)|^2\ll\int_{t_r-1/2}^{t_r+1/2}(|F(u)|^2+|F(u)F'(u)|)du
\end{equation}


follows by integrating the derivative of $|F|^2$. Apply the ordinary Dirichlet-polynomial mean value to $F=P_s^j$ and $F'$, whose coefficients acquire factors $\log n\ll L$. It follows that


\begin{equation}
|\mathcal I|\ll T_0L^{O(1)}L^{2A_sj}j!(2/P)^j.
\end{equation}


Here $j\ll L^{0.9}$ and $\log T_0-j\log P\ll\log P+j$, so the logarithm of this bound is $O(L^{0.9}\log L)$. The growing-order factorial has been retained.

For the second assertion choose a maximizing point for $|N_\xi|$ in each closed interval and again separate by three classes. The unweighted evaluation Gram entries are $\sum_{N<n\le2N}n^{i(t_r-t_s)}$. Put $T_*=\exp(L/(\log L)^2)$. For $1\le|u|\le T_*$, Euler summation and one integration by parts give the normalized bound $O(1/|u|+x^{-c_\delta})$; the discretization error is $O((1+|u|)/N)$. For $T_*<|u|\le x^{C_0}$, the second-derivative estimate when $|u|\le N$ gives $O(\sqrt{|u|}/N+|u|^{-1/2})$. When $|u|\ge N$, choose a fixed integer $k>C_0/\delta$ and the classical exponent pair


\begin{equation}
\kappa_k=(2^{k+2}-2)^{-1},\qquad\lambda_k=1-(k+1)\kappa_k.
\end{equation}


It bounds the normalized logarithmic-phase sum by $O(N^{-(k+2-C_0/\delta)\kappa_k}+N^{-1})$. Thus in the high range the normalized Gram entry is $O(\exp(-cL/(\log L)^2))$. The number of intervals in \eqref{v2:5-12} times either of the two error terms tends to zero. Separation gives row sum of $1/|t_r-t_s|$ at most $O(\log(2+|\mathcal I|))=O(L)$. The Gram row sums are therefore $O(NL)$. The Hermitian Schur bound and $\sum_n|\beta_n\xi(n)/N|^2\ll L^{O(1)}/N$ prove the second assertion of \eqref{v2:5-12}, with no dependence on the character conductor or the shift $v$.

For each small-conductor $\xi$, split the integral of $|P_sMN|^2$ into $|P_s|\le L^{-A_s}$ and its complement. The first part is $\ll L^{-2A_s+C_6}$ by the ordinary mean value of $MN$ over length $O(XL^{B_1})$. On the second part use \eqref{v2:5-11} for $M$ and the second assertion of \eqref{v2:5-12} for $N$, giving $\ll L^{-2A'+C_5}$. Choosing fixed $A_s,A'$ sufficiently large gives every fixed saving. Minkowski applied to \eqref{v2:5-9} and the ordinary dyadic prime decomposition then gives


\begin{equation}\label{v2:5-13}
\int_{|t|\le2XL^{B_1}}|B_{\xi_b}(t)M_\xi(t)N_\xi(t)|^2dt\ll_A L^{-A}.
\end{equation}


The argument is uniform in the bounded $b$-character and in the Mellin shift in \eqref{v2:5-9}. In \eqref{v2:5-11}, the extra frequency is the original $t$ plus the Vaughan Mellin twist, not that prime-polynomial shift. Summing \eqref{v2:5-13} over at most $L^{2A_1+1}$ small-conductor characters and dividing by $\varphi(H)$ yields $\ll_A L^{-A}/H$, since $H/\varphi(H)\ll\log\log(3H)\ll\log L$ with a harmless constant in the bounded-$H$ cases.

\begin{proposition}[General affine major arcs]\label{prop:hybrid-major}
Under $\mathrm H1$, with the preceding fixed choices, the
affine comparison square has bound
$|Q_Y^{\rm maj}|\ll_A XYL^{-A}$ at every fixed precision.
\end{proposition}
\begin{proof}
Combining both conductor ranges gives


\begin{equation}\label{v2:5-14}
\frac1{\varphi(H)}\sum_{\chi\bmod H}\int_{|t|\le2\mathcal T L^{B_1}}
 |B_{\chi\psi\psi'_b}(t)M_{\chi\psi}(t)N_{\chi\psi}(t)|^2dt
\ll_A L^{-A}/H.
\end{equation}


The primed endpoint, coefficient conjugations, translated intervals, and reversed signs of the frequency twists satisfy the same estimates. Cauchy in $t,\chi$, then the $s$ integration and the bounded-denominator character/arc sums in Section~\ref{subsec:arc-reduction}, cost only fixed logarithmic powers. Equations \eqref{v2:5-5} and \eqref{v2:5-14} give


\begin{equation}\label{v2:5-15}
|Q_Y^{\rm maj}|\ll_D\frac{YX^2}{\mathcal T H}L^{-D}
\le XYL^{-D},
\end{equation}


because $\mathcal T H=\max(X,H)$. The tail estimate in Section~\ref{subsec:hybrid} supplies the same precision. This is the signed comparison form defined in Proposition~\ref{prop:affine-II}, so the bound follows under $\mathrm H1$. The principal and low-conductor contribution is controlled by \eqref{v2:5-13}.

\end{proof}

\subsection{The small-radical input}\label{subsec:radical}
\begin{inputthm}[Iwaniec, 1974, Theorem 2]\label{input:I}
For $q\ge3$, put $d=\rad(q)$,
$\ell=\log(q(|t|+3))$ and
\begin{equation}\label{eq:Iwaniec}
 \vartheta^{-1}=4\cdot10^4
       \{\log d+(\ell\log(2\ell))^{3/4}\}.
\end{equation}
There is at most one nonprincipal character modulo $q$ and one
zero $\rho_0$ in $\Re\rho_0>1-\vartheta$; if present, the
character is real and the zero real and simple
\cite[Theorem~2, p.~97]{Iwaniec1974}.
\end{inputthm}
\begin{lemma}[Lemma Q]\label{lem:Q}
If $\rad(\GN)\le L^C$ and $k\le L^{A_0}$, every real primitive
character of conductor $q^*\mid k\GN$ has
\begin{equation}\label{eq:Q}
 q^*\le8\rad(k\GN)\le8L^{A_0+C}\le L^{A_1},
 \qquad A_1=A_0+C+1,\quad L\ge8.
\end{equation}
\end{lemma}
\begin{proof}
A real primitive character is a Kronecker symbol with fundamental
discriminant of absolute value $m$, $4m$ or $8m$, with $m$ odd and
squarefree \cite[Chapter~5]{Davenport}. Hence
$q^*\mid8\rad(q^*)\mid8\rad(k\GN)$.
\end{proof}
\begin{corollary}[Small-radical region]\label{cor:small-radical}
For $\rad(\GN)\le L^C$ and $A_1=A_0+C+1$,
\eqref{eq:Hstar} holds for every fixed
$C_{\mathrm{zf},1},C_{\mathrm{zf},2}$ once
$L\ge L_0(C_{\mathrm{zf},1},C_{\mathrm{zf},2},C,A_0)$.
\end{corollary}
\begin{proof}
For $q^*\mid k\GN$, $q^*>L^{A_1}$, apply
Input~\ref{input:I} with $q=q^*$.
Here $\rad(q^*)\le L^{A_0+C}$ and
$\ell=\log(q^*(|t|+3))\le2L$ at the fixed height $L^{C_{\mathrm{zf},2}}$.
Thus
\begin{equation}\label{eq:Iwaniec-width}
 \begin{split}
 \vartheta^{-1}&\le4\cdot10^4
 \{(A_0+C)\log L+(2L\log(4L))^{3/4}\},\\
 \vartheta&\ge c_I L^{-3/4}(\log L)^{-3/4}
       > C_{\mathrm{zf},1}\log L/L
 \end{split}
\end{equation}
for sufficiently large $L$, with absolute $c_I>0$ of size $10^{-5}$.
A real exceptional character would contradict Lemma~\ref{lem:Q},
so the exceptional case is absent. The strict width comparison
also gives the closed boundary in \eqref{eq:Hstar}.
\end{proof}
The explicit radical dependence is needed because $k$ contributes
radical up to $L^{A_0}$ even when $\GN=2^m$; a prime-power region
alone does not cover these conductors.

\begin{corollary}[The analytic alternatives]\label{cor:alternatives}
Theorem~\ref{thm:energy} supplies \eqref{eq:alpha-star} from the selected
$\mathrm H1^\star$, and therefore from $\mathrm H1$ for every even
$\GN$. In the class $\rad(\GN)\le L^C$ it follows from
Corollary~\ref{cor:small-radical}. For $0<|h|\le L^C$, all
characters have conductor at most $L^{A_0+C}$, but the scale-$1$
Mellin reduction is replaced in Section~\ref{subsec:small-major}
by the reconstructed hybrid small-conductor proof.
\end{corollary}

\subsection{Changes in version 2: radical criteria}\label{subsec:radical-v2}
For use on the larger radical classes, the elementary real-conductor
bound in Lemma~\ref{lem:Q} can be seen directly. At an odd
$p^e$, the units congruent to one modulo $p$ have odd order, so
a real character is trivial on them and factors modulo $p$.
At powers of two, units congruent to one modulo eight are squares;
the real conductor has 2-part at most eight. Thus
$q^*\le8\rad(q^*)\le8\rad(k\GN)$.

In the remainder of this subsection write $N$ for the even integer
$\GN$ (rather than a Vaughan polynomial length) and $L=\log N$.
\paragraph{The $\mathrm{LS}(c)$ criterion.}
Assume $\mathrm{LS}(c)$ as in Theorem~\ref{thm:F}. Fix the desired $\mathrm H1^\star$ width and height constants and the arc-denominator exponent $A_0$, and fix a nonnegative envelope $\eta(L)\to0$. Uniformly for even $N$ with $L=\log N$ and


\begin{equation}\label{v2:12-1}
\log\operatorname{rad}(N)\le\eta(L)\frac{L}{\log L},
\end{equation}


LS(c) and Iwaniec's 1974 Theorem 2 imply the selected $\mathrm H1^\star$, once $N$ exceeds a threshold depending on the envelope and fixed parameters.

\begin{proof} For a primitive conductor $q^*\mid kN$, $k\le L^{A_0}$,


\begin{equation}
\log\operatorname{rad}(q^*)\le\log\operatorname{rad}(N)+A_0\log L
=o(L/\log L)
\end{equation}


uniformly in the class. At the fixed height $|t|\le L^{C_2}$,
$\ell=\log(q^*(|t|+3))\le2L$ for sufficiently large $L$. The inverse of Iwaniec's width is


\begin{equation}
4\cdot10^4\{\log\operatorname{rad}(q^*)+(\ell\log(2\ell))^{3/4}\}
=o(L/\log L).
\end{equation}


Thus the Iwaniec region strictly contains the desired closed strip of width $C_1\log L/L$. Its only possible exception is a real simple zero of a real character. For a primitive real character, Lemma~\ref{lem:Q} gives


\begin{equation}
\log q^*\le\log8+\log\operatorname{rad}(q^*)=o(L/\log L).
\end{equation}


LS(c) therefore puts its real zero strictly farther from one than $C_1\log L/L$, uniformly. For $\mathrm H1^\star$ only conductors above the fixed threshold $L^{A_1}$ need this argument; they eventually exceed $q_0$. This removes the exception and proves the closed zero-free strip. \end{proof}

An explicit envelope is $\eta(L)=1/\log L$, giving


\begin{equation}\label{v2:12-2}
\operatorname{rad}(N)\le\exp\left(\frac{\log N}{(\log\log N)^2}\right).
\end{equation}


For every fixed $a_0<1$, the class $\operatorname{rad}(N)\le\exp((\log N)^{a_0})$ is another example. These classes can include $N=2^m r_m$ with odd primes $r_m\le\exp(m^\delta)$, fixed $\delta<1$, whose radicals need not be bounded by a fixed power of $\log N$.

For a fixed requested width $C_1$, a stronger explicit boundary is also permitted: let


\begin{equation}
\kappa=\frac{\min(c,1/(4\cdot10^4))}{8C_1}.
\end{equation}


If $\log\operatorname{rad}(N)\le\kappa L/\log L$, then for large $L$ the extra $A_0\log L$ and the $3/4$ term can each be bounded by a fixed fraction of $\kappa L/\log L$. Both the Iwaniec width and the real-zero gap exceed the requested width by a fixed strict margin. The exact threshold may depend on $\kappa,A_0,C_1,C_2$. This assertion provides one fixed selected precision; it does not supply arbitrary growing precision from one fixed $\kappa$.

\paragraph{The Landau--Page criterion.}
We use the classical theorem in the form recalled in
\cite[Lemma~7.1]{FGKMT}: for an absolute $c_{\rm LP}>0$,
among primitive conductors at most $Q$ at most one character
has a real zero $\beta>1-c_{\rm LP}/\log Q$; such a character
is real. This is the only part of that paper used here.
Fix $A_0,A_1,C_1,C_2>0$, the parameters of the selected $\mathrm H1^\star$ interface, and choose


\begin{equation}
0<\kappa\le\frac{\min(c_{\rm LP},1/(4\cdot10^4))}{32C_1}.
\end{equation}


For a dyadic scale $X$, put $\ell=\log X$ and


\begin{equation}
R_X=\exp(\kappa\ell/\log\ell),\qquad Q_X=8(2\ell)^{A_0}R_X.
\end{equation}


Let $q_X$ be the conductor of the possible Landau--Page character at $Q_X$, leaving $q_X$ undefined if there is no such character. For every sufficiently large $X$, the selected $\mathrm H1^\star$ holds uniformly for even $X\le N\le2X$ with $\operatorname{rad}(N)\le R_X$, provided either the exceptional character does not exist, or $q_X\le\ell^{A_1}$, or


\begin{equation}\label{v2:12-3}
\frac{q_X}{\gcd(q_X,N)}>(\log N)^{A_0}.
\end{equation}


\begin{proof} Write $L_N=\log N$, so $\ell\le L_N\le2\ell$ for large $X$. For $q^*\mid kN$, $k\le L_N^{A_0}$, one has


\begin{equation}
\log\operatorname{rad}(q^*)\le\kappa\ell/\log\ell+A_0\log(2\ell).
\end{equation}


At $|t|\le L_N^{C_2}$, the logarithm in Iwaniec's theorem is at most $2\ell$ eventually. Since $(2\ell\log(4\ell))^{3/4}=o(\ell/\log\ell)$, the denominator of Iwaniec's width is at most $8\cdot10^4\kappa\ell/\log\ell$ eventually. The resulting width is at least $16C_1\log\ell/\ell$, strictly greater than the requested $C_1\log L_N/L_N$. Thus only a real zero of a primitive real character can remain in the requested closed strip.

Such a character has $q^*\le8\operatorname{rad}(kN)\le Q_X$. Also $\log Q_X\le2\kappa\ell/\log\ell$ eventually. Every character other than the possible Landau--Page character therefore has real-zero distance at least


\begin{equation}
\frac{c_{\rm LP}}{\log Q_X}\ge16C_1\frac{\log\ell}{\ell},
\end{equation}


again a strict margin. If the exceptional character does not exist there is nothing to remove. If $q_X\le\ell^{A_1}$, its conductor is at most $L_N^{A_1}$, so it is outside the conductor range required by $\mathrm H1^\star$. In the remaining case, for positive integers $q,N$,


\begin{equation}
q\mid kN\quad\Longleftrightarrow\quad q/\gcd(q,N)\mid k.
\end{equation}


This follows prime by prime by comparing valuations. Consequently there is a positive integer $k\le L_N^{A_0}$ with $q_X\mid kN$ if and only if $q_X/\gcd(q_X,N)\le L_N^{A_0}$. Condition \eqref{v2:12-3} excludes precisely this possibility. Every required conductor is now covered, proving $\mathrm H1^\star$. All margins are uniform on the stated dyadic class. \end{proof}

\emph{Reflected consequence on $\mathrm H2'$.} The A and D conclusions and their functional rate follow on this dyadic class from Section~\ref{sec:small}, with Iwaniec, Landau--Page, Burgess and the other named classical inputs. Neither $\mathrm H1$ nor LS(c) is required here. This is an explicitly specified exceptional-conductor avoidance class, not an exception-free numerical list: the identity of $q_X$ may be unknown. The scale and the fixed width must be selected before $\kappa$. 

For reflected sign consequences the requested $C_1,C_2$ are the
fixed energy precision required by Theorem~\ref{thm:energy} and
Section~\ref{sec:assembly}; $\kappa$ is chosen afterwards.
This proves both analytic criteria of Theorem~\ref{thm:F}.
The LS criterion supplies every fixed requested precision for
each prescribed envelope; its threshold may depend on that
envelope. At conductor comparable with $N$, a gap $c/\log q$
alone does not give a width $C_1\log\log N/\log N$ or control
complex zeros. The reflected consequences are on the displayed
classes only.

\begin{remark}[A separate contour estimate]\label{rem:other-contour}
Fix $C_t\ge2$. Under \eqref{eq:H1}, formula~(5.15) of LWW,

 \sum_{p\in\mathcal P_1}\frac{\xi_b(p)p^{it'}}p
 \ll L^{C_{\mathrm{prime}}+1}\exp(-L^{a_1}/16)
\end{equation}
uniformly for $|t'|\le\GN^{C_t}$ and nonprincipal $\xi_b$ of
modulus $k\GN\le\GN^2$. Here $C_{\mathrm{prime}}$ is fixed,
depending on $C_t$ and the fixed exponent ranges.
The cited Perron argument uses $\Re s=15/16$ and

For the classical real-zero lower bound

$(1-\beta_1)L^{a_1}$ can tend to zero. It gives no divergence to
infinity, and leaves a possible exceptional conductor in
$(L^{C},\exp(L^{a_1/2})]$ outside this argument.
\end{remark}

\begin{proposition}[Obstruction (EC)]\label{prop:EC}
Let $\chi\bmod q$ be real primitive, with $q\mid\GN$.
For $p\nmid\GN$ one has
\begin{equation}\label{eq:EC-identity}
 \mu(\GN-p)=\chi(-1)\chi(p)\varepsilon(\GN-p)\mu^2(\GN-p),
 \qquad \varepsilon(n)=\prod_{p'\mid n}(-\chi(p')).
\end{equation}
Suppose every split prime ($\chi(p')=1$) at most $\log q$ divides
$\GN$, and at most $\pi(\GN)/20$ primes $p<\GN$ have either
$\chi(p)=1$ or a split prime factor exceeding $\log q$ in $\GN-p$.
Then, for sufficiently large such $\GN$,
\begin{equation}\label{eq:EC}
 \left|\sum_{p<\GN}\mu(\GN-p)\right|\ge0.27\pi(\GN).
\end{equation}
\end{proposition}
\begin{proof}
For squarefree $n$ coprime to $q$,
$\varepsilon(n)=\mu(n)\chi(n)$; for nonsquarefree $n$ both sides of
\eqref{eq:EC-identity} vanish after multiplication by $\mu^2(n)$.
Also $\chi(\GN-p)=\chi(-1)\chi(p)$ when $p\nmid\GN$,
proving the identity.
Outside the stated exceptional set, $\chi(p)=-1$.
No split prime can divide $\GN-p$: a small split prime divides
$\GN$ and would then divide $p$, contrary to $p\nmid\GN$;
a larger split prime is excluded by hypothesis.
Thus $\varepsilon(\GN-p)=1$, and each squarefree such value has
$\mu(\GN-p)=-\chi(-1)$.
Section~\ref{sec:density}, whose proof uses only Siegel--Walfisz,
gives $(A_{\GN}+o(1))\pi(\GN)$ squarefree values uniformly.
Removing the exceptional set loses at most $\pi(\GN)/20$ of them,
and their possible opposite contribution loses another
$\pi(\GN)/20$. The $O(\omega_{\rm all}(\GN))$ nonunit primes are
negligible. Hence the left side of \eqref{eq:EC} is at least
$(0.3739-2/20+o(1))\pi(\GN)>0.27\pi(\GN)$.
\end{proof}
This proposition establishes an obstruction under its split-prime
hypotheses. The classical lower bound
$1-\beta\gg_\varepsilon q^{-\varepsilon}$ does not by itself
exclude that scenario for a large conductor dividing $\GN$.
The possible zero-size condition $(1-\beta)\log\GN\to0$
does not replace the separate split-prime hypotheses in this proposition.
It identifies why the present uniform argument uses
\eqref{eq:Hstar}. In the small-radical class all the real
conductors dividing $k\GN$ are small by Lemma~\ref{lem:Q}.

\begin{remark}[Changes in version 2: the scope of EC]
The split-prime assumptions in Proposition~\ref{prop:EC} remain
separate from a zero assumption. Under $\mathrm H1$, the Perron
argument at $15/16$ in Section~\ref{subsec:short-prime}, applied
up to $\GN$, gives
$\sum_{p<\GN}\chi(p)=O(\GN^{15/16}(\log\GN)^{O(1)})$
uniformly for primitive nonprincipal conductors at most $\GN$.
For a real such character, half the primes are therefore split,
up to $o(\pi(\GN))$; this already excludes EC's bound
$\pi(\GN)/20$. A sufficiently strong selected
$\mathrm H1^\star$ gives the corresponding conclusion through
Lemma~\ref{lem:P}, with small conductors treated by
Siegel--Walfisz. The weaker-input statements apply only on their
specified classes. The LS criterion supplies the required wider
strip on its prescribed radical envelope; the avoidance criterion
excludes $q_X\mid\GN$ by \eqref{eq:F-avoid}.
For D at $z=2$, $V_\GN(2)=1$ and $U_\GN(2)=A_\GN$,
apart from the possible prime $2$ in passing to full roughness.
\end{remark}

\section{Type I and the common divisor}\label{sec:typeI}
\subsection{Finite presieving of the target}\label{subsec:presieve}
\begin{remark}[Changes in version 2: Type I]
This section replaces the version~1 reflected Type~I proof by a
proof for both orientations. Its common-divisor extraction is the
same; the finite presieve and the endpoint/Rankin bounds below
also cover the rough target. The retained exponents $2/5$ and
$1/20$ refer to the large variable, not to the marking-band ceiling.
\end{remark}
Assume $h$ is even and $2\le z\le\exp(L^a)$, $a<1/2$. Let $\mathcal Q=\{r\in\mathbb P:r\le z,r\nmid h\}$; it consists of odd primes. For an integer $R_z\ge0$ put


\begin{equation}
S_{R_z}(n)=\sum_{\substack{e\mid n,\ e\mid P_h(z)\\\omega_{\rm all}(e)\le R_z}}\mu(e),\qquad
E_{R_z}(n)=\sum_{\substack{e\mid n,\ e\mid P_h(z)\\\omega_{\rm all}(e)=R_z+1}}1.
\end{equation}


The binomial identity $\sum_{j=0}^R(-1)^j\binom{u}{j}=(-1)^R\binom{u-1}{R}$ for $u>0$ gives


\begin{equation}\label{v2:6-1}
|\mathcal R_{z,h}(n)-S_{R_z}(n)|\le E_{R_z}(n).
\end{equation}


For $u\le R$ the error is zero; for $u>R$, $\binom{u-1}{R}\le\binom{u}{R+1}$. This proves the pointwise inequality, including $u=0$. The same bound works in the independent divisor model with local probabilities $1/(r-1)$, because $h$ is coprime to every sieve prime. In particular


\begin{equation}\label{v2:6-2}
\sum_{\substack{e\mid P_h(z)\\\omega_{\rm all}(e)=R_z+1}}\frac1{\varphi(e)}
\le \frac{(\sum_{r\in\mathcal Q}1/(r-1))^{R_z+1}}{(R_z+1)!}.
\end{equation}


The numerator's base is at most $a\log L+O(1)$. Given a fixed desired precision $A$, choose a sufficiently large fixed $C_z$ and $R_z=\lceil C_z\log L\rceil$. Stirling's elementary lower bound $R!\ge(R/e)^R$ makes \eqref{v2:6-2} $O_A(L^{-A})$. Every retained $e$ obeys


\begin{equation}\label{v2:6-3}
e\le z^{R_z+1}\le\exp(O_A(L^a\log L))=x^{o(1)}.
\end{equation}


This is uniform for all $z$ in the stated range, including fixed $z$. Also


\begin{equation}\label{v2:6-4}
c/\log z\le V_h(z)\le1,\qquad V_h(z)\gg_a L^{-a}.
\end{equation}


Indeed omitting factors increases the product, and for odd primes


\begin{equation}
\frac{r-2}{r-1}=(1-1/r)\left(1-\frac1{(r-1)^2}\right).
\end{equation}


Mertens' product estimate and convergence of the second positive product prove \eqref{v2:6-4}. Thus any error $O_A(xL^{-A})$ with $A>a$ is negligible relative to $xV_h(z)$.

\subsection{Type I for both orientations}\label{subsec:general-I}
\begin{proposition}[Uniform Type I]\label{prop:I}
For the function $F$ below, both Type~I terms of
Proposition~\ref{prop:vaughan} have bound $O_A(yL^{-A})$
at every fixed precision, in the A range and, for the rough
target, in the D range. The same is true for the Liouville target.
\end{proposition}
\begin{proof}
Let $f=\mu$ or $\lambda$, and use the bounded function


\begin{equation}
F(n)=f(\varepsilon n+h)\mathcal R_{z,h}(\varepsilon n+h)W(\varepsilon n+h)\,1_{\varepsilon n+h\ge1}.
\end{equation}


For the unrough version omit $\mathcal R_{z,h}$. On a prime dyad of scale $y\ge x/L^B$, the assertion to prove is $T_1(F),T_2(F)\ll_A yL^{-A}$ at each fixed precision.

Write the outer Type I variable as $q\le y^{2/5}$ with $R_Z(q)=1$ and coefficient $|\alpha_q|\le O(L)$. Discard $q$ having a common prime with $h$. Such a prime is greater than $Z$, and summing reciprocal multiples costs $O(yL^3\omega_{\rm all}(h)/Z)$; the harmless endpoint counts are of smaller power size. This is negligible at every fixed logarithmic precision. Separate $k$ odd first, and expand the factors at odd primes in the inner roughness $R_Z(k)$ at depth $R_I=\lceil C_I\log L\rceil$, obtaining odd squarefree $r\le Z^{R_I+1}=x^{o(1)}$ with $r\mid k$. Keep $k$ odd. Expand $W$ at depth $R_W=\lceil C_W\log L\rceil$ as


\begin{equation}
W(n)=\sum_{d\mid n}c_d+\text{remainder},
\end{equation}


where $d$ is squarefree, uses only selected band primes, $d=\exp(O_K(R_WL^\theta))=x^{o(1)}$, $\sum_d|c_d|/d\ll_{K,\rho}1$, and the remainder is bounded by a positive finite divisor sum with reciprocal coefficient mass $O_{K,\rho}(1/R_W!)$. Here is the finite expansion used at this interface. In a single
band, writing $u=\omega_i(n)$ and $V=V_i$, Taylor's formula for
$(1-t)^{u-1}$ at $t=0$ gives
\begin{equation}\label{eq:finite-taylor}
 \frac{u\rho^{u-1}}V
 =\frac1V\sum_{j=0}^{R_W-1}(j+1)(\rho-1)^j
   \sum_{\substack{d\mid n,\ p\mid d\Rightarrow p\in\mathcal P_i\\
                    d\text{ squarefree},\ \omega_{\rm all}(d)=j+1}}1
   +\mathcal E_i(n).
\end{equation}
Indeed $u\binom{u-1}{j}=(j+1)\binom{u}{j+1}$, and
\begin{equation}
 |\mathcal E_i(n)|\le
 \frac{(R_W+1)(1-\rho)^{R_W}}V
 \sum_{\substack{d\mid n,\ p\mid d\Rightarrow p\in\mathcal P_i\\
                  d\text{ squarefree},\ \omega_{\rm all}(d)=R_W+1}}1.
\end{equation}
The reciprocal coefficient mass of this positive remainder is
$O(1/R_W!)$, also with $1/\varphi(d)$ in place of $1/d$,
because both band sums are eventually bounded above by one.
The absolute reciprocal mass of a retained single-band polynomial
is bounded by a fixed constant. Multiplying the $K$ expansions
and telescoping their errors proves the assertions for $W$.
This supplies directly the finite identity corresponding to
LWW (2.17) and its Type~I truncation.

Insert \eqref{v2:6-1} for the target roughness. The sieve divisor $e$ and list divisor $d$ are coprime, since every band prime exceeds $z$, and $D=de$ is squarefree and coprime to $h$. All of its primes are below $Z$. A term with $(r,D)>1$ is impossible: a common prime divides both $qk$ and $\varepsilon qk+h$, hence divides $h$, contrary to $(D,h)=1$. For the other terms the conditions $r\mid k$, $D\mid\varepsilon qk+h$, and $k$ odd give one class modulo $2rD$, because $(q,D)=1$. Their count is


\begin{equation}\label{v2:6-5}
O(y/(qrD)+1).
\end{equation}


Use \eqref{v2:6-5} for every finite Taylor or Bonferroni error. The reciprocal mass in $r$ is $O(\log Z)$, in $e$ is $O(\log z)$, and in $d$ is bounded. A tail of depth $C\log L$ has any requested fixed logarithmic saving by \eqref{v2:6-2} or the Taylor factorial bound. All $+1$ terms total $x^{2/5+o(1)}$, since the total absolute coefficient mass and the number of sieve divisors are $x^{o(1)}$. Choose the depths to absorb the fixed logarithmic factors. This proves the required truncation error estimate without summing over divisors of the full primorial.

Now put $k=rj$ and $g=(r,h)$. The integer $g$ is odd and squarefree, $g\mid\varepsilon qrj+h$, and $(D,g)=1$. For M\"obius, the exact local-factor identity gives


\begin{equation}\label{v2:6-6}
\mu(Dg v)=\mu(Dg)\sum_{\substack{b\mid v\\p\mid b\Rightarrow p\mid Dg}}\mu(v/b).
\end{equation}


Here the displayed support means every prime dividing $b$ divides $Dg$; arbitrary prime powers in $b$ are allowed. At a prime dividing $Dg$, the two potentially nonzero terms cancel when that prime divides $v$, proving the identity prime by prime. Thus write


\begin{equation}\label{v2:6-7}
\varepsilon qrj+h=Dgbt,\qquad
t\equiv(h/g)(Db)^{-1}\pmod{qr/g}.
\end{equation}


The inverse exists: $D$ is disjoint from $qr$, while the primes of $g$ have been removed from the squarefree $r$ and are below every prime dividing $q$. Also $(h/g,qr/g)=1$, since $q$ was made coprime to $h$ and $(r/g,h)=1$. Hence \eqref{v2:6-7} is reduced. There is no extra requirement $(g,Db)=1$. When $h$ is even, $j$ odd is equivalent to $t$ odd, giving a reduced class modulo $2qr/g$. The positive interval for $t$ is contained in


\begin{equation}\label{v2:6-8}
[1,Y_0],\qquad Y_0=(|h|+3y)/(Dgb).
\end{equation}


The sign $\varepsilon$ changes the order of the interval endpoints, not its modulus or reducedness.

For $Dgb\le y^{1/20}$, one has $Y_0\ge y^{19/20}$ and $\log Y_0\asymp L$. Moreover $2qr/g\le y^{2/5+o(1)}$, below the Bombieri–Vinogradov level $Y_0^{1/2}L^{-B_{\rm BV}}$ for every fixed $B_{\rm BV}$, once $x$ is sufficiently large. For fixed $g,D,b$, group by $qr/g$. The coefficient is at most $O(L\tau(qr/g))$; in fact the rough/small-prime factorization is unique. The weighted reduced-class M\"obius estimate follows from ordinary BV by


\begin{equation}
\Big(\sum_{m\le Q}\tau(m)E(Y_0;m)\Big)^2
\le\Big(\sum E(Y_0;m)\Big)\Big(\sum\tau(m)^2E(Y_0;m)\Big),
\end{equation}


using $E(Y_0;m)\ll Y_0/m+1$ and $\sum\tau(m)^2/m\ll L^4$. At sufficiently large fixed BV precision this gives $\ll Y_0L^{-A}$ for the weighted sum. Sum \emph{separately for each $g$}, since the level depends on it, and then use


\begin{equation}
\sum_{g\mid\operatorname{rad}(h)}1/g\ll\log L,\qquad
\sum_{b:\ p\mid b\Rightarrow p\mid Dg}1/b
\le\prod_{p\le Z}(1-1/p)^{-1}\ll\log Z.
\end{equation}


Together with $\sum_e1/e\ll\log z$, $\sum_d|c_d|/d\ll1$, and $(|h|+3y)/y\ll L^{B+C}$, these are only fixed logarithmic losses.

For $Dgb>y^{1/20}$, the factors $Dg=x^{o(1)}$ imply $b>y^{1/25}$ for sufficiently large $x$. Rankin at exponents $1/2$ and $1/20$ gives


\begin{equation}
\sum_{b>y^{1/25}}\frac1b\ll y^{-1/50}L^{O(1)},\qquad
\#\{b\le |h|+3y:p\mid b\Rightarrow p\mid Dg\}
\ll y^{1/20}L^{O(1)}.
\end{equation}


To justify the products, $Dg$ has $O_{K,A}(\log L)$ distinct prime divisors, all odd; each factor $(1-p^{-\eta})^{-1}$ is bounded by a constant depending only on fixed $\eta>0$. Thus the whole factor is $L^{O_{K,A}(1)}$. Counting \eqref{v2:6-7} absolutely gives $O(y/(Dbqr)+1)$. The reciprocal term has the displayed power saving; the endpoint term totals $y^{9/20+o(1)}$. Both absorb every fixed logarithmic loss. This completes the M\"obius Type~I bound. Partial summation for the factor $\log k$ supplies $T_1$ with one additional fixed power of $L$.

For Liouville use $\lambda(Dgt)=\lambda(Dg)\lambda(t)$, so there is no $b$-sum and $Dg=x^{o(1)}$ always lies in the small-divisor branch. The needed Liouville BV follows from


\begin{equation}
\lambda(n)=\sum_{b^2\mid n}\mu(n/b^2).
\end{equation}


For a reduced class modulo $m$, $(b,m)=1$ and the inner M\"obius class is $c\bar b^2\pmod m$. Truncate at $b\le L^{A+2}$. Use M\"obius BV at scale $Y_0/b^2$ with a larger fixed logarithmic level exponent. For the tail summed over $m\le Q$ use $Y_0/(b^2m)+1$, obtaining $O(Y_0L/L^{A+2}+Q\sqrt{Y_0})$; choose $Q\le\sqrt{Y_0}L^{-A-2}$ in addition to the head's BV restriction. The same Cauchy/divisor-square estimate gives the $\tau$-weighted version. This proves the Liouville Type I bound.

For the unrough A statement with \emph{odd} $h$, take $e=1$. Equation \eqref{v2:6-7} forces even $t$. Write $t=2t'$; for M\"obius use $\mu(2t')=-\mu(t')1_{t'\text{ odd}}$ and a reduced class modulo $2qr/g$. For Liouville use $\lambda(2t')=-\lambda(t')$, with no parity restriction, and modulus $qr/g$. All inverses exist because these moduli apart from the explicit 2 are odd. This supplies the odd-shift Type I case.

\end{proof}

\section{Removal of the auxiliary weights}\label{sec:weights}
\subsection{Positive moments on the eligible prime interval}\label{subsec:relative-moments}
In this section $L=\log x$ and $I$ is the eligible interval in
$[1,x]$. For D retain its even-shift and fixed-$a<1/2$ conditions.
For A omit every low-prime sieve factor and put $V=1$; no parity
assumption on $h$ is then needed.
\begin{remark}[Changes in version 2: relative measures]
The version~1 reduced-class mean and fixed-number variance are
used below with a finite positive presieve. The variance retains
the factor $V_h(z)$, and the later Cauchy inequality uses the rough
prime measure. The last subsection replaces the fixed-$b$ limit
by a uniform construction with growing $b_x$.
\end{remark}
Let $I$ be the interval of eligible primes in $[1,x]$; endpoints depend on $\varepsilon,h$. Put $\Theta_I=\sum_{p\in I}\log p$. For $F=1$, $W_j$, or $W_jW_k$ with $j\ne k$, the assertion is


\begin{equation}\label{v2:6-9}
\sum_{p\in I}(\log p)\mathcal R_{z,h}(\varepsilon p+h)F(\varepsilon p+h)
=V_h(z)m_F\Theta_I+O_A(xL^{-A}),
\end{equation}


where $m_1=1$, $m_{W_j}=m_j$, and $m_{W_jW_k}=m_jm_k$. The bound is uniform in $z,h$ and in both endpoints, for each fixed band array.

For completeness, the independent local divisor model assigns $r\mid\varepsilon p+h$ probability $1/(r-1)$ at any odd $r\nmid h$. The CRT proves independence at distinct primes, because the affine coefficient $\varepsilon$ is a unit and the required residue is reduced. Every band avoids $h$, and different bands are disjoint. Consequently


\begin{equation}\label{v2:6-10}
m_j=\prod_{i=1}^K\left[\frac1{V_i}\sum_{r\in\mathcal P_i}\frac1{r-1}
\prod_{\substack{q\in\mathcal P_i\\q\ne r}}
\left(1-\frac{1-\rho}{q-1}\right)\right]
=2^{K(\rho-1)}+o(1).
\end{equation}


The rough primes and all band primes are disjoint, so the local-model average of $\mathcal R_{z,h}F$ is exactly $V_h(z)m_F$.

To pass to primes, use the finite presieve \eqref{v2:6-1} and the finite Taylor divisor expansions from Section~\ref{subsec:general-I}, now with at most $2K$ bands. A retained product modulus is $ed=x^{o(1)}$ and coprime to $h$; its prime class is $p\equiv-\varepsilon h\pmod{ed}$. The modulus uniquely determines its low-prime factor $e$ and band factor $d$. The Taylor coefficient is bounded by a fixed power of $\log L$, so BV with a larger fixed logarithmic precision applies to the sum over all retained moduli, and is uniform in the two interval endpoints. Prime powers and the few primes dividing $h$ cost $x^{1/2+o(1)}$ and $x^{o(1)}$, respectively, after all finite coefficient sums.

The Taylor remainder has reciprocal coefficient mass $O(1/R_W!)$ also with $1/\varphi(d)$ instead of $1/d$, since each band's sum $\sum1/(r-1)$ tends to $\log2<1$. The roughness remainder multiplied by the absolute truncated Taylor polynomial has local-model mass bounded by \eqref{v2:6-2} times $\sum_d|c_d|/\varphi(d)\ll_{K,\rho}1$. Apply BV to these positive finite error sums as well. Choosing the truncation depths and BV precision proves errors $O_A(xL^{-A})$. Replacing the interval length by $\Theta_I$ uses the endpoint-uniform PNT error, again with enough logarithmic precision. This establishes \eqref{v2:6-9}.

Each $W_j/m_j$ is bounded by a constant depending on $K,\rho$ but not on $b$, after increasing the threshold for the fixed array. Thus its square is at most a fixed constant times itself. With


\begin{equation}
\mathcal W_b(n)=b^{-1}\sum_{j=1}^bW_j(n)/m_j,
\end{equation}


expand the square and apply \eqref{v2:6-9}. The cross terms for $j\ne k$ are $V_h(z)\Theta_I$ after normalization. The first moments and constant term have the same mean. The diagonal terms are bounded as just described. Hence


\begin{equation}\label{v2:6-11}
\sum_{p\in I}(\log p)\mathcal R_{z,h}(\varepsilon p+h)
 |\mathcal W_b(\varepsilon p+h)-1|^2
\le \frac{C_{K,\rho}}b V_h(z)\Theta_I+O_{A,b}(xL^{-A}).
\end{equation}


The constant $C_{K,\rho}$ is independent of $b$ and of the band locations. This is the relative variance needed for D; the factor $V_h(z)$ has not been discarded.

\begin{proposition}[Normalized fixed-number weight removal]\label{prop:weights}
For any fixed $b$, normalized weights
$\mathcal W_b=b^{-1}\sum_j W_j/m_j$ satisfy
\eqref{v2:6-11}. For a bounded target, Cauchy--Schwarz with
measure $(\log p)\mathcal R_{z,h}(\varepsilon p+h)$ gives
\begin{equation}\label{eq:weight-removal}
 \left|\sum_{p\in I}(\log p)f(\varepsilon p+h)
 \mathcal R_{z,h}(\varepsilon p+h)(1-\mathcal W_b(\varepsilon p+h))\right|
 \ll xV_h(z)b^{-1/2}+o_b(xV_h(z)).
\end{equation}
For the unrough statement replace $V_h(z)$ by one. This includes
the version~1 fixed-$b$ conclusion when $x=h=\GN$ and
$\varepsilon=-1$.
\end{proposition}
\begin{proof}
The total mass of this measure is
$V_h(z)\Theta_I+O_A(xL^{-A})$. Multiply its square root by
the square root of \eqref{v2:6-11}; use
$\Theta_I\ll x$ and $V_h(z)\gg L^{-a}$ with a sufficiently
large fixed precision. The constant is independent of $b$.
\end{proof}
\subsection{A common threshold for separated band families}\label{subsec:band-uniform}
Fix $K$, $\alpha>1/10$, an admissible ceiling $\theta$, and $\delta_b>0$. Permit the individual exponents to vary with $x$ subject to $a_1\ge\alpha$, $a_K\le\theta$ and $a_{i+1}-a_i\ge\delta_b$. All weighted estimates above have a common threshold and implied constant on this family.

The dependence on band locations is controlled as follows. Band masses converge uniformly because every lower endpoint exceeds $\exp(L^\alpha)$; puncturing costs $O(L\exp(-L^\alpha))$. Maximum product sizes, root moduli, coefficient masses, and boundary derivatives use only the fixed ceiling. Minimum-prime and overlap bounds use only $\alpha$. The sole hierarchy comparison in the second good-state test is bounded explicitly by


\begin{equation}
\log\#\{DZ\}\le C_K\log L+C_K L^{a_{i_*}-\delta_b},
\end{equation}


so its ratio to $L^{a_{i_*}}$ tends to zero uniformly. All finite memory identities and the positive bound in Section~\ref{subsec:retune} have constants depending only on the fixed normalization bounds and operator parameters. Prime-polynomial contours and mean values are uniform in their endpoints. In Type I and the auxiliary moments, at most $K$ or $2K$ groups occur in any one estimate; their coefficient bounds and factorial tails are uniform. Cross moments between two different copies require disjoint bands, not a fixed gap between those copies. The reflected weak-input interface has the same uniform large-value margin, as checked in the last item in Section~\ref{subsec:retune}. This accounts for every use of band locations in the cited arguments.

\subsection{Packing and the explicit rate}\label{subsec:packing}
Use $a=0$ for the unrough statement. After fixing $K$, choose an admissible $\theta>\max(a,1/10)$ and put


\begin{equation}
g=\frac{\theta-\max(a,1/10)}{4K+4},\quad
\alpha_i=\max(a,1/10)+(2i-1)g,\quad
b_x=\left\lfloor\frac{g\log L}{\log4}\right\rfloor.
\end{equation}


For $1\le j\le b_x$ choose the $i$th band by logarithmic endpoints


\begin{equation}\label{v2:10-1}
t_{i,j}=4^{j-1}L^{\alpha_i},\qquad
\exp(t_{i,j})<p\le\exp(2t_{i,j}),\quad p\nmid h.
\end{equation}


Its exponent $a_{i,j}=\alpha_i+(j-1)\log4/\log L$ lies in $[\alpha_i,\alpha_i+g]$. Each weight has consecutive exponent gaps exactly $2g$. Across different $i$, the neighborhoods are separated by $g$; within one $i$, adjacent logarithmic lower endpoints have ratio four, exceeding the previous band's upper/lower ratio two. Thus all $b_xK$ bands are disjoint for sufficiently large $x$, and each weight belongs to one fixed separated family. The marking primes exceed the rough threshold. Only the number of copies grows; $K$ and every analytic precision are fixed.

Apply \eqref{v2:6-11} with uniform errors to these copies. The $b_x^2$ mixed errors are divided by $b_x^2$; averaging the uniformly small marked sums also loses no factor $b_x$. Since $b_x\asymp\log L$, the positive-measure Cauchy argument gives


\begin{equation}\label{v2:10-2}
\left|\sum_{p\in I}(\log p)f(\varepsilon p+h)\mathcal R(p)\right|
\ll \frac{xV}{\sqrt{\log L}}+xL^{-A}+\sqrt{Vx\cdot xL^{-A}},
\end{equation}


where $V=1$ or $V_h(z)$. With $V\gg L^{-a}$, a sufficiently large \emph{fixed} $A$ absorbs the final terms. Low primes and removal of the prime logarithm contribute relative error $O(L^{a-1}\log L+L^{a+1-B})$, smaller than $(\log L)^{-1/2}$ for every fixed $a<1/2$ and sufficiently large fixed $B$.

Consequently the final errors are


\begin{equation}\label{v2:10-3}
O_C\left(\frac{\pi(x)}{\sqrt{\log\log x}}\right)\quad\text{in A},\qquad
O_{a,C}\left(\frac{V_h(z)\pi(x)}{\sqrt{\log\log x}}\right)\quad\text{in D}.
\end{equation}


The sign formulas use the exact main terms already computed, and have the same errors. For M\"obius relative to the whole rough population, replacing $U_h(z)$ by $V_h(z)$ adds $O(1/z)$ to the relative error. Constants and thresholds remain ineffective through the named classical inputs. This is an explicit functional rate, not a numerical effective estimate.

The original fixed-$b$ argument alone did not prove \eqref{v2:10-3}. The additional uniformity check and explicit packing are essential. No prime distribution estimate is applied at the full product of all marking primes; each analytic moment uses only its finite truncated divisors from at most two weight copies.

\section{Squarefree values}\label{sec:density}
\begin{proposition}[Uniform squarefree density]\label{prop:squarefree}
For even $\GN\to\infty$,
\begin{equation}\label{eq:squarefree-weighted}
 \sum_{\substack{p<\GN\\p\nmid\GN}}\mu^2(\GN-p)\log p
 =A_{\GN}\GN+O_A(\GN L^{-A}),
 \qquad A_{\GN}\ge0.3739.
\end{equation}
Consequently
\begin{equation}\label{eq:squarefree-count}
 \#\{p<\GN:\mu^2(\GN-p)=1\}=(A_{\GN}+o(1))\pi(\GN),
\end{equation}
uniformly in even $\GN$.
\end{proposition}
\begin{proof}
Use $\mu^2(n)=\sum_{d^2\mid n}\mu(d)$.
If $p\nmid\GN$ and $d^2\mid\GN-p$, then $(d,\GN)=1$.
For $d\le L^{A+2}$, Siegel--Walfisz in the reduced class
$p\equiv\GN\pmod{d^2}$, at a sufficiently large fixed precision,
gives main term
\begin{equation}\label{eq:squarefree-series}
 \GN\sum_{(d,\GN)=1}\frac{\mu(d)}{\ph(d^2)}=A_{\GN}\GN,
\end{equation}
with extension error $O(\GN L^{-A-1})$.
The tail has the finite range required by $\GN-p>0$:
\begin{equation}\label{eq:squarefree-tail}
 \sum_{L^{A+2}<d\le\sqrt{\GN}}(\GN/d^2+1)L
 \ll\GN L^{-A-1}+\sqrt{\GN}L.
\end{equation}
The last term is absorbed in $O_A(\GN L^{-A})$.
This is LWW Proposition~6.1 and (6.3), pp.~46--47, with the size
and reduced-class checks for the varying shift supplied here.
Primes dividing $\GN$ cost $O(\omega_{\rm all}(\GN)L)=O(L^2)$.
To pass to ordinary counts, use Lemma~\ref{lem:unweight} below
with the bounded target $\mu^2(\GN-p)$.
The same progression proof permits any endpoint $y\le\GN$.
\end{proof}

\begin{lemma}[Removing the prime logarithm]\label{lem:unweight}
For any bounded target $|f(p)|\le1$ on primes below $\GN$, possibly
depending on $\GN$, and fixed $B>2$,
\begin{equation}\label{eq:unweight}
 \sum_{p<\GN}f(p)
 =\frac1L\sum_{p<\GN}f(p)\log p
 +O_B\left(\frac{\GN\log L}{L^2}+\frac{\GN}{L^B}\right).
\end{equation}
\end{lemma}
\begin{proof}
For $p>\GN/L^B$, $\log p/L=1+O_B(\log L/L)$.
Chebyshev's bound gives $O(\GN/L)$ such primes.
For the remaining primes bound both sums by their number, at most
$\GN/L^B$; this proves the identity without varying the target
sequence in a partial-summation limit.
\end{proof}

\subsection{Changes for general \texorpdfstring{$h$}{h}: positive eligible counts}\label{subsec:general-density}
This assertion uses only the classical inputs. Uniformly for the entire A range,


\begin{equation}\label{v2:7-2}
\#\{p\in\mathcal P_{\varepsilon,h}(x):\mu^2(\varepsilon p+h)=1\}
=A_h\Pi_{\varepsilon,h}(x)+o(\pi(x)).
\end{equation}


Remove primes dividing $h$, at most $O(L)$ terms. For the others, $d^2\mid\varepsilon p+h$ forces $(d,h)=1$. Truncate the square-divisor expansion at $d\le L^D$; the unweighted tail is at most


\begin{equation}
\sum_{L^D<d\le\sqrt{x+|h|}}(x/d^2+1)
\ll xL^{-D}+\sqrt{xL^C},
\end{equation}


which is $o(\pi(x))$ for fixed $D>2$. For the head, apply Siegel–Walfisz to $p\equiv-\varepsilon h\pmod{d^2}$, a reduced class, at both endpoints of the eligible interval. The modulus is a fixed power of $\log x$; arbitrarily large fixed saving absorbs the $L^D$ divisors. Replace the interval's logarithmic-integral mass by its actual prime count using the same endpoint-uniform PNT estimate. Extending the absolutely convergent series gives


\begin{equation}
\sum_{(d,h)=1}\frac{\mu(d)}{\varphi(d^2)}
=\prod_{r\nmid h}(1-1/(r(r-1)))=A_h,
\end{equation}


with an admissible tail. Restoring the removed primes changes only $O(L)$ terms. This proves \eqref{v2:7-2}, including odd $h$ and empty eligible intervals.


The same argument with a larger fixed truncation exponent gives
any requested error $O_A(xL^{-A})$ in the unweighted count.
In particular its error is smaller than $\pi(x)E(x)$.
\subsection{Rough and squarefree-rough populations}\label{subsec:rough-density}
The positive population formula follows from \eqref{v2:6-9} with $F=1$ and the same removal of $\log p$:


\begin{equation}\label{v2:6-14}
\#\{p\in\mathcal P_{\varepsilon,h}(x):P^-(\varepsilon p+h)>z\}
=V_h(z)\Pi_{\varepsilon,h}(x)+o(V_h(z)\pi(x)).
\end{equation}


For squarefree rough values use $\mu^2(n)=\sum_{d^2\mid n}\mu(d)$. Remove the $O(\omega_{\rm all}(h))$ primes dividing $h$. On the remainder any contributing $d$ is coprime to $h$, and roughness forces $(d,P_h(z))=1$. Truncate at $d\le L^D$ with $D$ a sufficiently large fixed number. Since $1\le n\le x+|h|\ll xL^C$, the weighted discarded tail is


\begin{equation}\label{v2:6-15}
\ll L\sum_{L^D<d\le\sqrt{x+|h|}}(x/d^2+1)
\ll xL^{1-D}+\sqrt{xL^C}\,L.
\end{equation}


It is smaller than any prescribed fixed logarithmic error by choosing $D$ suitably. For the head expand roughness using \eqref{v2:6-1}. Moduli are $d^2e=x^{o(1)}$, with $(d,e)=1$, and the reduced prime residue is $-\varepsilon h$. The modulus has the unique decomposition into its square part $d^2$ and its squarefree part $e$; thus the sum of BV discrepancies has no uncontrolled multiplicity. Positive Bonferroni tails are bounded by \eqref{v2:6-2} times $\sum_d1/\varphi(d^2)\ll1$. Restore the finitely many omitted primes only after bounding the total finite divisor coefficient mass by $x^{o(1)}$; their error is negligible.

The main series factors as


\begin{equation}
V_h(z)\sum_{\substack{(d,hP_h(z))=1}}\frac{\mu(d)}{\varphi(d^2)}=U_h(z).
\end{equation}


Extending the $d$-series has error $O(L^{-D+1})$, since $d/\varphi(d)\ll\log(2d)$ suffices. Consequently


\begin{equation}\label{v2:6-16}
\#\{p\in\mathcal P_{\varepsilon,h}(x):P^-(\varepsilon p+h)>z,\ \mu^2(\varepsilon p+h)=1\}
=U_h(z)\Pi_{\varepsilon,h}(x)+o(V_h(z)\pi(x)).
\end{equation}


The proofs of \eqref{v2:6-14} and \eqref{v2:6-16} give arbitrarily strong fixed-logarithmic weighted remainders before removing the prime logarithm. After that removal their relative errors are bounded by $O(L^{a-1}\log L)$ plus arbitrarily small fixed powers of $L^{-1}$. They are therefore smaller than $E(x)$ throughout the final fixed-$a<1/2$ range, not merely unspecified little-o errors.

For $\kappa=\pm1$, the identities $1_{\mu(n)=\kappa}=(\mu^2(n)+\kappa\mu(n))/2$ and $1_{\lambda(n)=\kappa}=(1+\kappa\lambda(n))/2$, multiplied by the roughness indicator, prove the sign formulas from \eqref{eq:rough-main}, \eqref{v2:6-14}, and \eqref{v2:6-16}, at the precision of Section~\ref{subsec:packing}. Finally


\begin{equation}
0\le1-U_h(z)/V_h(z)\le\sum_{r>z}\frac1{r(r-1)}\ll1/z,
\end{equation}


uniformly in $h$. This is the exact reason both signs have density $1/2$ within the rough population when $z\to\infty$ and the eligible interval has a fixed positive proportion of the primes.

\subsection{The progression local factors}\label{subsec:progression-density}
Fix $C,C_q,A>0$. Uniformly for $0<|h|\le x(\log x)^C$, $q\le(\log x)^{C_q}$, $(a_0,q)=1$, and $\varepsilon=\pm1$, let $\Pi_{q,a_0;\varepsilon,h}(x)$ count the eligible primes additionally satisfying $p\equiv a_0\pmod q$. Then


\begin{equation}\label{v2:13-1}
\#\{p\in\mathcal P_{\varepsilon,h}(x):p\equiv a_0\ (q),\ \mu^2(\varepsilon p+h)=1\}
=\mathfrak A(q,a_0;\varepsilon,h)\Pi_{q,a_0;\varepsilon,h}(x)+O_{A,C,C_q}(x(\log x)^{-A}),
\end{equation}


where the product, interpreted as zero if any forced-square factor occurs, is


\begin{equation}\label{v2:13-2}
\mathfrak A(q,a_0;\varepsilon,h)=
\prod_{r\nmid qh}\left(1-\frac1{r(r-1)}\right)
\prod_{\substack{r\Vert q\\r\mid\varepsilon a_0+h}}\left(1-\frac1r\right)
\prod_{\substack{r^2\mid q\\r^2\mid\varepsilon a_0+h}}0.
\end{equation}


\begin{proof} Write $L=\log x$ and delete the $O(\log(2|h|))=O_C(L)$ primes dividing $h$. On the remaining primes, a square divisor $d^2\mid\varepsilon p+h$ necessarily has $(d,h)=1$. Truncate $\mu^2(n)=\sum_{d^2\mid n}\mu(d)$ at $d\le L^D$, with a fixed $D>A+C_q+3$. The absolute discarded sum is at most


\begin{equation}
\sum_{L^D<d\le\sqrt{x+|h|}}(x/d^2+1)
\ll xL^{-D}+\sqrt{xL^C}=O_{A,C,C_q}(xL^{-A}).
\end{equation}


For each retained squarefree $d$, the simultaneous conditions $p\equiv a_0\pmod q$ and $p\equiv-\varepsilon h\pmod{d^2}$ are either incompatible or specify one class modulo $m_d=\operatorname{lcm}(q,d^2)$. Compatibility means $a_0\equiv-\varepsilon h\pmod{\gcd(q,d^2)}$. When compatible the class is reduced: it is a unit at primes dividing $q$ by $(a_0,q)=1$, and at primes dividing $d$ by $(d,h)=1$. Since $m_d\le L^{C_q+2D}$, endpoint-uniform Siegel–Walfisz applies at arbitrarily high fixed precision. Summing its errors over $d\le L^D$ gives the required error after increasing that precision. A small interval endpoint causes no difficulty: use the trivial bound below $xL^{-B}$ and the usual Siegel–Walfisz estimate above it, with fixed $B$ chosen at the requested precision.

Relative to the prime progression's logarithmic-integral mass, the main coefficient is


\begin{equation}
\sum_{\substack{(d,h)=1\\a_0\equiv-\varepsilon h\ (\gcd(q,d^2))}}
\mu(d)\frac{\varphi(q)}{\varphi(\operatorname{lcm}(q,d^2))}.
\end{equation}


The series is absolutely convergent for fixed $q$. Its tail beyond $L^D$ is at most \[\varphi(q)\sum_{d>L^D}\frac1{\varphi(d^2)}\ll L^{C_q-D+1},\], uniformly in the displayed range. Even multiplying this by $x$ is within the requested error. Compatibility and the totient ratio factor prime by prime. If $r\nmid qh$, selecting $r\mid d$ costs $1/\varphi(r^2)=1/(r(r-1))$. If $r\mid h$, it cannot be selected. If $r\Vert q$, selection is compatible exactly when $r\mid\varepsilon a_0+h$, and then costs $\varphi(r)/\varphi(r^2)=1/r$. If $r^2\mid q$, selection is compatible exactly when $r^2\mid\varepsilon a_0+h$, and then costs one. These are precisely \eqref{v2:13-2}; in particular the coefficient lies in $[0,1]$.

Siegel–Walfisz for the original progression replaces its logarithmic-integral mass by the actual eligible-prime count at the same precision. Restoring the removed primes costs $O_C(L)$ in the original count; their contributions in the finite expansion are at most $O_C(L^{D+1})$, also admissible. This proves \eqref{v2:13-1}. \end{proof}

For $q=9$, $a_0=1$, $\varepsilon=1$, $h=8$, the last product is zero and every eligible shifted value is divisible by 9. Thus keeping $A_h$ as a progression sign-density constant would be false. Formula \eqref{v2:13-1} supplies the correct positive population but does not split it into two signs.

\section{Assembly of the Goldbach statements}\label{sec:assembly}
\begin{proof}[Proofs of Theorems~\ref{thm:G} and~\ref{thm:S}]
Choose $\rho$, $A^*>0$ and then $B>A^*+3$.
Let $C_{\mathrm{coef}}$ be the corresponding fixed coefficient and
smoothing-loss exponent for this choice of $B$.
Complete the parameter choices in the following order:
\begin{equation}\label{eq:parameter-order}
 \begin{gathered}
 \rho\ ;\ A^*\ ;\ B\ ;\ C_{\mathrm{coef}}\ ;\\
 D^*>2(A^*+B+C_{\mathrm{coef}}+c_*)\ ;\ E_0\ ;\ G\ ;\\
 A_0\ ;\ K\ ;\ C^*,C^\dagger,C_0\ ;\ B_0,B_1\ ;\\
 A_{\mathrm{energy}}=D^*+O(A_0+B_1)\ ;\\
 D,C_{\mathrm{zf},1},C_{\mathrm{zf},2},C_3,A_s,A_1\ ;\\
 \text{the fixed precision of Lemma~\ref{input:sifted}}\ ;\ a_i\ ;\ L_0.
 \end{gathered}
\end{equation}
Here $c_*$ is a fixed reserve for scale and dyadic losses, chosen
large enough before the later parameters.
The choices for the moment estimate are those in LWW
Section~4.7, p.~39; $A_0,K$ are independent of the band locations.
The zero-free constants are subsequently defined by
\eqref{eq:A-constants} with $A=A_{\mathrm{energy}}$ and do not
enter $C_{\mathrm{coef}}$ or the inequality for $D^*$.
For any fixed $b$ the $bK$ band exponents can be selected after these
saving exponents; their positions affect thresholds, not these choices.

On each of the $O(\log L)$ dyadic prime intervals, the Vaughan
reduction and Proposition~\ref{prop:I} reduce the weighted prime sum
to Type~II sums $T_Y$ on $O_K(L^\theta)$ product scales.
Propositions~\ref{prop:II} and~\ref{prop:arc-reduction}, followed by
Theorem~\ref{thm:energy}, give
\begin{equation}\label{eq:assembly-square}
 \QQ=\Qmaj+O(\GN YL^{-D^*}),\quad
 |\Qmaj|\ll\GN YL^{-D^*},\quad
 |T_Y|^2\ll(\GN/Y)\QQ\ll\GN^2L^{-D^*}.
\end{equation}
Thus $|T_Y|\ll\GN L^{-D^*/2}$, and the larger fixed budget in
\eqref{eq:parameter-order}, using $\GN/X\ll2L^B$, gives
$T_3(F)\ll xL^{-A^*}$.
For Theorem~\ref{thm:G}, use the selected $\mathrm H1^\star$;
\eqref{eq:H1} supplies it for every fixed selection of constants.
For Theorem~\ref{thm:S}, use Corollary~\ref{cor:small-radical} with
$A_1=A_0+C+1$ and the Burgess input in Theorem~\ref{thm:energy}.
In either case
\begin{equation}\label{eq:weighted-conclusion}
 \sum_{x<p\le2x}(\log p)\mu(\GN-p)W(\GN-p)=O(xL^{-A^*}),
 \qquad
 \sum_{p<\GN}(\log p)\mu(\GN-p)W(\GN-p)=o(\GN).
\end{equation}
The omitted small primes have the negligible bounded-weight cost
stated in Remark~\ref{rem:ranges}.

For each fixed $b$, every normalized weighted limit in
\eqref{eq:weight-removal} is zero.
Then let $b\to\infty$ to get
$\sum_{p<\GN}\mu(\GN-p)\log p=o(\GN)$.
Lemma~\ref{lem:unweight} and $\pi(\GN)\sim\GN/L$ give the initial
conclusion of \eqref{eq:G}.
Finally, for $\kappa=\pm1$,
\begin{equation}\label{eq:mu-sign-identity}
 \#\{p<\GN:\mu(\GN-p)=\kappa\}
 =\tfrac12\left(\#\{p<\GN:\mu^2(\GN-p)=1\}
                 +\kappa\sum_{p<\GN}\mu(\GN-p)\right).
\end{equation}
Proposition~\ref{prop:squarefree} proves the sign densities.

All comparisons use $\log x=L+O_B(\log L)$ on the stated range.
Puncturing, the large common-prime discard, the power errors in
Lemma~\ref{lem:R} and the $O(\log L)$ dyadic summation are uniform
in even $\GN$. Constants depend on the fixed bands, $K,\rho,b$,
and requested precisions, and in the small-radical case on $C$.
The limits are ineffective through Siegel--Walfisz and
Lemma~\ref{input:sifted}. A fixed $\mathrm H1^\star$ supplies the selected
fixed energy precision; the half-plane and the small-radical region
supply every fixed precision needed in this order.
\end{proof}

\subsection{Changes in version 2: assembly of A, C and D}\label{subsec:affine-assembly}
\begin{proof}[Proofs of Theorems~\ref{thm:affine}, \ref{thm:rate} and~\ref{thm:rough}]
Use $L=\log x$, $\Delta=-\varepsilon h$, $H=|h|$.
For D fix $a<1/2$ and the retuned exponents of
Section~\ref{subsec:retune}; for A one may use the same exponents
with any fixed auxiliary $a\in(0,1/10]$ and omit the rough sieve.
Choose a sufficiently large fixed lower-prime cutoff $B$ and
smoothing precision, and then the required saving $D_*$ so that
it absorbs the coefficient, Mellin, dyadic and $L^{B+C}$ losses.
Choose $E_0,G,A_0,K$ in the order of
\eqref{eq:parameter-order}, with $A_0>G+3$ and
$K>2(G+3)/j_*$. These choices precede the band locations.
Next choose the fixed transform exponents, conductor cutoff and
small-conductor, Type~I and moment precisions. Finally choose
the separated band family and its packing. For the reflected
branch choose the energy precision and zero-free constants by
\eqref{eq:A-constants} at this stage; they do not enter the
earlier coefficient-loss exponent.

The sifted Vaughan identity, Proposition~\ref{prop:I},
Proposition~\ref{prop:affine-II} and
Proposition~\ref{prop:hybrid-major} give, under $\mathrm H1$
and $\mathrm H2'$,
\begin{equation}\label{eq:affine-assembly-square}
 Q_Y\ll XYL^{-D_*},\qquad
 |T_Y|^2\ll\frac{X+H}{Y}Q_Y
       \ll X(X+H)L^{-D_*}\ll X^2L^{B+C-D_*}.
\end{equation}
The rough indicator transfers to the bounded common cofactor
in Proposition~\ref{prop:affine-square}; it gives precisely the
same upper square. A larger fixed $D_*$ absorbs all subsequent
losses. The number of product scales is $O_K(L^\theta)$ and
the number of retained prime dyads is $O_B(\log L)$.
For each fixed requested $A$, and with $B>A+2$, this proves
uniformly for every permitted single weight
\begin{equation}\label{v2:6-12}
 \sum_{p\in\mathcal P_{\varepsilon,h}(x)}(\log p)
 f(\varepsilon p+h)\mathcal R(\varepsilon p+h)W_j(\varepsilon p+h)
       \ll_A xL^{-A},
\end{equation}
where $\mathcal R=1$ for A and $\mathcal R=\mathcal R_{z,h}$
for D. The omitted low primes cost $O(xL^{1-B})$.

Section~\ref{subsec:band-uniform} supplies one threshold for
these estimates, and Section~\ref{subsec:packing} gives
$b_x\asymp\log L$ disjoint copies. Apply its Cauchy bound
with the rough prime measure when relevant. A sufficiently large
fixed $A,B$ yields
\begin{equation}\label{eq:affine-weighted-main}
 \left|\sum_{p\in\mathcal P_{\varepsilon,h}(x)}(\log p)
 f(\varepsilon p+h)\mathcal R(\varepsilon p+h)\right|
       \ll xV/\sqrt{\log L},
 \quad V=1\text{ or }V_h(z).
\end{equation}
For $p>x/L^B$, replace $\log p$ by $L$ with relative error
$O_B(\log L/L)$ using Chebyshev. Low primes contribute
$O(x/L^B)$. Relative to $V\pi(x)$ their total error is
$O(L^{a-1}\log L+L^{a+1-B})$ in D and the same expression
with $a=0$ in A. It is smaller than $(\log L)^{-1/2}$.
The $O(\log x)$ primes dividing $h$ can be restored on this
scale; on the others punctured and full roughness agree.
This proves A and D and the rate C.

The positive counts in Section~\ref{sec:density} and the exact
identities
\begin{equation}\label{eq:general-signs}
 \ind_{\mu(n)=\kappa}=\frac{\mu^2(n)+\kappa\mu(n)}2,
 \qquad
 \ind_{\lambda(n)=\kappa}=\frac{1+\kappa\lambda(n)}2
 \quad(n\ge1)
\end{equation}
give their sign formulas, with the actual eligible counts.
Section~\ref{sec:progressions} proves B with its own relative
measure and uses the same packing for its instance of C.
\end{proof}

\end{corollary}

\section{Liouville versions}\label{sec:liouville}
The function $\lambda(n)=(-1)^{\Omega(n)}$ is completely
multiplicative and takes values in $\{-1,1\}$ for every $n\ge1$.

from decomposing the prime variable.
For the shifted target define explicitly
\begin{equation}\label{eq:F-lambda}
 \begin{split}
 F_{\lambda,G}(n)&=\begin{cases}
 \lambda(\GN-n)W(\GN-n),&1\le n<\GN,\\0,&\text{otherwise},\end{cases}\\
 F_{\lambda,h}(n)&=\begin{cases}
 \lambda(n+h)W(n+h),&n\ge1,\ n+h\ge1,\\0,&\text{otherwise}.
 \end{cases}
 \end{split}
\end{equation}

For $\kappa=\pm1$, the three statements of
Theorem~\ref{thm:lambda} are explicitly
\begin{align}
 \sum_{p<\GN}\lambda(\GN-p)&=o(\pi(\GN)),&
 \#\{p<\GN:\lambda(\GN-p)=\kappa\}
       &=\left(\tfrac12+o(1)\right)\pi(\GN),\label{eq:Glambda}\\
 \sum_{\substack{p\le x\\p+h\ge1}}\lambda(p+h)&=o(\pi(x)),&
 \#\{p\le x:p+h\ge1,\ \lambda(p+h)=\kappa\}
       &=\left(\tfrac12+o(1)\right)\pi(x).\label{eq:Hlambda}
\end{align}
The former holds for all even $\GN$ under the hypotheses of
Theorem~\ref{thm:G}, and for the fixed-$C$ small-radical class
under those of Theorem~\ref{thm:S}; the latter is uniform for

\subsection{Expansion of the weight}\label{subsec:lambda-expansion}
The exact identities are
\begin{equation}\label{eq:lambda-weight-expansion}
 \begin{split}
 \omega_i(n)\rho^{\omega_i(n)-1}
    &=\sum_{\substack{p\in\mathcal P_i\\p\mid n}}\rho^{\omega_i(n)-1},\\
 W(n)&=\frac1{\prod_iV_i}\sum_{\substack{a\in\Aset\\a\mid n}}
                                \rho^{\omega(n)-K},\\
 \omega(ah)-K&=\omega(h)-s(a,h),\qquad
 s(a,h)=\#\{i:p_i\mid h\}\in\{0,\ldots,K\},
 \end{split}
\end{equation}
where $\omega_i(ah)=\omega_i(h)+\ind_{p_i\nmid h}$ and
$s(a,h)=0$ precisely when $(a,h)=1$.

\begin{lemma}[Liouville expansion]\label{lem:lambda-expansion}
Let $w$ denote $mn+2$, $\GN-mn$ or $mn+h$, with the positive
argument convention used above and the corresponding coefficients.
Then
\begin{equation}\label{eq:lambda-expansion}
 \begin{split}
 \sum_{\substack{m,n\\w\ge1}}\alpha_m\beta_n\lambda(w)W(w)
 ={}&\frac{(-1)^K}{\prod_iV_i}\sum_{\ell\ge1}\lambda(\ell)
                  \rho^{\omega(\ell)}
 \sum_{\substack{a\in\Aset,m,n\\w=a\ell,(a,\ell)=1}}\alpha_m\beta_n\\
 &+O_{K,\rho}(XL e^{-L^{0.1}}).
 \end{split}
\end{equation}
For a fixed positive $w$, the discrepancy before multiplication by
$\alpha_m\beta_n$ is exactly
\begin{equation}\label{eq:lambda-discrepancy}
 \frac{\lambda(w)}{\prod_iV_i}
 \sum_{\substack{a\in\Aset,a\mid w\\s(a,w/a)\ge1}}
                              \rho^{\omega(w)-K}.
\end{equation}
It lies in $\lambda(w)[0,W(w)]$; every summand weight is at most $1$.
\end{lemma}
\begin{proof}
For $a\in\Aset$, $\lambda(a)=(-1)^K$, and complete
multiplicativity gives $\lambda(w)=(-1)^K\lambda(w/a)$ for every
$a\mid w$, with no coprimality assumption.
On $(a,w/a)=1$, \eqref{eq:lambda-weight-expansion} gives exactly
the main term. The omitted summands are precisely
\eqref{eq:lambda-discrepancy}.
They have some $p_i^2\mid w$, and $\omega(w)\ge K$ makes their
weights at most $1$. Their absolute value is bounded, with constant
$1$, by
\[
 \frac1{\prod_iV_i}\sum_{\substack{a\in\Aset,a\mid w\\(a,w/a)>1}}1.
\]
Since $W(w)\ll_{K,\rho}1$, the absolute discrepancy is at most
\[
 C_{K,\rho}\sum_{p\in\cup_i\mathcal P_i}\ind_{p^2\mid w}.
\]
For fixed $m$ on the coefficient support, $(m,p)=1$ and the
condition $p^2\mid w$ selects one class of $n$ modulo $p^2$.
Since $N\gg p^2$, its count is at most $N/p^2+1\ll N/p^2$.
Summing with $|\alpha_m|\le1$, $|\beta_n|\ll L$ and
$\sum_{p\in\cup_i\mathcal P_i}p^{-2}\ll e^{-L^{0.1}}$
gives the claimed error for all three positive shifted arguments.
This is the bounded-weight square-divisibility argument of
LWW (4.3), p.~15.

If the restriction $(a,\ell)=1$ is then removed before Cauchy,
the added terms have $s\ge1$ and
$\rho^{\omega(\ell)}\le\rho^{\omega(w)-K}$, by
\eqref{eq:lambda-weight-expansion}. They are bounded by the same
count, at an additional fixed constant.
Cauchy uses only $|\lambda(\ell)\rho^{\omega(\ell)}|\le1$.
For the Goldbach shift use $Q_Y^+\le\QQ$ in
\eqref{eq:cauchy-all}; for small shifts use \eqref{v2:3-4}.
In each case the resulting square contains no target function.
\end{proof}

\subsection{Bombieri--Vinogradov for Liouville}\label{subsec:lambda-BV}
The M\"obius input is LWW (3.1), p.~10: for every $A>0$ there
exists $B=B(A)$ such that
\begin{equation}\label{eq:BVmu}
 \sum_{q\le x^{1/2}L^{-B}}
 \max_{(c,q)=1}\max_{y\le x}
 \left|\sum_{\substack{n\le y\\n\equiv c\ (q)}}\mu(n)\right|
 \ll_A xL^{-A}.
\end{equation}
\begin{lemma}[Liouville distribution]\label{lem:BVlambda}
For every $A>0$ there is $B'(A)$ such that
\begin{equation}\label{eq:BVlambda}
 \sum_{q\le x^{1/2}L^{-B'(A)}}
 \max_{(c,q)=1}\max_{y\le x}
 \left|\sum_{\substack{n\le y\\n\equiv c\ (q)}}\lambda(n)\right|
 \ll_A xL^{-A}.
\end{equation}
One may take $B'(A)=\max(B(A),A)+A+1$.
\end{lemma}
\begin{proof}
The multiplicative identity
\begin{equation}\label{eq:lambda-mu-convolution}
 \lambda(n)=\sum_{d^2\mid n}\mu(n/d^2)
\end{equation}
holds prime by prime: at $p^k$ the sum is $1$ for even $k$ and
$-1$ for odd $k$.
For $(c,q)=1$,
\[
 \sum_{\substack{n\le y\\n\equiv c\ (q)}}\lambda(n)
 =\sum_{d\le\sqrt y}\sum_{\substack{m\le y/d^2\\d^2m\equiv c\ (q)}}\mu(m).
\]
There is no solution when $(d,q)>1$; otherwise the class is
$m\equiv c\bar d^{\,2}\pmod q$, which is reduced.
Put $D_0=L^{A+1}$ and $Q=x^{1/2}L^{-B'}$.
For $d\le D_0$, use the discrepancy
\[
 E_\mu(x/d^2;q)=\max_{(c',q)=1}\max_{z\le x/d^2}
 \left|\sum_{\substack{m\le z\\m\equiv c'\ (q)}}\mu(m)\right|.
\]
For $d>D_0$ use $y/(d^2q)+1$.
The tail summed over $q\le Q$ is
\begin{equation}\label{eq:lambda-BV-tail}
 \ll xL/D_0+Q\sqrt x\le xL^{-A}+xL^{-B'}.
\end{equation}
For the head, $x_d=x/d^2\ge xL^{-2A-2}$ and
$\log x_d\ge L/2$. Condition $B'\ge B(A)+A+1$ ensures
$Q\le\sqrt{x_d}(\log x_d)^{-B(A)}$.
Apply \eqref{eq:BVmu} at $x_d$ and sum
$\sum_{d\le D_0}xd^{-2}L^{-A}\ll_A xL^{-A}$.
The stated $B'(A)$ also absorbs the tail.
\end{proof}

\begin{lemma}[Weighted Liouville distribution]\label{lem:BVlambda-weighted}
Put
\[
 E_\lambda(Y;m)=\max_{(c,m)=1}\max_{z\le Y}
 \left|\sum_{\substack{t\le z\\t\equiv c\ (m)}}\lambda(t)\right|.
\]
For fixed $A_{\rm BV}>2A+4$ and
$Q\le Y^{1/2}L^{-B'(A_{\rm BV})}$,
\begin{equation}\label{eq:BVlambda-weighted}
 \sum_{m\le Q}\tau(m)E_\lambda(Y;m)\ll_A YL^{-A}.
\end{equation}
Here $\log Y\asymp L$ in the applications, with fixed comparison
constants and a corresponding fixed precision margin.
\end{lemma}
\begin{proof}
The trivial bound is $E_\lambda(Y;m)\ll Y/m+1$.
As in the Cauchy step of LWW Section~3, p.~13,
\begin{equation}\label{eq:BVlambda-Cauchy}
 \left(\sum\tau E\right)^2
 \le\left(\sum E\right)\left(\sum\tau^2E\right),\quad
 \sum E\ll YL^{-A_{\rm BV}},\quad
 \sum\tau^2E\ll YL^4+QL^3.
\end{equation}
Use Lemma~\ref{lem:BVlambda} at precision $A_{\rm BV}$, rather
than $A$, and take square roots.
\end{proof}

\section{Progressions}\label{sec:progressions}
\begin{remark}[Changes in version 2: progression restriction]
This section adds the prime-variable character to the existing
determinant and analytic arguments. The local density was proved
in Section~\ref{subsec:progression-density}. Restriction to a
progression also requires the relative moment and logarithm-removal

\end{remark}
\begin{proof}[Proof of Theorem~\ref{thm:progression}]
\subsection*{The prime-variable character and Type I}
Let $\upsilon$ be any Dirichlet character modulo $q$. We first prove the marked estimate with the factor $\upsilon(p)$ at arbitrary fixed logarithmic precision, uniformly over $q,\upsilon$. Fix a sufficiently large low-prime cutoff exponent $B$. At the Vaughan level, $\upsilon(mn)=\upsilon(m)\upsilon(n)$, so replace the two coefficient sequences by $\alpha_m\upsilon(m)$ and $\beta_n\upsilon(n)$. Their absolute coefficient bounds are unchanged. Since $q<Z$ eventually, every sifted variable is already coprime to $q$.

The affine cofactor identity and Cauchy square are unchanged: the extra character factors belong to the coefficients, while the bounded target on the cofactor is still $f(\ell)$ times its damping. The signed root counting, labelled intermediate space, endpoint primitivity, comparison kernel, memory moments, goodness removal and minor-arc error in Section~\ref{subsec:affine-transport} apply to these coefficients with the same bounds. In particular no congruence modulo $q$ is added to the root-counting modulus.

For Type I the outer character factor has modulus at most one. To make the inner character completely explicit, in the notation of Section~\ref{subsec:general-I} one has $\varepsilon q_1rj+h=Dgbt$ (rename its outer variable $q_1$ to avoid the present modulus $q$). Thus the prime-variable character is


\begin{equation}
\upsilon(q_1rj)=\upsilon\big(\varepsilon(Dgbt-h)\big).
\end{equation}


Partition $t$ into its residue classes modulo $q$. On each it is constant and bounded. The reduced M\"obius/Liouville class modulo $2q_1r/g$ is intersected with one residue modulo $q$. These latter classes need not be reduced in $t$, so one must not invoke reduced-class BV on them directly. Instead, an equivalent simpler treatment is to partition the original inner variable $k$ modulo $q$ before the transformation. Since the sieve already gives $(q_1k,q)=1$, all these are unit classes for $k$, but the resulting target class can still be nonunit. To handle that case, absorb the fixed prime-power part forced by the modulus into the target using the same local-factor identity \eqref{v2:6-6}, as detailed next.

For a residue $t\equiv c\pmod q$, put $v=(c,q)$, $q'=q/v$, and write $t=v t_1$. Then $t_1\equiv c/v\pmod{q'}$ is reduced, but $(v,t_1)$ may be nontrivial. For M\"obius use the general identity


\begin{equation}\label{v2:14-2}
\mu(vt_1)=\begin{cases}
\mu(v)\displaystyle\sum_{\substack{b_1\mid t_1\\p\mid b_1\Rightarrow p\mid v}}\mu(t_1/b_1),&v\text{ squarefree},\\
0,&v\text{ not squarefree}.
\end{cases}
\end{equation}


If a prime divides both $b_1$ and $q'$, the original reduced class of $t_1$ is impossible, so only $(b_1,q')=1$ survives. The other original modulus $m=2q_1r/g$ is already reduced in $t$, hence $(v,m)=1$ is necessary, and any prime of $b_1$ dividing $m$ likewise gives an impossible term. On the surviving terms the new variable $t_2=t_1/b_1$ satisfies two compatible reduced classes, hence one reduced class modulo $\operatorname{lcm}(m,q')\le qm$. The numerator changes by inverses of $v,b_1$, which are units wherever needed. For Liouville simply use $\lambda(vt_1)=\lambda(v)\lambda(t_1)$, with no $b_1$ expansion.

All extra finite choices and multiplicities cost only fixed powers of $L$: there are $q\le L^{C_q}$ residue classes, $v\le q$, and at most $q$ preimages of a modulus under $m\mapsto\operatorname{lcm}(m,q')$. To justify the latter bound prime by prime, let e be the exponent in q'. If the fixed lcm has larger exponent, the exponent in m is forced; if it equals e, the exponent in m has at most e+1 choices. Thus there are at most $\tau(q')\le q'$ possible m. The small-divisor BV margin in Section~\ref{subsec:general-I} remains fixed after these logarithmic factors and replacing its divisor cutoff by $Dgbvb_1\le y^{1/20}$. The extra reciprocal sum over $b_1$ is at most $\prod_{p\mid q}(1-1/p)^{-1}\ll\log L$.

For the complementary smooth tail, $v\le L^{C_q}$ and all primes of $b_1$ divide $q$; the set of primes supporting $bb_1$ has $O_{K,A,C_q}(\log L)$ elements. Apply the same Rankin argument as in Section~\ref{subsec:general-I} to the combined smooth factor $bb_1$. Its convolution multiplicity contributes the square of the corresponding Euler product, still a fixed power of L at each fixed Rankin exponent. Thus the power-saving tail and endpoint estimates survive. Odd h uses the already separated factor 2 from Section~\ref{subsec:general-I} before this residue decomposition. This proves the uniformly twisted Type I estimate. The additional residue and divisor costs are absorbed by raising its fixed BV and truncation precisions, without changing the Type II coefficient budget.

\subsection*{The two character locations in the major arcs}
The finite fibre expansion and Mellin kernel are unchanged. Its characters now have the form


\begin{equation}\label{v2:14-3}
\xi=\chi\psi\upsilon\quad\hbox{on }m,n,\qquad
\xi_b=\chi\psi\psi'_b\quad\hbox{on the marking product }b.
\end{equation}


All are regarded on the common modulus $\operatorname{lcm}(kH,q)$. The prime-variable character $\upsilon$ occurs only in the two long polynomials. Choose a fixed $A_1>A_0+C_q+2$. If $\xi_b$ were principal, then $\chi\psi=(\psi'_b)^{-1}$ and $\xi=\upsilon(\psi'_b)^{-1}$ would have conductor dividing $\operatorname{lcm}(q,k)$, at most $qk\le L^{A_0+C_q}$. Therefore every $\xi$ above the chosen cutoff has nonprincipal $\xi_b$, and $\mathrm H1$ gives the same short-prime saving.

The hybrid mean value still averages over $\chi\pmod H$; the fixed additional character $\upsilon$ is absorbed into the coefficients and has modulus at most one. Orthogonality and the logarithmic spacing H/X are unchanged. In the small-conductor range the primitive inducer of $\xi$ determines $\chi$ for fixed $\psi,\upsilon$, so there are still at most $L^{2A_1+1}$ characters. The small-conductor M\"obius bound applies at the raised fixed cutoff. Primes dividing q cause no extra loss since they are below Z; primes dividing H are removed exactly as in Lemma~\ref{input:sifted}. The sparse-frequency estimate accepts arbitrary bounded coefficients in the long polynomial, including $\upsilon$, and is uniform in the marking character. Thus \eqref{v2:5-14}–\eqref{v2:5-15} retain their bounds.

Combining Type I and Type II gives, at every fixed precision A,


\begin{equation}
\sum_{p\in\mathcal P_{\varepsilon,h}(x)}(\log p)f(\varepsilon p+h)W_j(\varepsilon p+h)\upsilon(p)
\ll_{A,C,C_q}xL^{-A}.
\end{equation}


Character orthogonality modulo q and the factor $1/\varphi(q)$ then give the same bound for each marked progression sum. The uniform band-family proof applies unchanged after the fixed cutoff and precisions are raised; no parameter grows with q beyond its fixed logarithmic exponent.

\subsection*{Progression-relative weight removal}
Let $I$ be the eligible prime interval and put $\Theta_{I;q,a_0}=\sum_{p\in I,\,p\equiv a_0(q)}\log p$. For $F=1,W_j,W_jW_k$ with $j\ne k$, the finite Taylor expansion and CRT give


\begin{equation}\label{v2:14-4}
\sum_{\substack{p\in I\\p\equiv a_0\ (q)}}(\log p)F(\varepsilon p+h)
=m_F\Theta_{I;q,a_0}+O_A(xL^{-A}).
\end{equation}


Indeed every band prime exceeds $q$ and is coprime to $h$, so each retained band divisor $d$ is coprime to $q$. Combining its reduced shifted class with $a_0\pmod q$ gives one reduced class modulo $qd$. Different $d$ give different $qd$, all $x^{o(1)}$, so the summed BV discrepancies retain arbitrary fixed precision. The same positive Taylor remainder bound applies. The local probabilities at band primes remain $1/(r-1)$; hence the means $m_F$ are exactly those in \eqref{v2:6-10} and factor for disjoint copies. Endpoint-uniform Siegel–Walfisz gives $\Theta_{I;q,a_0}\ll x/\varphi(q)$.

Using the $b_x$ copies in Section~\ref{subsec:packing}, \eqref{v2:14-4} yields variance at most $C\Theta_{I;q,a_0}/b_x+O_A(xL^{-A})$. Cauchy with this progression measure and the uniformly small marked progression sums bounds the unmarked weighted sum by


\begin{equation}
\frac{Cx}{\varphi(q)\sqrt{\log L}}+O_A(xL^{-A})
+O_A\left(\sqrt{\frac{x}{\varphi(q)}\,xL^{-A}}\right).
\end{equation}


Choose fixed $A>C_q+4$. Both errors are absorbed by the first term. To remove $\log p$, use the progression bound $\pi(x;q,a_0)\ll x/(\varphi(q)L)$ for the high primes; this gives a relative error $O_B(\log L/L)$. The low-prime segment contributes at most $x/L^B$, a relative error $O(L^{C_q+1-B})$, so take fixed $B>C_q+4$. Using only the ambient Chebyshev bound here would fail when $C_q$ is large. This proves \eqref{eq:progression-main}. Finally choose the precision in \eqref{v2:13-1} larger than $C_q+3$ and use the exact sign identities, proving the stated sign counts. 

\end{proof}
The modulus range is $q\le(\log x)^{C_q}$ with fixed $C_q$.
